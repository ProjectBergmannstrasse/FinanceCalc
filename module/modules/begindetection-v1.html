function beginDetection(){
        var tries=0, told=false, loggedFace=false, blankFrames=0;
        var live=makeLivenessTracker();
        var tick=function(){
          if(!_autoFace.stream || !document.getElementById("lockScreen")){ _autoFace.stop(); return; }
          faceapi.detectSingleFace(video, new faceapi.TinyFaceDetectorOptions())
            .withFaceLandmarks(true).withFaceDescriptor()
            .then(function(det){
              if(!_autoFace.stream) return;
              if(!det){ blankFrames++; }
              if(!det && blankFrames===25){
                // 25 frames with nothing at all usually means no picture,
                // not a shy user — stop implying it's still looking
                setStatus && setStatus("Can\u2019t see anything \u2014 enter your PIN");
              }
              if(!det && blankFrames>=70){
                _autoFace.stop();
                setStatus && setStatus("No face seen \u2014 enter your PIN");
                return;
              }
              if(det){
                blankFrames=0;
                live.feed(det, video);
                var who=matchFace(Array.from(det.descriptor));
                if(who){
                  /* A matching descriptor is NOT enough. A printed photo or a
                     phone screen produces the same descriptor as the real
                     person — liveness is what separates them. */
                  var v=live.verdict();
                  if(v.state==="live"){
                    _autoFace.stop();
                    setStatus && setStatus("Welcome back, "+who.name);
                    setTimeout(function(){ onMatch(who); }, 350);
                    return;
                  }
                  if(v.state==="spoof"){
                    if(!loggedFace){
                      loggedFace=true;
                      recordFailedAttempt("spoof");
                      captureAttempt(video, "spoof");
                    }
                    setStatus && setStatus(v.why);
                    _autoFace.raf=setTimeout(tick, 700);
                    return;
                  }
                  // the randomised prompt, shown so they know what to do
                  setStatus && setStatus(v.state==="challenge"
                    ? (live.prompt() || "Blink or turn your head slightly")
                    : "Checking it\u2019s really you\u2026");
                  _autoFace.raf=setTimeout(tick, 220);
                  return;
                }
                /* An unrecognised face IS the signal worth recording. Give it a
                   few frames first — a bad angle or poor light shouldn't count as
                   someone else — then log it once per lock screen, not per frame. */
                tries++;
                if(tries===8 && !told){ told=true; setStatus && setStatus("Not recognised \u2014 enter your PIN"); }
                if(tries===12 && !loggedFace){
                  loggedFace=true;
                  recordFailedAttempt("face");
                  captureAttempt(video, "face");
                }
              }
              // no timeout: keep watching for as long as this screen is open
              _autoFace.raf=setTimeout(tick, 450);
            })
            .catch(function(){ _autoFace.raf=setTimeout(tick, 900); });
        };
        tick();
    }
