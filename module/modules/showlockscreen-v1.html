function showLockScreen(){
    if(!lockIsOn()) return;
    var ov=document.createElement("div");
    ov.id="lockScreen";
    ov.innerHTML='<div class="lk-box">'
      +'<div class="lk-icon">\uD83D\uDD12</div>'
      +'<div id="lkPadHost"></div>'
      +'</div>';
    document.body.appendChild(ov);
    document.body.style.overflow="hidden";

    function unlock(){
      _autoFace.stop();
      if(pad) pad.dispose();
      setTimeout(function(){
        // These both opened an overlay at the same moment, so the plain-text
        // banner landed on top of the one with the photos — which is why a
        // face failure looked like it had no image. One screen now.
        if(unseenShots().length) showIntrusionReview();
        else showAttemptBanner();
      }, 700);
      ov.style.opacity="0";
      document.body.style.overflow="";
      document.removeEventListener("visibilitychange", onHide);
      setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); }, 220);
    }

    // extras below the pad: face unlock and the recovery route
    /* No "use face unlock" button. If a face is enrolled the camera just
       looks for you while this screen is open — a button to start something
       that should be automatic was the wrong design. */
    var extra="";
    if(recoveryOn())
      extra+='<button class="kp-link" id="lkForgot">Forgot your PIN?</button>';

    var pad=buildKeypad({
      title:"My Money",
      sub:"Enter your PIN",
      len:4, max:12,
      wrongText:"Wrong PIN \u2014 try again",
      extra:extra,
      validate:function(d, prog){
        var cd=lockCooldownLeft();
        if(cd>0) return Promise.resolve(false);
        return new Promise(function(res){
          verifyPin(d, function(ok){
            if(!ok){
              recordFailedAttempt("pin");
              var left=lockCooldownLeft();
              if(left>0) setTimeout(function(){ pad.say("Too many tries \u2014 wait "+left+"s"); }, 460);
            }
            res(ok);
          }, prog);
        });
      },
      onDone:unlock
    });
    document.getElementById("lkPadHost").appendChild(pad.el);

    var fg=pad.el.querySelector("#lkForgot");
    if(fg) fg.addEventListener("click", function(){ openRecoveryUnlock(function(){ unlock(); }); });

    // look for the owner automatically, only while this screen is up
    startAutoFace(ov, unlock, function(msg){ pad.say(msg, /Welcome/.test(msg)?"ok":""); });

    /* The camera must never outlive this screen. Three independent triggers,
       because one missed path leaves the light on: leaving the tab, the
       element being removed by anything at all, and page unload. */
    var onHide=function(){ if(document.hidden) _autoFace.stop(); else if(document.getElementById("lockScreen")) startAutoFace(ov, unlock, function(m){ pad.say(m, /Welcome/.test(m)?"ok":""); }); };
    document.addEventListener("visibilitychange", onHide);
    window.addEventListener("pagehide", function(){ _autoFace.stop(); });
    try{
      new MutationObserver(function(_,obs){
        if(!document.getElementById("lockScreen")){
          _autoFace.stop();
          document.removeEventListener("visibilitychange", onHide);
          obs.disconnect();
        }
      }).observe(document.body,{childList:true});
    }catch(e){}
  }
