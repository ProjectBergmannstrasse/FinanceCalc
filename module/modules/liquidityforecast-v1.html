function liquidityForecast(ahead){
    var s=monthlySeries();
    if(s.months.length<4) return {ok:false, months:s.months.length};
    var inM=fitSeries(s.inflow, 12), outM=fitSeries(s.outflow, 12);
    if(!inM||!outM) return {ok:false, months:s.months.length};
    var fin=forecastSeries(inM, ahead), fout=forecastSeries(outM, ahead);
    var t=totals();
    var bal=t.balance+t.savings;
    var path=[], worst=bal, worstPath=[], dryMonth=null, dryWorst=null;
    var wbal=bal;
    for(var i=0;i<ahead;i++){
      var net=fin[i].mid-fout[i].mid;
      bal+=net; path.push({h:i+1, bal:bal, net:net});
      // pessimistic: low income, high spending
      wbal += (fin[i].lo - fout[i].hi);
      worstPath.push({h:i+1, bal:wbal});
      if(dryMonth===null && bal<0) dryMonth=i+1;
      if(dryWorst===null && wbal<0) dryWorst=i+1;
    }
    // seasonal months worth flagging
    var peak=0, trough=0;
    outM.seasonal.forEach(function(v,idx){
      if(v>outM.seasonal[peak]) peak=idx;
      if(v<outM.seasonal[trough]) trough=idx;
    });
    return {ok:true, path:path, worstPath:worstPath, start:t.balance+t.savings,
            dryMonth:dryMonth, dryWorst:dryWorst,
            inModel:inM, outModel:outM, series:s,
            peakIdx:peak, troughIdx:trough,
            peakAmt:outM.seasonal[peak], troughAmt:outM.seasonal[trough],
            fin:fin, fout:fout};
  }
