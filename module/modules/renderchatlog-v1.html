function renderChatLog(){
    var box=document.getElementById("chatLog"); if(!box) return;
    if(!chatLog.length){
      box.innerHTML='<div class="chat-empty">'
        +'<div class="ce-ic">\uD83D\uDCAC</div>'
        +'<div class="ce-t">Ask about your money</div>'
        +'<div class="ce-d">Everything stays on this device. Attach a statement or receipt if it helps.</div>'
        +'<div class="chat-chips">'
        + CHAT_SUGGESTIONS.map(function(q){
            return '<button class="chat-sugg" data-q="'+esc(q)+'">'+esc(q)+'</button>';
          }).join("")
        +'</div></div>';
      box.querySelectorAll(".chat-sugg").forEach(function(b){
        b.addEventListener("click", function(){
          var inp=document.getElementById("chatInput");
          if(inp){ inp.value=b.getAttribute("data-q"); if(window._chatSync) window._chatSync(); }
          chatSend();
        });
      });
      return;
    }
    box.innerHTML=chatLog.map(function(m){
      var atts=(m.files&&m.files.length)
        ? '<div class="chat-att">'+m.files.map(function(f){
            return '<span>'+chatFileIcon(f.ext)+" "+esc(f.name)+'</span>'; }).join("")+'</div>'
        : "";
      if(m.pending){
        return '<div class="chat-turn ai"><div class="chat-who">AI</div>'
          +'<div class="chat-typing"><i></i><i></i><i></i></div></div>';
      }
      return '<div class="chat-turn '+(m.role==="user"?"me":"ai")+'">'
        +'<div class="chat-who">'+(m.role==="user"?"You":"AI")+'</div>'
        + atts
        +'<div class="chat-bub">'+esc(m.text)+'</div></div>';
    }).join("");
    box.scrollTop=box.scrollHeight;
  }
