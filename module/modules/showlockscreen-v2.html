function showLockScreen(){
    if(!lockIsOn()) return;
    var ov=document.createElement("div");
    ov.id="lockScreen";
    ov.innerHTML='<div class="lk-box">'
      +'<div class="lk-icon">\uD83D\uDD12</div>'
      +'<div id="lkPadHost"></div>'
      +'</div>';
    document.body.appendChild(ov);
    document.body.style.overflow="hidden";

    function unlock(){
      _autoFace.stop();
      if(pad) pad.dispose();
      setTimeout(showAttemptBanner, 700);
      ov.style.opacity="0";
      document.body.style.overflow="";
      document.removeEventListener("visibilitychange", onHide);
      setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); }, 220);
    }

    // extras below the pad: face unlock and the recovery route
    var extra="";
    if(faceCfg.on && facePeople().length)
      extra+='<button class="btn ghost sm" id="lkBio">\uD83D\uDE00 Use face unlock</button>';
    if(recoveryOn())
      extra+='<button class="kp-link" id="lkForgot">Forgot your PIN?</button>';

    var pad=buildKeypad({
      title:"My Money",
      sub:"Enter your PIN",
      len:4, max:12,
      wrongText:"Wrong PIN \u2014 try again",
      extra:extra,
      validate:function(d, prog){
        var cd=lockCooldownLeft();
        if(cd>0) return Promise.resolve(false);
        return new Promise(function(res){
          verifyPin(d, function(ok){
            if(!ok){
              recordFailedAttempt("pin");
              var left=lockCooldownLeft();
              if(left>0) setTimeout(function(){ pad.say("Too many tries \u2014 wait "+left+"s"); }, 460);
            }
            res(ok);
          }, prog);
        });
      },
      onDone:unlock
    });
    document.getElementById("lkPadHost").appendChild(pad.el);

    var bio=pad.el.querySelector("#lkBio");
    if(bio) bio.addEventListener("click", function(){
      if(inFrame() && !window.faceapi){
        pad.say("Face unlock needs the app in its own tab"); return;
      }
      pad.say("Opening the camera\u2026");
      bio.disabled=true;
      var answered=false;
      var settle=function(ok){
        if(answered) return; answered=true; bio.disabled=false;
        if(ok) unlock();
        else { recordFailedAttempt("face"); pad.say("Face didn\u2019t match \u2014 use your PIN"); }
      };
      setTimeout(function(){
        if(answered) return; answered=true; bio.disabled=false;
        var sh=document.getElementById("faceOv");
        if(sh && sh.parentNode) sh.parentNode.removeChild(sh);
        pad.say("Camera didn\u2019t start \u2014 use your PIN");
      }, 15000);
      verifyFace(settle);
    });
    var fg=pad.el.querySelector("#lkForgot");
    if(fg) fg.addEventListener("click", function(){ openRecoveryUnlock(function(){ unlock(); }); });

    // look for the owner automatically, only while this screen is up
    startAutoFace(ov, unlock, function(msg){ pad.say(msg, /Welcome/.test(msg)?"ok":""); });
    var onHide=function(){ if(document.hidden) _autoFace.stop(); };
    document.addEventListener("visibilitychange", onHide);
  }
