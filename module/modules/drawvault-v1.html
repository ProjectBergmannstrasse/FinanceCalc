function drawVault(){
    var list=vaultTier();
    if(!list.length)
      return '<div class="hs-empty"><div class="he-ic">\uD83D\uDDC4\uFE0F</div>'
        +'<div class="he-t">The vault is empty</div>'
        +'<div class="he-d">Deleted entries are kept here even after<br>the normal bin is emptied.</div></div>';
    var html='<div class="rc-secd" style="margin-bottom:10px">Kept even after the bin is emptied. '
      +'Large deletions are marked, because those are the ones worth a second look.</div>';
    var lastDay="";
    list.forEach(function(b,i){
      var day=dayHeading(b.at||Date.now());
      if(day!==lastDay){ html+='<div class="hs-day">deleted '+esc(day)+'</div>'; lastDay=day; }
      var sum=b.items.reduce(function(a,t){ return a+Math.abs(num(t.amount)); },0);
      var names=b.items.slice(0,2).map(function(t){ return t.note||t.cat||"entry"; }).join(", ");
      if(b.items.length>2) names+=" +"+(b.items.length-2)+" more";
      html+='<div class="hs-card" style="animation-delay:'+Math.min(i*22,220)+'ms">'
        +'<div class="hs-ic del">'+(b.mass?"\u26A0\uFE0F":"\uD83D\uDDD1\uFE0F")+'</div>'
        +'<div class="hs-main"><div class="hs-t">'+esc(names)+'</div>'
        +'<div class="hs-sub"><span>'+b.items.length+' entr'+(b.items.length===1?"y":"ies")+'</span>'
        +'<span class="hs-dot"></span><span>'+esc(agoText(b.at))+'</span>'
        +(b.mass?'<span class="hs-pill soon">large deletion</span>':'')
        +'</div></div>'
        +'<div class="hs-amt">'+fmt(sum,state.currency)+'</div>'
        +'<button class="hs-btn" data-restore="'+b.id+'" data-tier="deep">Restore</button>'
        +'</div>';
    });
    return html;
  }
