function mtOcrUpscale(dataUrl, cb){
    var img=new Image();
    img.onload=function(){
      var w=img.naturalWidth,h=img.naturalHeight;
      if(!w||!h){cb(dataUrl,1);return;}
      var sc=w<900?Math.min(3,Math.max(1,Math.ceil(900/w))):1;
      if(sc===1){cb(dataUrl,1);return;}
      try{
        var c=document.createElement("canvas");
        c.width=w*sc;c.height=h*sc;
        var x=c.getContext("2d");
        x.imageSmoothingEnabled=true;x.imageSmoothingQuality="high";
        x.drawImage(img,0,0,c.width,c.height);
        cb(c,sc);
      }catch(e){cb(dataUrl,1);}
    };
    img.onerror=function(){cb(dataUrl,1);};
    img.src=dataUrl;
  }
