function mtOcrStart(){
    if(mtOcrMode()==="off"||!mtOcrIsTradeType()) return;
    var imgs=mtOcrImages();
    if(!imgs.length){mtOcr.state="idle";mtOcr.result=null;mtOcrUI();return;}
    var myToken=++mtOcr.token;
    mtOcr.state="loading";mtOcr.progress=0;mtOcr.result=null;mtOcr.error=null;mtOcrUI();
    mtOcrLoadEngine(function(ok){
      if(myToken!==mtOcr.token) return;
      if(!ok){
        mtOcr.state="failed";
        mtOcr.error=navigator.onLine?"The reader couldn't load — try again.":"You're offline — the reader needs a connection once, then it's cached.";
        mtOcrUI();return;
      }
      mtOcr.state="running";mtOcrUI();
      var allTrades=[], idx=0, per=1/imgs.length;
      function next(){
        if(myToken!==mtOcr.token) return;
        if(idx>=imgs.length){
          mtOcr.result={trades:allTrades,isStatement:allTrades.length>0};
          mtOcr.state=allTrades.length?"done":"nota";
          mtOcrUI();
          if(mtOcr.state==="done"&&(mtOcrMode()==="auto"||mtOcr.wantFill)){mtOcr.wantFill=false;mtOcrFill();}
          if(mtOcr.state==="nota") mtOcr.wantFill=false;
          return;
        }
        var pp=imgs[idx], n=idx;
        mtOcrUpscale(pp.dataUrl,function(input,sc){
          if(myToken!==mtOcr.token) return;
          Tesseract.recognize(input,"eng",{logger:function(m){
            if(m.status==="recognizing text"&&myToken===mtOcr.token){
              mtOcr.progress=(n+m.progress)*per;mtOcrUI();
            }
          }}).then(function(res){
            if(myToken!==mtOcr.token) return;
            var words=(res.data.words||[]).map(function(w){
              return{text:w.text,x:w.bbox.x0/sc,y:w.bbox.y0/sc,w:(w.bbox.x1-w.bbox.x0)/sc,h:(w.bbox.y1-w.bbox.y0)/sc};
            }).filter(function(w){return w.text&&w.text.trim().length;});
            var r=mtOcrParse(words);
            // v1.98: file the analyzed image into the sandboxed bucket with its verdict
            mtSandboxSave({ts:Date.now(),name:pp.name||("image_"+(n+1)),dataUrl:pp.dataUrl,
                           verdict:r.trades.length?"accepted":"ignored",
                           tradeCount:r.trades.length,
                           reason:r.trades.length?"parsed as MT4/MT5":"not MT4/MT5 — no instrument + buy/sell and no statement header"});
            allTrades=allTrades.concat(r.trades);
            idx++;next();
          }).catch(function(){ if(myToken===mtOcr.token){idx++;next();} });
        });
      }
      next();
    });
  }
