function start(){
      if(inFrame() && !_faceApiReady && !window.faceapi){
        blocked.style.display=""; video.style.display="none";
        stepEl.textContent="Not available in a frame";
        stepEl.style.color="var(--amber)";
        hintEl.textContent="The face model can't download inside a frame. Open the file in its own tab.";
        fix.style.display="";
        host.querySelector("#enRetry").style.display="none";
        return;
      }
      if(cameraBlocked()){
        blocked.style.display=""; video.style.display="none";
        stepEl.textContent="Camera access is switched off";
        stepEl.style.color="var(--amber)";
        hintEl.textContent="Turn it back on in Settings, or use a photo.";
        fix.style.display="";
        host.querySelector("#enRetry").style.display="none";
        return;
      }
      if(!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia){
        blocked.style.display=""; video.style.display="none";
        stepEl.textContent="No camera available";
        hintEl.textContent="You can still enrol from photos.";
        fix.style.display="";
        host.querySelector("#enRetry").style.display="none";
        return;
      }
      stepEl.style.color=""; blocked.style.display="none"; video.style.display="";
      fix.style.display="none"; help.style.display="none";
      hintEl.textContent="Starting the camera\u2026";
      navigator.mediaDevices.getUserMedia({video:{facingMode:"user",width:480,height:480}})
        .then(function(s){
          stream=s; video.srcObject=s; stopped=false;
          hintEl.textContent="Loading the face model\u2026";
          return loadFaceApi(function(m){ hintEl.textContent=m; });
        })
        .then(function(){
          running=true; startedAt=Date.now();
          stepEl.textContent="Look straight at the camera";
          hintEl.textContent="Then turn your head in a slow circle";
          loop();
        })
        .catch(function(err){
          var m=String(err&&err.name||err);
          if(/NotAllowed|Permission/i.test(m)){ permissionRefused(); return; }
          stepEl.textContent="Couldn't start the camera";
          stepEl.style.color="var(--down)";
          hintEl.textContent=String(err&&err.message||m).slice(0,110);
          fix.style.display="";
        });
    }
