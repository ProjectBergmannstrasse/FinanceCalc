function loadWebllm(){
    var btn=document.getElementById("wllmLoadBtn");
    var prog=document.getElementById("wllmProgress");
    var bar=document.getElementById("wllmBar");
    var sta=document.getElementById("wllmLoadStatus");
    // WebLLM needs its own runtime (~2MB JS) — unavoidable for in-browser LLM
    // The model weights (~660MB for Llama 3.2 1B Q4) are downloaded once and cached
    var SCRIPT_URL="https://esm.run/@mlc-ai/web-llm";
    var MODEL="Llama-3.2-1B-Instruct-q4f16_1-MLC";
    if(!btn||!prog||!bar||!sta) return;
    btn.disabled=true; btn.textContent="Loading runtime…";
    prog.style.display=""; sta.style.display="";
    sta.textContent="Loading WebLLM runtime (~2MB)…";
    // dynamically load the WebLLM runtime — the only external load in the whole app
    var script=document.createElement("script");
    script.type="module";
    script.textContent=[
      'import * as webllm from "'+SCRIPT_URL+'";',
      'const engine=await webllm.CreateMLCEngine("'+MODEL+'",{',
      '  initProgressCallback:function(p){',
      '    var b=document.getElementById("wllmBar");',
      '    var s=document.getElementById("wllmLoadStatus");',
      '    if(b) b.style.width=(p.progress*100).toFixed(0)+"%";',
      '    if(s) s.textContent=p.text||"Loading…";',
      '  }',
      '});',
      'window._wllmEngine=engine;',
      'window._wllmLoaded=true;',
      'document.getElementById("wllmLoadBtn").style.display="none";',
      'document.getElementById("wllmProgress").style.display="none";',
      'document.getElementById("wllmLoadStatus").textContent="Llama 3.2 1B ready (WebGPU)";',
      'document.getElementById("wllmLoadStatus").style.color="var(--up)";',
      'document.getElementById("audWebllmChat").style.display="";',
      'document.getElementById("audEngineLabel").textContent="WebLLM";',
    ].join("\n");
    script.onerror=function(){
      btn.disabled=false; btn.textContent="Load model (660MB, once)";
      prog.style.display="none"; sta.textContent="Failed to load runtime. Check your connection.";
    };
    document.head.appendChild(script);
  }
