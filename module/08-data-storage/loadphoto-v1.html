function loadPhoto(id, cb){
    if(photoCache[id]){ cb(photoCache[id]); return; }
    var handled=false;
    try{
      if(window.storage && typeof window.storage.get==="function"){
        handled=true;
        var p=window.storage.get(photoKey(id), false);
        if(p && p.then){
          p.then(function(r){ if(r && r.value){ photoCache[id]=r.value; cb(r.value); return; } loadPhotoViaManager(id, cb); })
           .catch(function(){ loadPhotoViaManager(id, cb); });
          return;
        }
      }
    }catch(e){}
    if(!handled) loadPhotoViaManager(id, cb);
  }
