function exportWithProgress(getJsonOrBlob, filename, label){
    var ov=document.createElement("div");
    ov.className="exp-ov";
    ov.innerHTML='<div class="exp-card">'
      +'<div class="exp-title">'+esc(label||"Preparing backup")+'</div>'
      +'<div class="exp-track"><div class="exp-fill" id="expFill"></div></div>'
      +'<div class="exp-meta"><span id="expPct">0%</span><span id="expEta">measuring\u2026</span></div>'
      +'</div>';
    document.body.appendChild(ov);
    requestAnimationFrame(function(){ ov.classList.add("in"); });
    var fill=ov.querySelector("#expFill"), pct=ov.querySelector("#expPct"), eta=ov.querySelector("#expEta");
    var t0=performance.now(), done=false;

    function finishWith(blob){
      done=true;
      var total=blob.size||1, start=performance.now();
      // Stream the blob through a reader so the bar reflects real bytes moved, and
      // the ETA is derived from measured MB/s rather than assumed.
      var reader=blob.stream ? blob.stream().getReader() : null;
      if(!reader){
        // older engine: no streams — write straight out, show a quick determinate fill
        fill.style.width="100%"; pct.textContent="100%"; eta.textContent="done";
        downloadBlob(blob, filename);
        setTimeout(close, 500); return;
      }
      var moved=0;
      (function pump(){
        reader.read().then(function(r){
          if(r.done){
            fill.style.width="100%"; pct.textContent="100%"; eta.textContent="done";
            downloadBlob(blob, filename);
            setTimeout(close, 500); return;
          }
          moved += r.value.length;
          var frac=Math.min(1, moved/total);
          var el=(performance.now()-start)/1000;
          var rate=el>0 ? moved/el : 0;                 // bytes/sec, measured
          var remain=rate>0 ? (total-moved)/rate : 0;    // seconds left, from real rate
          fill.style.width=(frac*100).toFixed(1)+"%";
          pct.textContent=Math.round(frac*100)+"%";
          eta.textContent = frac>=1 ? "done"
            : (rate>0 ? ("~"+(remain<1?"<1":Math.ceil(remain))+"s left · "+fmtRate(rate)) : "measuring\u2026");
          pump();
        }).catch(function(){ downloadBlob(blob, filename); close(); });
      })();
    }
    function fmtRate(bps){
      if(bps>1048576) return (bps/1048576).toFixed(1)+" MB/s";
      if(bps>1024) return (bps/1024).toFixed(0)+" KB/s";
      return Math.round(bps)+" B/s";
    }
    function close(){ ov.classList.remove("in"); setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); }, 240); }

    // getJsonOrBlob may be sync-returning or callback-based (exportData is a callback)
    var maybe=getJsonOrBlob(function(result){
      var blob=(result instanceof Blob) ? result : new Blob([result],{type:"application/json"});
      finishWith(blob);
    });
    if(maybe!==undefined && !done){
      var blob=(maybe instanceof Blob) ? maybe : new Blob([maybe],{type:"application/json"});
      finishWith(blob);
    }
  }
