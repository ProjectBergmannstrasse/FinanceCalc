function loadWebllm(opts){
    opts=opts||{};
    var silent=!!opts.silent;                 // true when re-attaching on boot
    var btn=document.getElementById("wllmLoadBtn");
    var prog=document.getElementById("wllmProgress");
    var bar=document.getElementById("wllmBar");
    var sta=document.getElementById("wllmLoadStatus");
    if(window._wllmEngine){ if(opts.onReady) opts.onReady(); return; }
    if(window._wllmLoading){ return; }
    window._wllmLoading=true;

    var cached=wllmRemembered();
    if(btn && !silent){
      btn.disabled=true;
      btn.textContent = cached ? "Reconnecting…" : "Starting…";
    }
    if(prog && !silent) prog.style.display="";
    if(sta && !silent){
      sta.style.display="";
      sta.textContent = cached
        ? "Reconnecting to the model already on this device…"
        : "Fetching the runtime (~2MB)…";
    }
    var SCRIPT_URL="https://esm.run/@mlc-ai/web-llm";
    var script=document.createElement("script");
    script.type="module";
    script.textContent=[
      'import * as webllm from "'+SCRIPT_URL+'";',
      'try{',
      '  const engine=await webllm.CreateMLCEngine("'+WLLM_MODEL+'",{',
      '    initProgressCallback:function(p){',
      '      var b=document.getElementById("wllmBar");',
      '      var s=document.getElementById("wllmLoadStatus");',
      '      if(b) b.style.width=((p.progress||0)*100).toFixed(0)+"%";',
      '      if(s) s.textContent=p.text||"Loading…";',
      '      var cs=document.getElementById("chatSheetStatus");',
      '      if(cs) cs.textContent=(p.text||"loading").slice(0,42);',
      '    }',
      '  });',
      '  window._wllmEngine=engine;',
      '  window._wllmLoading=false;',
      '  window.dispatchEvent(new CustomEvent("wllm-ready"));',
      '}catch(err){',
      '  window._wllmLoading=false;',
      '  window.dispatchEvent(new CustomEvent("wllm-failed",{detail:String(err&&err.message||err)}));',
      '}'
    ].join("\n");
    script.onerror=function(){
      window._wllmLoading=false;
      if(btn){ btn.disabled=false; btn.textContent="Load model (660MB, once)"; }
      if(prog) prog.style.display="none";
      if(sta) sta.textContent="Couldn't reach the runtime — check your connection.";
    };
    document.head.appendChild(script);
  }
