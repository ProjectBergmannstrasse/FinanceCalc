function durabilityReport(){
    return Promise.all([persistState(), isPrivateWindow(), storageEstimate()])
      .then(function(r){
        var persisted=r[0], priv=r[1], est=r[2];
        var usingFolder = (typeof TierFSA!=="undefined" && TierFSA._dirHandle) ? true : false;
        var level, headline, detail, actions=[];
        if(priv){
          level="bad";
          headline="Private window — everything is lost when you close it";
          detail="Nothing written here survives this tab. Export a backup before you finish, or reopen the app in a normal window.";
          actions.push("backup");
        } else if(usingFolder){
          level="good";
          headline="Saving to a folder on your disk";
          detail="A real file, outside the browser's control. Not affected by eviction, Safari's 7-day rule, or clearing browser data.";
        } else if(persisted==="granted"){
          level="ok";
          headline="Storage marked persistent";
          detail="The browser has agreed not to evict this app's data when disk space runs low."
            + (isSafari() ? " Safari still clears script storage after 7 days without a visit, so keep backups." : "");
          if(isSafari()) { level="warn"; actions.push("folder","backup"); } else actions.push("folder");
        } else {
          level="warn";
          headline="Storage is evictable";
          detail = isSafari()
            ? "Safari deletes this app's storage after 7 days without opening it, and may clear it earlier if disk space runs low."
            : "If your disk fills up, the browser can delete this app's data to make room. It picks the sites you haven't opened recently.";
          actions.push("persist","folder","backup");
        }
        var pct=null;
        if(est && est.quota) pct=Math.min(100, Math.round((est.usage||0)/est.quota*100));
        return {level:level, headline:headline, detail:detail, actions:actions,
                persisted:persisted, priv:priv, folder:usingFolder,
                usage:est&&est.usage, quota:est&&est.quota, pct:pct};
      });
  }
