function openAiCompatible(o){
    return {
      label:o.label, model:o.model, keyUrl:o.keyUrl, keyHint:o.keyHint, canList:true,
      base:o.base, extraHeaders:o.extraHeaders||{},
      _headers(key){
        return Object.assign({"Content-Type":"application/json","Authorization":"Bearer "+key}, this.extraHeaders);
      },
      async listModels(key){
        var r=await aiFetch(this.label, this.base+"/models",{headers:this._headers(key)});
        if(!r.ok) throw await aiHttpErr(this.label,r);
        var j=await r.json();
        var list=(j.data||j.models||[]).map(function(m){ return {id:m.id, display:m.name||m.id}; });
        if(o.filter) list=list.filter(function(m){ return o.filter(m.id); });
        return list.sort(function(a,b){ return a.id<b.id?1:-1; });
      },
      async run(key, prompt, images, model){
        var content=[{type:"text",text:prompt}];
        images.forEach(function(im){ content.push({type:"image_url", image_url:{url:"data:"+im.mime+";base64,"+im.b64}}); });
        var body={model:model||this.model, temperature:0, messages:[{role:"user",content:content}]};
        // json mode isn't universal across OpenRouter's catalogue — drop it and retry
        var attempts=[Object.assign({response_format:{type:"json_object"}}, body), body];
        var lastErr=null;
        for(var i=0;i<attempts.length;i++){
          var r=await aiFetch(this.label, this.base+"/chat/completions",{
            method:"POST", headers:this._headers(key), body:JSON.stringify(attempts[i])
          });
          if(!r.ok){ lastErr=await aiHttpErr(this.label,r); continue; }
          var j=await r.json();
          if(j.error) { lastErr=new Error(this.label+": "+(j.error.message||"error")); continue; }
          var c=j.choices&&j.choices[0];
          if(!c) { lastErr=new Error(this.label+" returned no answer."); continue; }
          return c.message.content;
        }
        throw lastErr;
      }
    };
  }
