function renderTagCard(){
    var card=document.getElementById("tagCard");
    if(!card) return;
    var all=tradeList();
    if(!all.length){ card.style.display="none"; return; }
    card.style.display="";
    var untagged=all.filter(function(t){ return !t.inst; }).length;
    document.getElementById("tagCount").textContent=all.length+" trade"+(all.length===1?"":"s");
    var sel=document.getElementById("tagScope");
    var prev=sel.value;
    sel.innerHTML=tagScopeOptions();
    sel.value = [].slice.call(sel.options).some(function(o){ return o.value===prev; })
      ? prev : (untagged?"untagged":"all");
    if(sel._xsync) sel._xsync();
    document.getElementById("tagWhy").innerHTML = untagged
      ? "<b>"+untagged+"</b> of your <b>"+all.length+"</b> trades have no instrument — the field is optional, so anything typed without one saved blank. That's why <b>By instrument</b> shows a single “untagged” group, and why your plan can't tell you where your edge is."
      : "Every trade is tagged. Use this to <b>change</b> or <b>clear</b> an instrument in bulk.";
    renderTagSuggest();
    updateTagPreview();
  }
