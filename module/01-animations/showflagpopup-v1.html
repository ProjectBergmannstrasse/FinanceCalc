function showFlagPopup(flags){
    var listHtml=flags.map(function(f){
      return '<div class="aud-finding flag" style="margin-bottom:8px">'
        +'<span class="af-icon">'+f.icon+'</span>'
        +'<div class="af-title">'+esc(f.title)+'</div>'
        +'<div class="af-body">'+esc(f.body)+'</div></div>';
    }).join("");

    var ov=document.createElement("div");
    ov.className="ov in";
    ov.style.cssText="position:fixed;inset:0;z-index:300;background:rgba(0,0,0,.72);"
      +"display:flex;align-items:center;justify-content:center;padding:16px";
    ov.innerHTML='<div class="sheet" style="max-width:460px">'
      +'<div class="sheet-hd"><h3>⚠️ Something needs your attention</h3>'
        +'<button class="x-btn" id="fpX">✕</button></div>'
      +'<div class="sheet-bd">'+listHtml
        +'<div style="font-size:11px;color:var(--mut);margin-top:6px;line-height:1.6">'
          +'Spotted automatically from your own figures. Ask AI to talk it through, '
          +'or open Money to see the entries behind it.</div>'
      +'</div>'
      +'<div class="sheet-ft">'
        +'<button class="btn ghost sm" id="fpDismiss">Dismiss</button>'
        +'<button class="btn ghost sm" id="fpAudit">See details</button>'
        +'<button class="btn primary sm" id="fpAsk">Ask AI</button>'
      +'</div></div>';
    document.body.appendChild(ov);

    var close=function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); };
    ov.querySelector("#fpX").onclick=close;
    ov.querySelector("#fpDismiss").onclick=close;
    ov.addEventListener("click", function(e){ if(e.target===ov) close(); });

    ov.querySelector("#fpAudit").onclick=function(){
      close(); openAiSettings();
    };
    ov.querySelector("#fpAsk").onclick=function(){
      close();
      // route to whichever local engine is actually available
      var q="I have these issues with my finances: "
        + flags.map(function(f){ return f.title+" — "+f.body; }).join(" | ")
        + ". What should I do about it?";
      askBestLocalEngine(q);
    };
  }
