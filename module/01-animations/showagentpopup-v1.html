function showAgentPopup(findings, fromWatch){
    if(document.getElementById("agentPopOv")) return;
    var host=document.createElement("div");
    host.className="ov in"; host.id="agentPopOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:440px">'
      +'  <div class="sheet-hd"><h3>'+(fromWatch?"\u26a0\ufe0f Something worth a look":"Deep check results")+'</h3>'
      +'    <button class="x-btn" id="apX">\u2715</button></div>'
      +'  <div class="sheet-bd">'
      +    findings.map(function(f){
            return '<div class="ag-find '+esc(f.level)+'">'
              +'<div class="ag-ft">'+esc(f.title)+'</div>'
              +'<div class="ag-fb">'+esc(f.body)+'</div></div>';
          }).join("")
      +'  </div>'
      +'  <div class="sheet-ft">'
      +(fromWatch?'<button class="btn ghost sm" id="apOff">Stop watching</button>':'')
      +'    <button class="btn ghost sm" id="apDeep">Run deep check</button>'
      +'    <button class="btn primary sm" id="apOk">Got it</button></div>'
      +'</div>';
    document.body.appendChild(host);
    var close=function(){ if(host.parentNode) host.parentNode.removeChild(host); };
    host.querySelector("#apX").onclick=close;
    host.querySelector("#apOk").onclick=close;
    host.querySelector("#apDeep").onclick=function(){ close(); openDeepCheck(); };
    var off=host.querySelector("#apOff");
    if(off) off.onclick=function(){
      agentCfg.watch=false; saveAgentCfg(); refreshAgentPanel(); close();
      flashSaved("background review off");
    };
  }
