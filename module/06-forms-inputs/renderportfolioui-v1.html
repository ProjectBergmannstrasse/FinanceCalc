function renderPortfolioUI(){
    var btn=document.getElementById("portfolioBtn");
    var p=activePortfolio();
    var nm=btn && btn.querySelector(".pf-switch-nm");
    if(nm) nm.textContent=(p&&p.name)||"Portfolio";
    else if(btn) btn.textContent="📁 "+((p&&p.name)||"Portfolio"); // fallback if markup changes
    var list=document.getElementById("portfolioList");
    if(!list) return;
    var html="";
    PORTFOLIOS.forEach(function(pf){
      var st=portfolioStats(pf);
      var on=pf.id===ACTIVE_PID;
      html+='<div class="pf-row'+(on?' pf-on':'')+'" data-pid="'+esc(pf.id)+'">'
        +'<div class="pf-main" data-act="switch">'
          +'<div class="pf-name">'+(on?'✓ ':'')+esc(pf.name)+'</div>'
          +'<div class="pf-sub">'+st.count+' '+(st.count===1?'entry':'entries')+'</div>'
        +'</div>'
        +'<div class="pf-acts">'
          +'<button type="button" class="icon-btn" data-act="dup" title="Duplicate">⧉</button>'
          +'<button type="button" class="icon-btn" data-act="ren" title="Rename">✎</button>'
          +'<button type="button" class="icon-btn" data-act="del" title="Delete">🗑</button>'
        +'</div>'
      +'</div>';
    });
    list.innerHTML=html;
    refreshAdvancedUI();
  }
