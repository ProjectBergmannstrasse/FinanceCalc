function openDeepCheck(){
    var host=document.createElement("div");
    host.className="ov in page-mode"; host.id="deepOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:480px">'
      +'  <div class="sheet-hd"><h3>Advanced check</h3>'
      +'    <button class="x-btn" id="dpX">\u2715</button></div>'
      +'  <div class="sheet-bd">'
      +'    <div class="ag-intro">Each agent takes one question and works through your whole history.</div>'
      +'    <div class="ag-list" id="agList"></div>'
      +'    <div class="ag-meter"><div class="ag-meter-bar" id="agBar"></div></div>'
      +'    <div class="ag-now" id="agNow">Ready.</div>'
      +'  </div>'
      +'  <div class="sheet-ft">'
      +'    <button class="btn ghost sm" id="dpStop" style="display:none">Stop</button>'
      +'    <button class="btn ghost sm" id="dpClose">Close</button>'
      +'    <button class="btn primary sm" id="dpGo">Start</button></div>'
      +'</div>';
    document.body.appendChild(host);
    var listEl=host.querySelector("#agList");
    function drawList(activeId, doneIds){
      doneIds=doneIds||[];
      listEl.innerHTML=AGENTS.map(function(a){
        var st = doneIds.indexOf(a.id)>=0 ? "done" : (a.id===activeId ? "on" : "");
        return '<div class="ag-row '+st+'"><span class="ag-ic">'+a.icon+'</span>'
          +'<span class="ag-nm">'+esc(a.name)
          +'<span class="ag-brief">'+esc(a.brief)+'</span></span>'
          +'<span class="ag-st">'+(st==="done"?"\u2713":st==="on"?'<i class="ag-spin"></i>':"\u00b7")+'</span></div>';
      }).join("");
    }
    drawList();
    var close=function(){ cancelAgents(); if(host.parentNode) host.parentNode.removeChild(host); };
    host.querySelector("#dpX").onclick=close;
    host.querySelector("#dpClose").onclick=close;
    host.querySelector("#dpStop").onclick=function(){
      cancelAgents();
      host.querySelector("#agNow").textContent="Stopped.";
      host.querySelector("#dpStop").style.display="none";
      host.querySelector("#dpGo").style.display="";
    };
    host.querySelector("#dpGo").onclick=function(){
      var go=host.querySelector("#dpGo"), stop=host.querySelector("#dpStop");
      go.style.display="none"; stop.style.display="";
      var done=[];
      var ids=AGENTS.map(function(a){ return a.id; });
      host.querySelector("#agNow").textContent="Starting "+agentConcurrency()+" agent"
        +(agentConcurrency()===1?"":"s")+" at "+agentCfg.intensity+"% intensity\u2026";
      runAgents(ids, function(a,i,total){
        if(i>1) done.push(AGENTS[i-2].id);
        drawList(a.id, done);
        host.querySelector("#agBar").style.width=Math.round((i-1)/total*100)+"%";
        host.querySelector("#agNow").textContent=a.icon+"  "+a.name+" \u2014 "+a.brief;
      }).then(function(findings){
        drawList(null, ids);
        host.querySelector("#agBar").style.width="100%";
        stop.style.display="none"; go.style.display=""; go.textContent="Run again";
        agentCfg.lastDeep=Date.now(); saveAgentCfg();
        host.querySelector("#agNow").textContent=findings.length
          ? findings.length+" finding"+(findings.length===1?"":"s")+"."
          : "Nothing found \u2014 your records look clean.";
        if(findings.length){ setTimeout(function(){ close(); showAgentPopup(findings,false); },600); }
      });
    };
  }
