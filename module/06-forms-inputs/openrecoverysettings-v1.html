function openRecoverySettings(){
    var cfg=recoveryCfg();
    var host=document.createElement("div");
    host.className="ov in"; host.id="rcSetOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:400px">'
      +'  <div class="sheet-hd"><h3>Recovery settings</h3>'
      +'    <button class="x-btn" id="rsX">\u2715</button></div>'
      +'  <div class="sheet-bd">'
      +'    <div class="fld"><label>Rollback windows (minutes, comma separated)</label>'
      +'      <input class="inp" id="rsWins" value="'+esc((cfg.windows||RECOVERY_WINDOWS).join(", "))+'"></div>'
      +'    <div class="fld"><label>Time zone for the log</label>'
      +'      <select class="inp" id="rsTz"></select></div>'
      +'    <div class="rq-warn" style="margin:14px 0 0">Recommended: keep at least one short window '
      +'      (5 min) for slips and one long (1 hour) for a bad session. Nothing is ever destroyed by a '
      +'      rollback \u2014 it is logged and reversible.</div>'
      +'  </div>'
      +'  <div class="sheet-ft"><button class="btn ghost sm" id="rsCancel">Cancel</button>'
      +'    <button class="btn primary sm" id="rsSave">Save</button></div>'
      +'</div>';
    document.body.appendChild(host);
    var tzSel=host.querySelector("#rsTz");
    var zones=["", "UTC","Asia/Dubai","Europe/London","America/New_York","America/Los_Angeles",
               "Asia/Karachi","Asia/Singapore","Australia/Sydney"];
    try{ var local=Intl.DateTimeFormat().resolvedOptions().timeZone;
      if(local && zones.indexOf(local)<0) zones.splice(1,0,local); }catch(e){}
    tzSel.innerHTML=zones.map(function(z){
      return '<option value="'+esc(z)+'"'+(z===cfg.tz?" selected":"")+'>'+(z||"Device default")+'</option>';
    }).join("");
    function close(){ if(host.parentNode) host.parentNode.removeChild(host); }
    host.querySelector("#rsX").onclick=close;
    host.querySelector("#rsCancel").onclick=close;
    host.querySelector("#rsSave").onclick=function(){
      var wins=host.querySelector("#rsWins").value.split(",")
        .map(function(x){ return parseInt(x,10); })
        .filter(function(x){ return x>0 && x<=1440; });
      cfg.windows = wins.length ? wins : RECOVERY_WINDOWS.slice();
      cfg.tz = tzSel.value;
      persistNow(); close(); flashSaved("recovery settings saved"); openRecoverySheet();
    };
  }
