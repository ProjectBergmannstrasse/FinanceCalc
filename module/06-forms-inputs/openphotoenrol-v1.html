function openPhotoEnrol(opts){
    opts=opts||{};
    var captured={};    // slot key -> {descriptor, thumb}
    var host=document.createElement("div");
    host.className="ov in page-mode"; host.id="photoEnrolOv";
    host.innerHTML=
       '<div class="sheet">'
      +'  <div class="sheet-hd"><h3>Set up from photos</h3>'
      +'    <button class="x-btn" id="peX" title="Cancel">\u2715</button></div>'
      +'  <div class="sheet-bd">'
      +'    <div class="pe-intro">Add a photo for each angle below. Tested on real faces, one '
      +'      straight-on photo <b>stops matching</b> when you tilt your head \u2014 several angles is '
      +'      what makes it reliable. Straight ahead is required; the rest each make it better.</div>'
      +'    <div class="pe-grid" id="peGrid"></div>'
      +'    <div class="pe-tips">'
      +'      <div class="pe-tip">\u2600\ufe0f Even light on your face \u2014 no strong backlight</div>'
      +'      <div class="pe-tip">\uD83D\uDC53 Same glasses you normally wear</div>'
      +'      <div class="pe-tip">\uD83D\uDC64 One face per photo, filling most of the frame</div>'
      +'      <div class="pe-tip">\uD83D\uDCF7 Photos of a screen work poorly \u2014 use originals</div>'
      +'    </div>'
      +'  </div>'
      +'  <div class="sheet-ft"><button class="btn ghost sm" id="peCancel">Cancel</button>'
      +'    <button class="btn primary sm" id="peSave" disabled>Save face</button></div>'
      +'  <input type="file" id="peFile" accept="image/*" style="display:none">'
      +'</div>';
    document.body.appendChild(host);

    var grid=host.querySelector("#peGrid");
    var saveBtn=host.querySelector("#peSave");
    var fileInput=host.querySelector("#peFile");
    var pendingSlot=null;

    function draw(){
      grid.innerHTML=PHOTO_SLOTS.map(function(s){
        var got=captured[s.k];
        return '<button class="ps-card'+(got?" done":"")+'" data-k="'+s.k+'">'
          +'<div class="ps-thumb">'
          + (got ? '<img src="'+got.thumb+'" alt="">' : slotArt(s.art))
          + (got ? '<span class="ps-tick">\u2713</span>' : '')
          +'</div>'
          +'<div class="ps-l">'+esc(s.label)+(s.k==="center"?' <span class="ps-req">required</span>':'')+'</div>'
          +'<div class="ps-h">'+esc(got?"Tap to replace":s.hint)+'</div>'
          +'</button>';
      }).join("");
      grid.querySelectorAll(".ps-card").forEach(function(b){
        b.addEventListener("click", function(){
          pendingSlot=b.getAttribute("data-k");
          fileInput.click();
        });
      });
      var n=Object.keys(captured).length;
      saveBtn.disabled=!captured.center;
      saveBtn.textContent = n ? "Save face ("+n+" angle"+(n===1?"":"s")+")" : "Save face";
    }
    draw();

    fileInput.onchange=function(e){
      var f=e.target.files&&e.target.files[0]; if(!f||!pendingSlot) return;
      e.target.value="";
      var slot=pendingSlot;
      var card=grid.querySelector('.ps-card[data-k="'+slot+'"]');
      if(card) card.classList.add("busy");
      var r=new FileReader();
      r.onload=function(){
        var img=new Image();
        img.onload=function(){
          loadFaceApi().then(function(){
            return faceapi.detectSingleFace(img, new faceapi.TinyFaceDetectorOptions())
              .withFaceLandmarks(true).withFaceDescriptor();
          }).then(function(det){
            if(card) card.classList.remove("busy");
            if(!det){ flashSaved("No face found in that photo \u2014 try a clearer one"); return; }
            // check the angle roughly matches the slot, and say so if not
            var pose=poseFromLandmarks(det.landmarks);
            var want=ENROL_STEPS.filter(function(s){ return s.k===slot; })[0];
            var offAngle = want && pose && !poseMatches(want, pose);
            captured[slot]={descriptor:Array.from(det.descriptor), thumb:shrinkThumb(img)};
            draw();
            flashSaved(offAngle
              ? "Saved \u2014 though that looks more front-on than "+slot
              : "angle captured");
          }).catch(function(err){
            if(card) card.classList.remove("busy");
            appModal({title:"Couldn't read that photo",
              message:String(err&&err.message||err).slice(0,200), confirmText:"OK"});
          });
        };
        img.onerror=function(){ if(card) card.classList.remove("busy"); flashSaved("couldn't open that image"); };
        img.src=r.result;
      };
      r.readAsDataURL(f);
    };
    function shrinkThumb(img){
      try{
        var c=document.createElement("canvas"); c.width=c.height=96;
        var x=c.getContext("2d");
        var side=Math.min(img.width, img.height);
        x.drawImage(img,(img.width-side)/2,(img.height-side)/2,side,side,0,0,96,96);
        return c.toDataURL("image/jpeg",0.7);
      }catch(e){ return ""; }
    }

    var close=function(){ if(host.parentNode) host.parentNode.removeChild(host); };
    host.querySelector("#peX").onclick=function(){ close(); opts.onCancel&&opts.onCancel(); };
    host.querySelector("#peCancel").onclick=function(){ close(); opts.onCancel&&opts.onCancel(); };
    saveBtn.onclick=function(){
      var angles=PHOTO_SLOTS.filter(function(s){ return captured[s.k]; })
        .map(function(s){ return {pose:s.k, descriptor:captured[s.k].descriptor}; });
      if(!angles.length) return;
      close();
      opts.onResult&&opts.onResult(angles, angles.length<3);
    };
  }
