function openRecoverySheet(){
    var host=document.createElement("div");
    host.className="ov in page-mode"; host.id="recovOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:520px;height:min(86vh,680px)">'
      +'  <div class="sheet-hd"><h3>Recovery</h3>'
      +'    <button class="x-btn" id="rcGear" title="Recovery settings">\u2699</button>'
      +'    <button class="x-btn" id="rcX" title="Close">\u2715</button></div>'
      +'  <div class="rc-tabs" id="rcTabs">'
      +'    <button class="rc-tab on" data-v="log">Actions</button>'
      +'    <button class="rc-tab" data-v="deep">Vault</button>'
      +'    <button class="rc-tab" data-v="time">Rollback</button>'
      +'  </div>'
      +'  <div class="sheet-bd" id="rcBody"></div>'
      +'</div>';
    document.body.appendChild(host);
    var view="log";
    function close(){ if(host.parentNode) host.parentNode.removeChild(host); }
    host.querySelector("#rcX").onclick=close;
    host.addEventListener("click", function(e){ if(e.target===host) close(); });
    host.querySelector("#rcGear").onclick=function(){ close(); openRecoverySettings(); };
    host.querySelectorAll(".rc-tab").forEach(function(b){
      b.addEventListener("click", function(){
        view=b.getAttribute("data-v");
        host.querySelectorAll(".rc-tab").forEach(function(x){ x.classList.toggle("on", x===b); });
        draw();
      });
    });
    function draw(){
      var box=host.querySelector("#rcBody");
      if(view==="log")  box.innerHTML=drawActionLog();
      if(view==="deep") box.innerHTML=drawVault();
      if(view==="time") box.innerHTML=drawRollback();
      wire(box);
    }
    function wire(box){
      box.querySelectorAll("[data-revert]").forEach(function(b){
        b.addEventListener("click", function(){
          var a=actionLog.filter(function(x){ return x.id===b.getAttribute("data-revert"); })[0];
          if(!a) return;
          if(revertAction(a)){ flashSaved("reverted"); draw(); }
          else flashSaved("couldn't revert that one");
        });
      });
      box.querySelectorAll("[data-restore]").forEach(function(b){
        b.addEventListener("click", function(){
          var n=vaultRestore(b.getAttribute("data-tier"), b.getAttribute("data-restore"));
          flashSaved(n?("restored "+n):"nothing to restore"); draw();
        });
      });
      box.querySelectorAll("[data-mins]").forEach(function(b){
        b.addEventListener("click", function(){
          var m=+b.getAttribute("data-mins");
          appModal({title:"Roll back "+m+" minutes?",
            message:"Everything logged in that window is undone. The rollback itself is logged, so you can undo it too.",
            confirmText:"Roll back", cancelText:"Cancel",
            onConfirm:function(){ rollbackMinutes(m, function(n){
              flashSaved(n?("undid "+n+" action"+(n===1?"":"s")):"nothing in that window"); draw(); }); }});
        });
      });
    }
    draw();
  }
