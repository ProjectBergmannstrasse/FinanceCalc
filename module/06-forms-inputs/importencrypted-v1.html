function importEncrypted(file){
    openOverlay(
      '<div class="sheet"><div class="sheet-hd"><h3>🔓 Decrypt backup</h3><button class="x-btn" id="deX">✕</button></div>'
      +'<div class="sheet-bd">'
        +'<div class="fld"><label>Password</label>'
          +'<input class="inp" id="dePw" type="password" placeholder="the password you used to encrypt"></div>'
        +'<div class="lock-err" id="deErr"></div>'
      +'</div>'
      +'<div class="sheet-ft"><button class="btn sm" id="deCancel">Cancel</button>'
        +'<button class="btn sm solid" id="deGo">Decrypt &amp; import</button></div></div>',
      {ready:function(ov,close){
        ov.querySelector("#deX").onclick=close;
        ov.querySelector("#deCancel").onclick=close;
        ov.querySelector("#deGo").onclick=async function(){
          var pw=ov.querySelector("#dePw").value; if(!pw) return;
          var btn=ov.querySelector("#deGo"); btn.textContent="Decrypting…"; btn.disabled=true;
          var err=ov.querySelector("#deErr");
          try{
            var buf=await file.arrayBuffer();
            var json=await decryptBytes(new Uint8Array(buf),pw);
            applyLoaded(json); persist(); render();
            close(); flashSaved("backup decrypted and loaded");
          }catch(e){
            err.textContent="Wrong password or corrupted file"; err.style.color="var(--down)";
            btn.textContent="Decrypt & import"; btn.disabled=false;
          }
        };
      }}
    );
  }
