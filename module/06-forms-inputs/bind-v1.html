function bind(){
    // chart toggle type
    Array.prototype.forEach.call(document.querySelectorAll("#chartSeg button"), function(b){
      b.addEventListener("click", function(){ state.chartType=b.getAttribute("data-t"); persist(); render(); });
    });
    // collapse chart
    document.getElementById("chartToggle").addEventListener("click", function(){
      state.chartCollapsed=!state.chartCollapsed; persist(); buildChart();
    });

    // type -> show/hide category & adjust label
    var typeSel=document.getElementById("fType");
    function typeUI(){
      var v=typeSel.value;
      document.getElementById("catFld").style.display = (v==="spend")?"flex":"none";
      var lbl=document.getElementById("amtLbl");
      lbl.textContent = (v==="adjust")?"Amount (use − for a drop)":"Amount";
    }
    typeSel.addEventListener("change", typeUI); typeUI();

    // photo picker
    var pendingPhoto=null;
    var fPhoto=document.getElementById("fPhoto");
    document.getElementById("fPhotoBtn").addEventListener("click", function(){ fPhoto.click(); });
    fPhoto.addEventListener("change", function(){
      var f=fPhoto.files && fPhoto.files[0]; if(!f) return;
      document.getElementById("fPhotoName").textContent="reading…";
      processImage(f, function(res){
        pendingPhoto=res;
        var nm=res.name||"photo";
        document.getElementById("fPhotoName").textContent = nm.length>22? nm.slice(0,20)+"…" : nm;
        var prev=document.getElementById("fPhotoPrev");
        prev.src=res.dataUrl; prev.style.display="block";
        document.getElementById("fPhotoClear").style.display="inline-block";
      });
    });
    function clearPhotoPick(){
      pendingPhoto=null; fPhoto.value="";
      document.getElementById("fPhotoName").textContent="";
      document.getElementById("fPhotoPrev").style.display="none";
      document.getElementById("fPhotoClear").style.display="none";
    }
    document.getElementById("fPhotoClear").addEventListener("click", clearPhotoPick);

    // entry submit
    document.getElementById("entryForm").addEventListener("submit", function(e){
      e.preventDefault();
      var type=typeSel.value;
      var date=document.getElementById("fDate").value || todayStr();
      var raw=num(document.getElementById("fAmt").value);
      if(type!=="adjust") raw=Math.abs(raw);
      if(!raw){ shake(document.getElementById("fAmt")); return; }
      var tx={ id:Date.now()+Math.floor(Math.random()*1000), date:date, type:type, amount:raw };
      if(type==="spend") tx.cat=document.getElementById("fCat").value;
      var note=document.getElementById("fNote").value.trim(); if(note) tx.note=note;
      if(pendingPhoto){ tx.photo=tx.id; savePhoto(tx.id, pendingPhoto.dataUrl); }
      state.transactions.push(tx);
      state.selectedMonth=monthKey(date);
      persist(); render();
      document.getElementById("fAmt").value=""; document.getElementById("fNote").value="";
      clearPhotoPick();
      document.getElementById("fAmt").focus();
    });

    // month select
    document.getElementById("monthSel").addEventListener("change", function(e){
      state.selectedMonth=e.target.value; persist(); render();
    });

    // salary / allowance inputs
    [["sAlw","allowance"],["sDebt","toDebt"],["sTrade","toTrading"],["sSave","toSavings"]].forEach(function(pair){
      document.getElementById(pair[0]).addEventListener("input", function(e){
        state.settings[pair[1]]=num(e.target.value); persist(); render();
      });
    });

    // apply allowance — only once per calendar month (max 12 a year)
    document.getElementById("applyAlw").addEventListener("click", function(){
      if(appliedThisMonth()) return;
      var sd=num(state.settings.toDebt), stt=num(state.settings.toTrading), ss=num(state.settings.toSavings);
      if(sd+stt+ss<=0) return;
      var today=todayStr(), curM=monthKey(today), base=Date.now();
      if(sd>0) state.transactions.push({id:base+1,date:today,type:"alw_debt",amount:sd,note:"Allowance → debt",auto:true});
      if(stt>0) state.transactions.push({id:base+2,date:today,type:"alw_trading",amount:stt,note:"Allowance → trading",auto:true});
      if(ss>0) state.transactions.push({id:base+3,date:today,type:"alw_savings",amount:ss,note:"Allowance → savings",auto:true});
      state.selectedMonth=curM;
      persist(); render();
      flashSaved("added");
    });

    // setup save
    document.getElementById("saveSetup").addEventListener("click", function(){
      state.opening.balance=num(document.getElementById("oBal").value);
      state.opening.debt=num(document.getElementById("oDebt").value);
      state.opening.savings=num(document.getElementById("oSav").value);
      var c=document.getElementById("oCur").value.trim(); state.currency=c||"$";
      persist(); render(); flashSaved("saved");
    });

    // build a backup string that also bundles photos
    function exportData(cb){
      var ids=[]; state.transactions.forEach(function(t){ if(t.photo) ids.push(t.photo); });
      var photos={}, i=0;
      (function next(){
        if(i>=ids.length){ cb(JSON.stringify({__mymoney:1, state:state, photos:photos}, null, 2)); return; }
        var id=ids[i++]; loadPhoto(id, function(v){ if(v) photos[id]=v; next(); });
      })();
    }
    function fallbackShow(json){
      var w=document.getElementById("impWrap"); w.style.display="block";
      document.getElementById("impBox").value=json;
    }
    // export copy
    document.getElementById("expCopy").addEventListener("click", function(){
      flashSaved("preparing…");
      exportData(function(json){
        if(navigator.clipboard && navigator.clipboard.writeText){
          navigator.clipboard.writeText(json).then(function(){ flashSaved("copied"); }, function(){ fallbackShow(json); });
        } else fallbackShow(json);
      });
    });
    // export file
    document.getElementById("expFile").addEventListener("click", function(){
      flashSaved("preparing…");
      exportData(function(json){
        try{
          var blob=new Blob([json],{type:"application/json"});
          var url=URL.createObjectURL(blob);
          var a=document.createElement("a"); a.href=url; a.download="my-money-backup.json";
          document.body.appendChild(a); a.click(); document.body.removeChild(a);
          setTimeout(function(){ URL.revokeObjectURL(url); },1000);
          flashSaved("downloaded");
        }catch(e){ alert("Couldn't download here — use Copy backup instead."); }
      });
    });
    // import
    document.getElementById("impToggle").addEventListener("click", function(){
      var w=document.getElementById("impWrap"); w.style.display=(w.style.display==="none"?"block":"none");
    });
    document.getElementById("impLoad").addEventListener("click", function(){
      var txt=document.getElementById("impBox").value.trim(); if(!txt) return;
      try{
        var obj=JSON.parse(txt);
        if(obj && obj.__mymoney && obj.state){
          if(obj.photos){ for(var pid in obj.photos){ savePhoto(pid, obj.photos[pid]); } }
          applyLoaded(JSON.stringify(obj.state));
        } else {
          applyLoaded(JSON.stringify(obj));
        }
        state.selectedMonth=null;
        persist(); render();
        document.getElementById("impWrap").style.display="none";
        flashSaved("loaded");
      }catch(e){ alert("That backup didn't look right. Check you pasted the whole thing."); }
    });
    // reset
    document.getElementById("resetBtn").addEventListener("click", function(){
      if(!confirm("This clears your changes and restores the full history logged from your note. Continue?")) return;
      state.transactions.forEach(function(t){ if(t.photo) delPhoto(t.photo); });
      state=clone(DEFAULTS); state.selectedMonth=null;
      persist(); render(); flashSaved("reset");
    });

    // image viewer
    document.getElementById("viewerClose").addEventListener("click", closeViewer);
    document.getElementById("viewer").addEventListener("click", function(e){ if(e.target===this) closeViewer(); });
    document.getElementById("viewerDel").addEventListener("click", function(){
      if(viewingId==null) return;
      var tx=state.transactions.filter(function(x){ return x.id===viewingId; })[0];
      if(tx && tx.photo){ delPhoto(tx.photo); delete tx.photo; }
      persist(); render(); closeViewer();
    });
    document.addEventListener("keydown", function(e){ if(e.key==="Escape") closeViewer(); });
  }
