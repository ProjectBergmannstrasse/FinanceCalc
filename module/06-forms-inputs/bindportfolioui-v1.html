function bindPortfolioUI(){
    var btn=document.getElementById("portfolioBtn");
    if(btn) btn.addEventListener("click", openPortfolioPanel);
    var closeBtn=document.getElementById("portfolioPanelClose");
    if(closeBtn) closeBtn.addEventListener("click", closePortfolioPanel);
    var ov=document.getElementById("portfolioPanelOverlay");
    if(ov) ov.addEventListener("click", closePortfolioPanel);
    var addBtn=document.getElementById("portfolioAddBtn");
    if(addBtn) addBtn.addEventListener("click", createPortfolio);
    var list=document.getElementById("portfolioList");
    if(list) list.addEventListener("click", function(e){
      var row=e.target.closest(".pf-row"); if(!row) return;
      var pid=row.getAttribute("data-pid");
      var act=e.target.closest("[data-act]"); act=act?act.getAttribute("data-act"):null;
      if(act==="dup") duplicatePortfolio(pid);
      else if(act==="ren") renamePortfolio(pid);
      else if(act==="del") deletePortfolio(pid);
      else switchPortfolio(pid);
    });
  }
