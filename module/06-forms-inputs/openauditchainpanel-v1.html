function openAuditChainPanel(){
    var stored=loadChain();
    var sealedAt=stored?new Date(stored.at).toLocaleString():"Never";
    openOverlay(
      '<div class="sheet"><div class="sheet-hd"><h3>⛓️ Audit Chain</h3><button class="x-btn" id="acX">✕</button></div>'
      +'<div class="sheet-bd">'
        +'<div style="font-size:12px;color:var(--mut);margin-bottom:12px;line-height:1.65">'
          +'SHA-256 hash chain across all your transactions. Seal it now, then verify later '
          +'to prove no entries were modified — useful for tax records or sharing with an accountant.</div>'
        +'<div class="mc-stat" style="margin-bottom:10px">'
          +'<div class="mk">Chain status</div>'
          +'<div id="acStatus" class="mv" style="font-size:14px">'+(stored?"Sealed "+stored.count+" entries":"Not sealed")+'</div>'
          +'<div style="font-size:11px;color:var(--mut);margin-top:3px">Last sealed: '+sealedAt+'</div></div>'
        +'<div id="acResult" style="font-size:13px;margin-top:8px"></div>'
      +'</div>'
      +'<div class="sheet-ft">'
        +'<button class="btn sm" id="acVerify">Verify chain</button>'
        +'<button class="btn sm solid" id="acSeal">Seal now</button>'
        +'<button class="btn sm" id="acX2">Close</button></div></div>',
      {ready:function(ov,close){
        ov.querySelector("#acX").onclick=close;
        ov.querySelector("#acX2").onclick=close;
        ov.querySelector("#acSeal").onclick=async function(){
          var btn=ov.querySelector("#acSeal"); btn.disabled=true; btn.textContent="Sealing…";
          var c=await sealLedger();
          ov.querySelector("#acStatus").textContent="Sealed "+c.count+" entries";
          ov.querySelector("#acResult").innerHTML='<span style="color:var(--up)">✅ Chain sealed. Tip: '+c.tip.slice(0,16)+"…</span>";
          btn.disabled=false; btn.textContent="Seal now";
        };
        ov.querySelector("#acVerify").onclick=async function(){
          var btn=ov.querySelector("#acVerify"); btn.disabled=true; btn.textContent="Verifying…";
          var r=await verifyLedger();
          ov.querySelector("#acResult").innerHTML=r.ok
            ?'<span style="color:var(--up)">✅ Chain intact — '+r.count+' entries verified. No tampering detected.</span>'
            :'<span style="color:var(--down)">🚨 '+esc(r.reason)+'</span>';
          btn.disabled=false; btn.textContent="Verify chain";
        };
      }}
    );
  }
