function bindAuditor(){
    var head=document.getElementById("audHead");
    var body=document.getElementById("audBody");
    var caret=document.getElementById("audCaret");
    if(!head||!body) return;
    var _tier="rules";
    function showTier(t){
      _tier=t;
      ["rules","ai","webllm"].forEach(function(id){
        var btn=document.getElementById("audBtn"+id.charAt(0).toUpperCase()+id.slice(1));
        var panel=document.getElementById("audPanel"+id.charAt(0).toUpperCase()+id.slice(1));
        if(btn) btn.classList.toggle("on",id===t);
        if(panel) panel.style.display=id===t?"":"none";
      });
    }
    document.querySelectorAll(".aud-tier-btn").forEach(function(b){
      b.addEventListener("click",function(){ showTier(b.getAttribute("data-tier")); });
    });
    head.addEventListener("click",function(){
      var open=body.style.display==="none";
      body.style.display=open?"":"none";
      if(caret) caret.style.transform=open?"rotate(90deg)":"";
      head.setAttribute("aria-expanded",open?"true":"false");
      if(open) initBuiltinAi().then(function(){});
    });
    // rules
    var runBtn=document.getElementById("audRunRules");
    if(runBtn) runBtn.addEventListener("click",function(){
      runBtn.textContent="Auditing…"; runBtn.disabled=true;
      setTimeout(function(){
        var f=runAuditRules();
        renderAuditFindings(f);
        runBtn.textContent="Run audit"; runBtn.disabled=false;
      },20);
    });
    // Chrome AI
    var askBtn=document.getElementById("audAskBtn");
    if(askBtn) askBtn.addEventListener("click",async function(){
      var q=document.getElementById("audQuery").value.trim();
      var resp=document.getElementById("audAiResp"); if(!resp||!q) return;
      resp.innerHTML='<div class="aud-llm-resp"><span class="aud-spinner"></span> Thinking…</div>';
      try{
        var ans=await askBuiltinAi(q);
        resp.innerHTML=ans
          ?'<div class="aud-llm-resp">'+esc(ans)+'</div>'
          :'<div class="aud-llm-resp" style="color:var(--amber)">Chrome built-in AI not available. Try the Rules tab or load WebLLM.</div>';
      }catch(e){ resp.innerHTML='<div class="aud-llm-resp" style="color:var(--out)">Error: '+esc(String(e).slice(0,100))+'</div>'; }
    });
    document.getElementById("audQuery").addEventListener("keydown",function(e){ if(e.key==="Enter") askBtn.click(); });
    // WebLLM
    var wLoad=document.getElementById("wllmLoadBtn");
    if(wLoad) wLoad.addEventListener("click",loadWebllm);
    var wAsk=document.getElementById("wllmAskBtn");
    if(wAsk) wAsk.addEventListener("click",async function(){
      var q=document.getElementById("wllmQuery").value.trim();
      var resp=document.getElementById("wllmResp"); if(!resp||!q) return;
      resp.innerHTML='<div class="aud-llm-resp"><span class="aud-spinner"></span> Generating…</div>';
      try{
        var ans=await askWebllm(q);
        resp.innerHTML=ans?'<div class="aud-llm-resp">'+esc(ans)+'</div>'
          :'<div class="aud-llm-resp" style="color:var(--amber)">No response. Is the model loaded?</div>';
      }catch(e){ resp.innerHTML='<div class="aud-llm-resp" style="color:var(--out)">Error: '+esc(String(e).slice(0,100))+'</div>'; }
    });
    document.getElementById("wllmQuery").addEventListener("keydown",function(e){ if(e.key==="Enter") wAsk.click(); });
  }
