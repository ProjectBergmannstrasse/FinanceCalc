function enhanceSelect(sel){
    if(!sel || sel._xs) return;
    sel._xs=true;
    var wrap=document.createElement("div");
    wrap.className="xsel";
    sel.parentNode.insertBefore(wrap, sel);
    wrap.appendChild(sel);
    sel.classList.add("xsel-native");
    var btn=document.createElement("button");
    btn.type="button"; btn.className="xsel-btn";
    btn.innerHTML='<span class="xsel-val"></span><span class="xsel-car">▾</span>';
    wrap.appendChild(btn);
    function sync(){
      var o=sel.options[sel.selectedIndex];
      var vis=xselVisible(sel);
      // an empty picker is a dead button — say so rather than render a lone caret
      btn.querySelector(".xsel-val").textContent = o ? o.textContent : (vis.length ? "" : "—");
      btn.disabled = sel.disabled || !vis.length;
      btn.classList.toggle("placeholder", !o || o.value==="");
    }
    sel._xsync=sync;
    sync();
    sel.addEventListener("change", sync);
    /* Several selects start empty and are filled later — the copy/mirror pickers from the
       portfolio list, the CSV column pickers from the uploaded file's headers. Hunting
       down every rebuild site by hand doesn't work (I missed three), so watch the select
       itself: any change to its options or its hidden/disabled state re-syncs the button.
       MutationObserver fires on a microtask, i.e. after the rebuild AND the .value= that
       follows it, which is exactly the moment the label needs re-reading. */
    if(typeof MutationObserver!=="undefined"){
      new MutationObserver(sync).observe(sel, {
        childList:true, subtree:true, characterData:true,
        attributes:true, attributeFilter:["hidden","disabled","value","selected"]
      });
    }
    btn.addEventListener("click", function(e){ e.preventDefault(); openXsel(sel, btn); });
  }
