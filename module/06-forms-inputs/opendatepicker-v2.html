function openDatePicker(input){
    if(!input) return;
    var today=new Date();
    var selected=dpParse(input.value) || new Date();
    var viewY=selected.getFullYear(), viewM=selected.getMonth();
    var minD=dpParse(input.getAttribute("min"));
    var maxD=dpParse(input.getAttribute("max"));

    var ov=document.createElement("div");
    ov.className="ov in dp-ov";
    ov.innerHTML=
       '<div class="dp-inner">'
      +'  <div class="dp-bar">'
      +'    <button class="dp-nav" id="dpPrev" title="Previous month">\u2039</button>'
      +'    <button class="dp-title" id="dpTitle"></button>'
      +'    <button class="dp-nav" id="dpNext" title="Next month">\u203a</button>'
      +'  </div>'
      +'  <div class="dp-dow">'+DP_DOW.map(function(d){ return '<span>'+d+'</span>'; }).join("")+'</div>'
      +'  <div class="dp-grid" id="dpGrid"></div>'
      +'  <div class="dp-yearwrap" id="dpYearWrap" style="display:none"><div class="dp-years" id="dpYears"></div></div>'
      +'  <div class="dp-foot">'
      +'    <button class="btn ghost sm" id="dpToday">Today</button>'
      +'    <button class="btn ghost sm" id="dpCancel">Cancel</button>'
      +'    <button class="btn primary sm" id="dpOk">Choose</button>'
      +'  </div>'
      +'</div>';
    document.body.appendChild(ov);

    var grid=ov.querySelector("#dpGrid");
    var title=ov.querySelector("#dpTitle");
    var yearWrap=ov.querySelector("#dpYearWrap");
    var yearBox=ov.querySelector("#dpYears");
    var showingYears=false;

    function blocked(d){
      if(minD && d<minD) return true;
      if(maxD && d>maxD) return true;
      return false;
    }
    function draw(){
      title.textContent=DP_MONTHS[viewM]+" "+viewY;
      grid.innerHTML="";
      // Monday-first: getDay() is 0=Sun, so shift it
      var first=new Date(viewY, viewM, 1);
      var lead=(first.getDay()+6)%7;
      var daysIn=new Date(viewY, viewM+1, 0).getDate();
      var prevDays=new Date(viewY, viewM, 0).getDate();
      for(var i=0;i<lead;i++){
        var pd=prevDays-lead+i+1;
        grid.insertAdjacentHTML("beforeend",'<button class="dp-day dp-out" data-off="-1" data-d="'+pd+'">'+pd+'</button>');
      }
      for(var day=1; day<=daysIn; day++){
        var d=new Date(viewY, viewM, day);
        var cls="dp-day";
        if(dpSameDay(d, selected)) cls+=" on";
        if(dpSameDay(d, today))    cls+=" today";
        if(blocked(d))             cls+=" off";
        grid.insertAdjacentHTML("beforeend",
          '<button class="'+cls+'" data-off="0" data-d="'+day+'"'+(blocked(d)?" disabled":"")+'>'+day+'</button>');
      }
      var used=lead+daysIn, tail=(7-(used%7))%7;
      for(var t=1;t<=tail;t++)
        grid.insertAdjacentHTML("beforeend",'<button class="dp-day dp-out" data-off="1" data-d="'+t+'">'+t+'</button>');

      grid.querySelectorAll(".dp-day").forEach(function(b){
        b.addEventListener("click", function(){
          var off=+b.getAttribute("data-off"), dd=+b.getAttribute("data-d");
          var d=new Date(viewY, viewM+off, dd);
          if(blocked(d)) return;
          selected=d; viewY=d.getFullYear(); viewM=d.getMonth();
          draw();
        });
      });
    }
    function drawYears(){
      yearBox.innerHTML="";
      var base=viewY-6;
      for(var y=base; y<base+16; y++){
        yearBox.insertAdjacentHTML("beforeend",
          '<button class="dp-year'+(y===viewY?" on":"")+'" data-y="'+y+'">'+y+'</button>');
      }
      yearBox.querySelectorAll(".dp-year").forEach(function(b){
        b.addEventListener("click", function(){
          viewY=+b.getAttribute("data-y");
          showingYears=false; yearWrap.style.display="none"; grid.style.display="";
          ov.querySelector(".dp-dow").style.display="";
          draw();
        });
      });
    }
    title.addEventListener("click", function(){
      showingYears=!showingYears;
      yearWrap.style.display=showingYears?"":"none";
      grid.style.display=showingYears?"none":"";
      ov.querySelector(".dp-dow").style.display=showingYears?"none":"";
      if(showingYears) drawYears();
    });
    ov.querySelector("#dpPrev").addEventListener("click", function(){
      if(showingYears){ viewY-=16; drawYears(); return; }
      viewM--; if(viewM<0){ viewM=11; viewY--; } draw();
    });
    ov.querySelector("#dpNext").addEventListener("click", function(){
      if(showingYears){ viewY+=16; drawYears(); return; }
      viewM++; if(viewM>11){ viewM=0; viewY++; } draw();
    });
    ov.querySelector("#dpToday").addEventListener("click", function(){
      selected=new Date(); viewY=selected.getFullYear(); viewM=selected.getMonth(); draw();
    });

    function close(){
      ov.classList.remove("in");
      document.removeEventListener("keydown", onKey);
      setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); },150);
    }
    function commit(){
      input.value=dpFmt(selected);
      // let the app's own listeners run exactly as if it had been typed
      input.dispatchEvent(new Event("input",{bubbles:true}));
      input.dispatchEvent(new Event("change",{bubbles:true}));
      close();
    }
    var onKey=function(e){
      if(e.key==="Escape"){ e.preventDefault(); close(); }
      else if(e.key==="Enter"){ e.preventDefault(); commit(); }
      else if(e.key==="ArrowLeft"||e.key==="ArrowRight"||e.key==="ArrowUp"||e.key==="ArrowDown"){
        e.preventDefault();
        var step=e.key==="ArrowLeft"?-1:e.key==="ArrowRight"?1:e.key==="ArrowUp"?-7:7;
        var n=new Date(selected); n.setDate(n.getDate()+step);
        if(blocked(n)) return;
        selected=n; viewY=n.getFullYear(); viewM=n.getMonth(); draw();
      }
    };
    document.addEventListener("keydown", onKey);
    ov.querySelector("#dpCancel").addEventListener("click", close);
    ov.querySelector("#dpOk").addEventListener("click", commit);
    ov.addEventListener("click", function(e){ if(e.target===ov) close(); });
    draw();
  }

  /* Take over every date field, present and future. The input keeps
     type="date" so its value format and validation are untouched — we
     just stop the native picker from appearing. */
  function attachDatePicker(el){
    if(!el || el._dpBound) return;
    el._dpBound=true;
    el.setAttribute("readonly","readonly");   // stops the OS keyboard/picker
    el.classList.add("dp-input");
    var open=function(e){ e.preventDefault(); e.stopPropagation(); openDatePicker(el); };
    el.addEventListener("click", open);
    el.addEventListener("focus", function(){ el.blur(); });
    el.addEventListener("keydown", function(e){
      if(e.key==="Enter"||e.key===" "){ e.preventDefault(); openDatePicker(el); }
    });
  }
  function bindAllDatePickers(root){
    (root||document).querySelectorAll('input[type="date"]').forEach(attachDatePicker);
  }

  /* ============================================================
     PIN RECOVERY — optional security questions

     Without this, a forgotten PIN means clearing the app's data and
     restoring a backup. A recovery answer avoids that.

     It is deliberately weaker than the PIN and the UI says so: an
     answer like a birth year is guessable, so anyone who knows you
     could get in. That's the trade being offered — losing everything
     to a forgotten PIN is the bigger risk for most people, but it
     should be a choice, not a default. Off unless you turn it on.

     The answer is stretched and hashed exactly like the PIN, never
     stored in the clear.
     ============================================================ */
  var RECOVERY_QUESTIONS=[
    "What year were you born?",
    "What was the name of your first pet?",
    "What street did you grow up on?",
    "What was your first job?",
    "What city were you born in?",
    "What was your first car?"
  ];
  function recoveryOn(){ return !!(lockCfg.recQ && lockCfg.recHash); }
  function normaliseAnswer(a){
    // forgiving about case and spacing — nobody remembers "  Rex " exactly
    return String(a||"").trim().toLowerCase().replace(/\s+/g," ");
  }
  function setRecovery(question, answer, cb){
    var salt=randHex(16);
    stretchPin(normaliseAnswer(answer), salt, lockCfg.rounds, function(h){
      lockCfg.recQ=question; lockCfg.recSalt=salt; lockCfg.recHash=h;
      saveLock(); cb&&cb(true);
    });
  }
  function clearRecovery(){
    delete lockCfg.recQ; delete lockCfg.recSalt; delete lockCfg.recHash;
    saveLock();
  }
  function verifyRecovery(answer, cb, onProgress){
    if(!recoveryOn()){ cb(false); return; }
    stretchPin(normaliseAnswer(answer), lockCfg.recSalt, lockCfg.rounds, function(h){
      cb(h===lockCfg.recHash);
    }, onProgress);
  }

  /* Setting it up: pick a question, answer it, confirm the answer. Same
     no-silent-failure rule as the PIN — every problem is stated. */
  function openRecoverySetup(){
    if(!lockIsOn()){
      // don't dead-end them — offer the step they actually need
      appModal({title:"Set a PIN first",
        message:"A recovery question is how you get back in if you forget your PIN \u2014 so there needs to be a PIN first. Set one now?",
        confirmText:"Set a PIN", cancelText:"Not now",
        onConfirm:function(){ openPinSetup(); }});
      return;
    }
    var host=document.createElement("div");
    host.className="ov in"; host.id="recSetupOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:400px">'
      +'  <div class="sheet-hd"><h3>Recovery question</h3>'
      +'    <button class="x-btn" id="rqX">\u2715</button></div>'
      +'  <div class="sheet-bd">'
      +'    <div class="pin-sub">If you forget your PIN, answering this lets you set a new one '
      +'      without wiping your data.</div>'
      +'    <div class="rq-warn">A birth year or a pet\u2019s name is easier to guess than a PIN. '
      +'      Anyone who knows you well could use this to get in \u2014 pick something only you know.</div>'
      +'    <div class="fld"><label>Question</label>'
      +'      <select class="inp" id="rqQ">'
      +        RECOVERY_QUESTIONS.map(function(q){ return '<option>'+esc(q)+'</option>'; }).join("")
      +'      </select></div>'
      +'    <div class="fld"><label>Your answer</label>'
      +'      <input class="inp" id="rqA" autocomplete="off" spellcheck="false" placeholder="answer"></div>'
      +'    <div class="fld"><label>Type it again</label>'
      +'      <input class="inp" id="rqA2" autocomplete="off" spellcheck="false" placeholder="repeat the answer"></div>'
      +'    <div class="pin-err" id="rqErr"></div>'
      +'  </div>'
      +'  <div class="sheet-ft"><button class="btn ghost sm" id="rqCancel">Cancel</button>'
      +'    <button class="btn primary sm" id="rqSave">Save</button></div>'
      +'</div>';
    document.body.appendChild(host);
    var err=host.querySelector("#rqErr");
    function showErr(m){
      err.textContent=m||""; err.style.display=m?"block":"none";
      if(m){ var sh=host.querySelector(".sheet");
        sh.classList.remove("shake"); void sh.offsetWidth; sh.classList.add("shake"); }
    }
    function close(){ if(host.parentNode) host.parentNode.removeChild(host); }
    host.querySelector("#rqX").onclick=close;
    host.querySelector("#rqCancel").onclick=close;
    host.querySelector("#rqSave").onclick=function(){
      var q=host.querySelector("#rqQ").value;
      var a=host.querySelector("#rqA").value;
      var a2=host.querySelector("#rqA2").value;
      if(normaliseAnswer(a).length<2){ showErr("Give an answer of at least 2 characters."); return; }
      if(normaliseAnswer(a)!==normaliseAnswer(a2)){ showErr("Those two answers don\u2019t match."); return; }
      var btn=host.querySelector("#rqSave");
      btn.disabled=true; btn.textContent="Saving\u2026";
      setRecovery(q, a, function(){
        close(); refreshSecUI(); flashSaved("recovery question saved");
      });
    };
  }
