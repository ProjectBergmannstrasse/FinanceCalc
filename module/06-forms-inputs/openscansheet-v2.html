function openScanSheet(file, dataUrl){
    var isTextDetectorSupported=typeof window.TextDetector!=="undefined";
    var sym=state.currency;

    // build the sheet immediately and show the image
    var sheetHtml='<div class="sheet" style="max-width:600px">'
      +'<div class="sheet-hd"><h3>📷 Scan receipt</h3>'
        +'<button class="x-btn" id="scX">✕</button></div>'
      +'<div class="sheet-bd" style="max-height:none;overflow:visible">'
        +'<div style="display:flex;gap:12px;flex-wrap:wrap;align-items:flex-start">'
          +'<div style="flex:1;min-width:140px;max-width:220px">'
            +'<img id="scImg" src="'+dataUrl+'" alt="receipt" '
              +'style="width:100%;border-radius:10px;display:block">'
          +'</div>'
          +'<div style="flex:2;min-width:180px">'
            +'<div id="scStatus" style="font-size:12px;color:var(--mut);margin-bottom:10px">'
              +(isTextDetectorSupported?"Reading text…":"Text reader not available in this browser — fill in manually.")
            +'</div>'
            +'<div class="fld"><label>Amount</label>'
              +'<input class="inp" id="scAmt" inputmode="decimal" placeholder="e.g. 24.99"></div>'
            +'<div class="fld"><label>Date</label>'
              +'<input class="inp" id="scDate" type="date"></div>'
            +'<div class="fld"><label>Note / merchant</label>'
              +'<input class="inp" id="scNote" placeholder="e.g. Carrefour"></div>'
            +'<div class="fld"><label>Category</label>'
              +'<select class="inp" id="scCat">'
                +'<option>Food</option><option>Snacks</option><option>Shopping</option>'
                +'<option>Transport</option><option>Utilities</option><option>Entertainment</option>'
                +'<option>Other</option></select></div>'
          +'</div>'
        +'</div>'
      +'</div>'
      +'<div class="sheet-ft">'
        +'<button class="btn sm" id="scCancel">Cancel</button>'
        +'<button class="btn sm solid" id="scAdd">Add expense</button>'
      +'</div></div>';

    var pair=openOverlay(sheetHtml, {ready:function(ov,close){
      // pre-fill date with today
      document.getElementById("scDate").value=todayStr();
      attachDatePicker(document.getElementById("scDate"));   // sheet is built at runtime
      document.getElementById("scX").onclick=close;
      document.getElementById("scCancel").onclick=close;
      document.getElementById("scAdd").onclick=function(){
        var amt=parseFloat(document.getElementById("scAmt").value);
        if(!isFinite(amt)||amt<=0){ shake(document.getElementById("scAmt")); return; }
        var tx={
          id:"t"+Date.now()+Math.floor(Math.random()*9999),
          type:"expense", amount:amt,
          date:document.getElementById("scDate").value||todayStr(),
          note:document.getElementById("scNote").value.trim(),
          cat:document.getElementById("scCat").value
        };
        state.transactions.unshift(tx);
        persist(); close(); render();
        flashSaved("expense logged from receipt");
      };

      // run OCR after the image has painted
      if(!isTextDetectorSupported) return;
      var img=document.getElementById("scImg");
      var status=document.getElementById("scStatus");
      var run=function(){
        try{
          var cv=preprocessReceiptCanvas(img);
          var detector=new window.TextDetector();
          detector.detect(cv).then(function(blocks){
            var lines=blocks.map(function(b){ return b.rawValue; })
              .sort(function(a,b){
                // sort roughly top-to-bottom by bounding box Y
                return (a._y||0)-(b._y||0);
              });
            // preserve vertical order via blocks directly
            var sorted=blocks.slice().sort(function(a,b){
              return (a.boundingBox.top)-(b.boundingBox.top);
            });
            var sortedLines=sorted.map(function(b){ return b.rawValue; });
            var parsed=parseReceiptText(sortedLines.length?sortedLines:lines);
            // fill in whatever we found
            if(parsed.total){
              document.getElementById("scAmt").value=parsed.total.toFixed(2);
            }
            if(parsed.date){
              document.getElementById("scDate").value=parsed.date;
            }
            if(parsed.merchant||parsed.note){
              document.getElementById("scNote").value=parsed.merchant||parsed.note;
            }
            var found=[];
            if(parsed.total) found.push(sym+parsed.total.toFixed(2));
            if(parsed.date)  found.push(parsed.date);
            if(parsed.merchant) found.push('"'+parsed.merchant+'"');
            status.textContent=found.length
              ? "Found: "+found.join(" · ")+" — check and adjust if needed."
              : "Couldn't read any amounts — fill in manually.";
            status.style.color=found.length?"var(--up)":"var(--amber)";
          }).catch(function(err){
            status.textContent="OCR error — fill in manually. ("+String(err).slice(0,60)+")";
            status.style.color="var(--amber)";
          });
        }catch(err){
          status.textContent="Couldn't run OCR — fill in manually.";
          status.style.color="var(--amber)";
        }
      };
      if(img.complete) run(); else img.onload=run;
    }});
  }
