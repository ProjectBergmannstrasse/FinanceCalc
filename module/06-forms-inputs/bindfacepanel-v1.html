function bindFacePanel(){
    bindCameraSwitch();
    var addBtn=document.getElementById("faceAddBtn");
    if(addBtn) addBtn.addEventListener("click", function(){ enrolFace(function(){ renderFacePeople(); }); });
    var photoBtn=document.getElementById("facePhotoBtn");
    if(photoBtn && !photoBtn._bound){
      photoBtn._bound=true;
      photoBtn.addEventListener("click", function(){
        openPhotoEnrol({
          onResult:function(angles, thin){
            appModal({title:"Whose face is this?",
              message:(thin?"Only "+angles.length+" angle"+(angles.length===1?"":"s")
                        +" \u2014 adding the others makes it far more reliable. ":"Captured "+angles.length+" angles. ")
                     +"Give it a name so you can tell people apart later.",
              input:{placeholder:"e.g. Me, or a family member"}, confirmText:"Save",
              onConfirm:function(name){
                addFacePerson((name||"").trim()||"Me", angles);
                flashSaved("face saved \u2014 "+angles.length+" angle"+(angles.length===1?"":"s"));
                refreshSecUI(); refreshFacePanel(); renderFacePeople();
              }});
          }
        });
      });
    }
    renderFacePeople();
    var mBtn=document.getElementById("faceModelBtn");
    if(mBtn) mBtn.addEventListener("click", function(){
      if(mBtn.onclick) return;        // the unavailable path handles its own click
      doFaceModelDownload();
    });
    refreshFacePanel();
  }
