function openBigExport(){
    var rows=sortedTx();
    var host=document.createElement("div");
    host.className="ov in"; host.id="bigExpOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:400px">'
      +'  <div class="sheet-hd"><h3>Export all entries</h3>'
      +'    <button class="x-btn" id="beX">\u2715</button></div>'
      +'  <div class="sheet-bd">'
      +'    <div class="rc-secd">'+rows.length.toLocaleString()+' entr'+(rows.length===1?"y":"ies")
      +'      to write. Large exports are built in slices so the app stays responsive \u2014 '
      +'      it won\u2019t be instant, but you can watch it and stop it.</div>'
      +'    <div class="ag-meter" style="margin-top:14px"><div class="ag-meter-bar" id="beBar"></div></div>'
      +'    <div class="ag-now" id="beNow">Ready.</div>'
      +'  </div>'
      +'  <div class="sheet-ft">'
      +'    <button class="btn ghost sm" id="beCancel">Close</button>'
      +'    <button class="btn primary sm" id="beGo">Export CSV</button></div>'
      +'</div>';
    document.body.appendChild(host);
    var close=function(){ _exportCancel=true; if(host.parentNode) host.parentNode.removeChild(host); };
    host.querySelector("#beX").onclick=close;
    host.querySelector("#beCancel").onclick=close;
    host.querySelector("#beGo").onclick=function(){
      var go=host.querySelector("#beGo"), bar=host.querySelector("#beBar"), now=host.querySelector("#beNow");
      go.disabled=true; go.textContent="Working\u2026";
      host.querySelector("#beCancel").textContent="Stop";
      var t0=Date.now();
      exportRowsChunked(rows,
        ["date","type","amount","category","note","split"],
        function(t){ return [t.date||"", t.type||"", num(t.amount), t.cat||"", t.note||"", t.splitWays||1]; },
        "mymoney-export-"+todayStr()+".csv",
        function(done,total){
          var pct=Math.round(done/total*100);
          bar.style.width=pct+"%";
          now.textContent=done.toLocaleString()+" of "+total.toLocaleString()+" \u00b7 "+pct+"%";
        }
      ).then(function(r){
        var secs=((Date.now()-t0)/1000).toFixed(1);
        now.textContent="Done \u2014 "+r.rows.toLocaleString()+" rows, "
          +fmtBytes(r.bytes)+", "+secs+"s";
        go.textContent="Export again"; go.disabled=false;
        host.querySelector("#beCancel").textContent="Close";
      }).catch(function(e){
        now.textContent = /cancel/i.test(String(e.message)) ? "Stopped." : "Failed: "+String(e.message).slice(0,60);
        go.textContent="Export CSV"; go.disabled=false;
        host.querySelector("#beCancel").textContent="Close";
      });
    };
  }
