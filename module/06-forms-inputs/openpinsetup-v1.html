function openPinSetup(){
    var stage=1, first="", entered="";
    var host=document.createElement("div");
    host.className="ov in page-mode"; host.id="pinSetupOv";
    host.innerHTML=
       '<div class="sheet pin-sheet">'
      +'  <div class="sheet-hd"><h3 id="pinTitle">Choose a PIN</h3>'
      +'    <button class="x-btn" id="pinX">✕</button></div>'
      +'  <div class="sheet-bd" style="text-align:center">'
      +'    <div class="pin-sub" id="pinSub">4 to 12 digits. There is no recovery — if you forget it you\u2019d have to clear the app\u2019s data and restore a backup.</div>'
      +'    <div class="lk-dots" id="pinDots"></div>'
      +'    <div class="pin-err" id="pinErr"></div>'
      +'    <div class="lk-pad" id="pinPad"></div>'
      +'  </div>'
      +'  <div class="sheet-ft"><button class="btn ghost sm" id="pinCancel">Cancel</button>'
      +'    <button class="btn solid sm" id="pinNext" disabled>Next</button></div>'
      +'</div>';
    document.body.appendChild(host);

    var dots=host.querySelector("#pinDots"), err=host.querySelector("#pinErr");
    var title=host.querySelector("#pinTitle"), sub=host.querySelector("#pinSub");
    var next=host.querySelector("#pinNext");
    function close(){ if(host.parentNode) host.parentNode.removeChild(host); }
    host.querySelector("#pinX").onclick=close;
    host.querySelector("#pinCancel").onclick=close;

    function showErr(msg){
      err.textContent=msg||"";
      err.style.display=msg?"block":"none";
      if(msg){
        var sh=host.querySelector(".pin-sheet");
        sh.classList.remove("shake"); void sh.offsetWidth; sh.classList.add("shake");
      }
    }
    function drawDots(){
      dots.innerHTML="";
      var n=Math.max(4, entered.length);
      for(var i=0;i<n;i++){
        var s=document.createElement("span");
        s.className="lk-dot"+(i<entered.length?" on":"");
        dots.appendChild(s);
      }
      next.disabled = entered.length<4;
      next.textContent = stage===1 ? "Next" : "Set PIN";
    }
    function press(k){
      showErr("");
      if(k==="⌫"){ entered=entered.slice(0,-1); drawDots(); return; }
      if(entered.length>=12){ showErr("A PIN can be at most 12 digits."); return; }
      entered+=k; drawDots();
    }
    var pad=host.querySelector("#pinPad");
    ["1","2","3","4","5","6","7","8","9","","0","⌫"].forEach(function(k){
      var b=document.createElement("button");
      b.className="lk-key"+(k===""?" lk-blank":"");
      b.textContent=k; if(k==="") b.disabled=true;
      b.addEventListener("click", function(){ press(k); });
      pad.appendChild(b);
    });
    host.addEventListener("keydown", function(e){
      if(/^[0-9]$/.test(e.key)) press(e.key);
      else if(e.key==="Backspace") press("⌫");
      else if(e.key==="Enter" && entered.length>=4) next.click();
    });
    host.tabIndex=-1; setTimeout(function(){ try{ host.focus(); }catch(e){} },50);

    next.onclick=function(){
      if(entered.length<4){ showErr("A PIN needs at least 4 digits — you've entered "+entered.length+"."); return; }
      if(stage===1){
        first=entered; entered=""; stage=2;
        title.textContent="Confirm your PIN";
        sub.textContent="Type the same PIN again so we know it's what you meant.";
        showErr(""); drawDots();
        return;
      }
      if(entered!==first){
        showErr("Those two didn't match. Starting again.");
        first=""; entered=""; stage=1;
        title.textContent="Choose a PIN";
        sub.textContent="4 to 12 digits. There is no recovery.";
        drawDots();
        return;
      }
      next.disabled=true; next.textContent="Saving…";
      setLockPin(first, function(){
        close();
        refreshSecUI();
        flashSaved("app lock on");
        // offer face unlock only if a camera is genuinely there
        cameraCheck().then(function(c){
          if(!c.ok) return;
          appModal({title:"Add face unlock?",
            message:"Unlock with your face instead of typing the PIN. Needs a one-time ~6MB model download, then it runs entirely on your device. Your PIN keeps working either way.",
            confirmText:"Set it up", cancelText:"PIN only",
            onConfirm:function(){ enrolFace(function(){ refreshSecUI(); }); }});
        });
      });
    };
    drawDots();
  }
