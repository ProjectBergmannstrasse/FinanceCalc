function openCameraSheet(opts){
    // opts: {title, action:"enrol"|"verify", onResult(descriptorOrMatch), onCancel}
    var host=document.createElement("div");
    host.className="ov in"; host.id="faceOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:400px">'
      +'  <div class="sheet-hd"><h3>'+esc(opts.title||"Face")+'</h3>'
      +'    <button class="x-btn" id="faceX">✕</button></div>'
      +'  <div class="sheet-bd" style="text-align:center">'
      +'    <div class="face-frame"><video id="faceVideo" autoplay playsinline muted></video>'
      +'      <div class="face-ring" id="faceRing"></div></div>'
      +'    <div class="face-status" id="faceStatus">Starting the camera…</div>'
      +'  </div>'
      +'  <div class="sheet-ft"><button class="btn ghost sm" id="faceCancel">Cancel</button>'
      +'    <button class="btn solid sm" id="faceGo" disabled>'
      +      (opts.action==="enrol"?"Capture":"Unlock")+'</button></div>'
      +'</div>';
    document.body.appendChild(host);
    var stream=null, stopped=false;
    var video=host.querySelector("#faceVideo");
    var status=host.querySelector("#faceStatus");
    var go=host.querySelector("#faceGo");
    var ring=host.querySelector("#faceRing");
    function say(t,kind){
      status.textContent=t;
      status.style.color = kind==="bad"?"var(--down)" : kind==="ok"?"var(--up)" : "var(--mut)";
    }
    function cleanup(){
      stopped=true;
      if(stream) stream.getTracks().forEach(function(t){ t.stop(); });
      if(host.parentNode) host.parentNode.removeChild(host);
    }
    host.querySelector("#faceX").onclick=function(){ cleanup(); opts.onCancel&&opts.onCancel(); };
    host.querySelector("#faceCancel").onclick=function(){ cleanup(); opts.onCancel&&opts.onCancel(); };

    navigator.mediaDevices.getUserMedia({video:{facingMode:"user", width:320, height:240}})
      .then(function(s){
        stream=s; video.srcObject=s;
        say("Line your face up inside the circle.");
        go.disabled=false;
        // live "face in frame?" hint where the browser can do it natively
        if(typeof window.FaceDetector!=="undefined"){
          var fd=new window.FaceDetector({fastMode:true, maxDetectedFaces:1});
          var tick=function(){
            if(stopped) return;
            fd.detect(video).then(function(faces){
              if(stopped) return;
              var has=faces&&faces.length;
              ring.classList.toggle("on", !!has);
              say(has?"Face detected — ready.":"No face in frame yet.", has?"ok":null);
              setTimeout(tick, 400);
            }).catch(function(){ setTimeout(tick, 800); });
          };
          setTimeout(tick, 500);
        }
      })
      .catch(function(err){
        var m=String(err&&err.name||err);
        say(/NotAllowed/.test(m) ? "Camera permission was refused."
          : /NotFound/.test(m)   ? "No camera found on this device."
          : /NotReadable/.test(m)? "The camera is in use by another app."
          : "Couldn't start the camera: "+m, "bad");
        go.disabled=true;
      });

    go.onclick=function(){
      go.disabled=true;
      say("Reading your face…");
      loadFaceApi(function(msg){ say(msg); }).then(function(){
        return faceapi.detectSingleFace(video, new faceapi.TinyFaceDetectorOptions())
          .withFaceLandmarks(true).withFaceDescriptor();
      }).then(function(det){
        if(!det){ say("Couldn't see a face clearly — try better light.", "bad"); go.disabled=false; return; }
        var desc=Array.from(det.descriptor);
        if(opts.action==="enrol"){
          cleanup(); opts.onResult&&opts.onResult(desc);
        } else {
          var saved=faceCfg.descriptor;
          if(!saved){ say("No face is enrolled.", "bad"); go.disabled=false; return; }
          var dist=Math.sqrt(desc.reduce(function(sum,v,i){ var d=v-saved[i]; return sum+d*d; },0));
          if(dist<=faceCfg.threshold){ say("Matched.", "ok"); setTimeout(function(){ cleanup(); opts.onResult&&opts.onResult(true); },400); }
          else { say("That's not a match (distance "+dist.toFixed(2)+"). Use your PIN.", "bad"); go.disabled=false; }
        }
      }).catch(function(e){
        say(String(e&&e.message||e).slice(0,110), "bad");
        go.disabled=false;
      });
    };
  }
