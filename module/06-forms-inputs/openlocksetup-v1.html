function openLockSetup(){
    var ls=lockState();
    var hasBio=webAuthnSupported();
    openOverlay(
      '<div class="sheet"><div class="sheet-hd"><h3>🔐 App Lock</h3><button class="x-btn" id="lsX">✕</button></div>'
      +'<div class="sheet-bd">'
        +'<div class="fld"><label>New PIN (4–6 digits)</label>'
          +'<input class="inp" id="lsPin1" type="password" inputmode="numeric" maxlength="6" placeholder="4-6 digits"></div>'
        +'<div class="fld"><label>Confirm PIN</label>'
          +'<input class="inp" id="lsPin2" type="password" inputmode="numeric" maxlength="6" placeholder="repeat PIN"></div>'
        +(hasBio?'<div class="fld" style="display:flex;align-items:center;gap:10px;padding:10px 0">'
          +'<label style="flex:1;cursor:pointer"><b>Also use Face ID / Fingerprint</b>'
            +'<span class="mut" style="display:block;font-size:11px">Registers your biometric after saving</span></label>'
          +'<input type="checkbox" id="lsBio" class="pf-switch" style="width:auto;height:auto"></div>':"")
        +(ls.pin?'<button class="btn danger wide" id="lsRemove" style="margin-top:8px">Remove lock</button>':"")
      +'</div>'
      +'<div class="sheet-ft"><button class="btn sm" id="lsCancel">Cancel</button>'
        +'<button class="btn sm solid" id="lsSave">Save lock</button></div></div>',
      {ready:async function(ov,close){
        ov.querySelector("#lsX").onclick=close;
        ov.querySelector("#lsCancel").onclick=close;
        var rmBtn=ov.querySelector("#lsRemove");
        if(rmBtn) rmBtn.onclick=function(){
          saveLockState({}); _lockSession=true; close(); flashSaved("lock removed");
        };
        ov.querySelector("#lsSave").onclick=async function(){
          var p1=ov.querySelector("#lsPin1").value;
          var p2=ov.querySelector("#lsPin2").value;
          if(!/^\d{4,6}$/.test(p1)){ shake(ov.querySelector("#lsPin1")); return; }
          if(p1!==p2){ shake(ov.querySelector("#lsPin2")); return; }
          var salt=Math.random().toString(36).slice(2)+Date.now().toString(36);
          var hashed=await hashPin(p1,salt);
          var newState={pin:hashed,salt:salt};
          var bioCb=ov.querySelector("#lsBio");
          if(bioCb&&bioCb.checked){
            var credId=await registerBiometric();
            if(credId) newState.credId=credId;
          }
          saveLockState(newState);
          close(); flashSaved("lock set");
        };
      }}
    );
  }
