function renderTradingPage(){
    var all=tradeList(), sym=state.currency;
    document.getElementById("trEmpty").style.display=all.length?"none":"";
    document.getElementById("trWrap").style.display=all.length?"":"none";
    if(!all.length) return;

    var sel=document.getElementById("trInst"), prev=sel.value;
    var insts=tradeByInstrument(all).map(function(r){ return r.inst; });
    sel.innerHTML='<option value="__all">All instruments</option>'+insts.map(function(i){ return '<option value="'+esc(i)+'">'+esc(i)+'</option>'; }).join("");
    sel.value=insts.indexOf(prev)>=0?prev:"__all";
    if(sel._xsync) sel._xsync();   // options were just rebuilt

    var list=trScopedList();
    var wrEl=document.getElementById("trWr"), subEl=document.getElementById("trWrSub");
    if(!list.length){
      wrEl.textContent="—"; wrEl.className="tr-wr mono";
      subEl.textContent="No trades in this period.";
      document.getElementById("trCurve").innerHTML="";
      document.getElementById("trGrid").innerHTML="";
      document.getElementById("trInstPanel").style.display="none";
      document.getElementById("trInsights").innerHTML='<div class="ins ins-info">Nothing to read in this period — try a longer one.</div>';
      return;
    }
    var s=tradeStats(list);
    wrEl.textContent=(s.winRate*100).toFixed(1)+"%";
    wrEl.className="tr-wr mono "+(s.expectancy>=0?"up":"down");
    subEl.innerHTML="win rate · "+s.wins+"W / "+s.losses+"L over "+s.n+" trade"+(s.n===1?"":"s");
    document.getElementById("trCurve").innerHTML=equityCurveSvg(s.curve);

    function cell(k,v,cls){ return '<div class="tr-cell"><div class="tr-k">'+k+'</div><div class="tr-v mono '+(cls||"")+'">'+v+'</div></div>'; }
    var pf=s.profitFactor===Infinity?"∞":s.profitFactor.toFixed(2);
    document.getElementById("trGrid").innerHTML=
       cell("Net P&amp;L",fmt(s.net,sym),s.net>=0?"up":"down")
      +cell("Expectancy / trade",fmt(s.expectancy,sym),s.expectancy>=0?"up":"down")
      +cell("Profit factor",pf,s.profitFactor>=1?"up":"down")
      +cell("Max drawdown",fmt(s.maxDD,sym),s.maxDD>0?"down":"")
      +cell("Avg win",fmt(s.avgWin,sym),"up")
      +cell("Avg loss",fmt(s.avgLoss,sym),"down")
      +cell("Best trade",fmt(s.best,sym),"up")
      +cell("Worst trade",fmt(s.worst,sym),"down")
      +cell("Longest win streak",s.bestWinStreak)
      +cell("Longest loss streak",s.bestLossStreak);

    var rows=tradeByInstrument(list);
    var ip=document.getElementById("trInstPanel");
    if(rows.length>1 && sel.value==="__all"){
      ip.style.display="";
      document.getElementById("trTbl").innerHTML=rows.map(function(r){
        var st=r.stats;
        return '<div class="tr-row"><div class="tr-i">'+esc(r.inst)+'<span class="tr-n">'+st.n+' trade'+(st.n===1?"":"s")+'</span></div>'
          +'<div class="tr-wr2 mono '+(st.winRate>=0.5?"up":"down")+'">'+(st.winRate*100).toFixed(0)+'%</div>'
          +'<div class="tr-pl mono '+(st.net>=0?"up":"down")+'">'+fmt(st.net,sym)+'</div></div>';
      }).join("");
    } else ip.style.display="none";

    renderTradeForecast(list, sym);
    renderPlan(tradeList(), sym);   // the plan is your whole record, not the ticked period
    renderTagCard();

    var ICON={good:"✅",bad:"⚠️",warn:"⚡",info:"ℹ️"};
    // Seventeen rules can match at once. Fifteen paragraphs of advice is a wall nobody
    // reads, so show the few that matter most: problems first, then warnings, then
    // observations. The rest are one tap away rather than shouting over the important ones.
    var RANK={bad:0, warn:1, good:2, info:3};
    var all=tradeInsights(list,sym).map(function(x,i){ x._i=i; return x; })
      .sort(function(a,b){ return (RANK[a.kind]-RANK[b.kind]) || (a._i-b._i); });
    var top=all.slice(0, TR_INSIGHT_MAX), rest=all.slice(TR_INSIGHT_MAX);
    var draw=function(x){
      return '<div class="ins ins-'+x.kind+'"><span class="ins-ic">'+ICON[x.kind]+'</span><span>'+x.text+'</span></div>';
    };
    var h=top.map(draw).join("");
    if(rest.length){
      h+='<div id="insMore" style="display:none">'+rest.map(draw).join("")+'</div>'
        +'<button type="button" class="ins-more-btn" id="insMoreBtn">'+rest.length+' more</button>';
    }
    document.getElementById("trInsights").innerHTML=h;
    var mb=document.getElementById("insMoreBtn");
    if(mb) mb.addEventListener("click", function(){
      var box=document.getElementById("insMore");
      var open=box.style.display!=="none";
      box.style.display=open?"none":"block";
      mb.textContent=open?(rest.length+" more"):"Show less";
    });
  }
