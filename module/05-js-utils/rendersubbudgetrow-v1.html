function renderSubBudgetRow(cat, sym, thisM){
    var cap=budgetFor(cat), spent=spentInCat(cat,null,thisM);
    var subs=subCatsFor(cat);
    var pct=cap>0?Math.min(100,spent/cap*100):0;
    var over=cap>0&&spent>cap;
    var subRows=subs.map(function(sub){
      var ss=state.transactions.filter(function(tx){ return tx.subcat===sub&&balDelta(tx)<0&&monthKey(tx.date||"")===thisM; })
        .reduce(function(a,tx){ return a+Math.abs(balDelta(tx)); },0);
      if(!ss) return "";
      return '<div class="sub-bud-row"><span class="sub-bud-name mut">↳ '+esc(sub)+'</span><span class="sub-bud-amt mut">'+sym+ss.toFixed(2)+'</span></div>';
    }).join("");
    return '<div class="bud-row'+(over?" bud-over":"")+'">'+
      '<div class="bud-row-hd"><span>'+esc(cat)+(cap?'<span class="bud-cap"> / '+sym+cap.toFixed(0)+'</span>':"")+'</span>'+
      '<span class="bud-amt'+(over?" bud-over-t":"")+'">'+sym+spent.toFixed(2)+'</span></div>'+
      (cap?'<div class="bud-bar"><div class="bud-fill'+(over?" bud-fill-over":"")
        +'" style="width:'+pct+'%"></div></div>':"")
      +subRows+'</div>';
  }
