function csvPreview(){
    var rows=csvBuild();
    var good=rows.filter(function(r){ return !r.skip && !r.dupe; });
    var dupes=rows.filter(function(r){ return r.dupe; }).length;
    var bad=rows.filter(function(r){ return r.skip; }).length;
    var box=document.getElementById("csvPreview");
    if(!rows.length){ box.innerHTML='<div class="csv-note">Pick a date column and an amount column.</div>'; return; }
    var html='<div class="csv-tbl"><div class="csv-r hd"><span class="csv-c1">Date</span><span class="csv-c2">Amount</span><span class="csv-c3">Type</span><span class="csv-c4">Instrument / note</span></div>';
    rows.slice(0,6).forEach(function(r){
      if(r.skip){ html+='<div class="csv-r dupe"><span class="csv-c4 csv-skip">skipped — couldn\'t read a date and amount</span></div>'; return; }
      var meta=TYPE_META[r.type]||{label:r.type};
      html+='<div class="csv-r'+(r.dupe?" dupe":"")+'"><span class="csv-c1">'+esc(r.date)+'</span>'
        +'<span class="csv-c2">'+fmt(r.amount,state.currency)+'</span>'
        +'<span class="csv-c3">'+esc(meta.label)+'</span>'
        +'<span class="csv-c4">'+esc(r.inst||"")+(r.dupe?' <span class="csv-skip">· already added</span>':'')+'</span></div>';
    });
    html+='</div>';
    var bits=["<b>"+good.length+"</b> to import"];
    if(dupes) bits.push("<b>"+dupes+"</b> already in your ledger (skipped)");
    if(bad) bits.push("<b>"+bad+"</b> unreadable (skipped)");
    html+='<div class="csv-note" style="margin-top:8px">'+bits.join(" · ")+(rows.length>6?" — showing first 6":"")+'</div>';
    box.innerHTML=html;
    var btn=document.getElementById("csvImportBtn");
    btn.disabled=!good.length;
    btn.textContent=good.length?("Import "+good.length):"Nothing to import";
  }
