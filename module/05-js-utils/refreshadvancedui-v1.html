function refreshAdvancedUI(){
    var wrap=document.getElementById("advWrap");
    if(!wrap || wrap.style.display!=="block") return;
    var others=PORTFOLIOS.filter(function(p){ return p.id!==ACTIVE_PID; });
    var copySel=document.getElementById("copyFromSel");
    var mirSel=document.getElementById("mirrorSel");
    var p=activePortfolio();
    if(copySel){
      copySel.innerHTML = others.length
        ? others.map(function(o){ return '<option value="'+esc(o.id)+'">'+esc(o.name)+'</option>'; }).join("")
        : '<option value="">No other portfolios</option>';
    }
    if(mirSel){
      mirSel.innerHTML = '<option value="">Off — don\'t mirror</option>'
        + others.map(function(o){
            var sel=(p && p.mirrorFrom===o.id) ? " selected" : "";
            return '<option value="'+esc(o.id)+'"'+sel+'>'+esc(o.name)+'</option>';
          }).join("");
    }
    var st=document.getElementById("mirrorState");
    if(st){
      if(p && p.mirrorFrom){
        var src=findPortfolio(p.mirrorFrom);
        st.innerHTML = src
          ? '✅ Mirroring <b>'+esc(src.name)+'</b>. New entries added there will appear here automatically. Entries added <i>here</i> do not go back the other way.'
          : '⚠️ The portfolio this was mirroring no longer exists.';
      } else st.textContent="";
    }
  }
