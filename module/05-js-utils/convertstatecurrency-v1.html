function convertStateCurrency(fromSym, toSym){
    var fromCode=SYMBOL_TO_CODE[fromSym], toCode=SYMBOL_TO_CODE[toSym];
    if(!fromCode || !toCode || fromCode===toCode) return false;
    state.transactions.forEach(function(t){ t.amount=roundCents(convertAmount(t.amount, fromCode, toCode)); });
    state.opening.balance=roundCents(convertAmount(state.opening.balance, fromCode, toCode));
    state.opening.debt=roundCents(convertAmount(state.opening.debt, fromCode, toCode));
    state.opening.savings=roundCents(convertAmount(state.opening.savings, fromCode, toCode));
    state.settings.allowance=roundCents(convertAmount(state.settings.allowance, fromCode, toCode));
    state.settings.toDebt=roundCents(convertAmount(state.settings.toDebt, fromCode, toCode));
    state.settings.toTrading=roundCents(convertAmount(state.settings.toTrading, fromCode, toCode));
    state.settings.toSavings=roundCents(convertAmount(state.settings.toSavings, fromCode, toCode));

    // Everything below holds money too. Missing any of them leaves that feature quietly
    // denominated in the OLD currency — a $200 cap becomes a €200 cap, a restored entry
    // comes back at the wrong value, and a rent rule keeps charging the old number.
    if(state.budgets && typeof state.budgets==="object"){
      Object.keys(state.budgets).forEach(function(k){
        var v=num(state.budgets[k]);
        if(v>0) state.budgets[k]=roundCents(convertAmount(v, fromCode, toCode));
      });
    }
    if(Array.isArray(state.recurring)){
      state.recurring.forEach(function(r){ r.amount=roundCents(convertAmount(num(r.amount), fromCode, toCode)); });
    }
    if(Array.isArray(state.goals)){
      state.goals.forEach(function(g){ if(g && g.target!=null) g.target=roundCents(convertAmount(num(g.target), fromCode, toCode)); });
    }
    if(Array.isArray(state.trash)){
      state.trash.forEach(function(item){
        if(item && item.tx) item.tx.amount=roundCents(convertAmount(num(item.tx.amount), fromCode, toCode));
      });
    }
    return true;
  }
