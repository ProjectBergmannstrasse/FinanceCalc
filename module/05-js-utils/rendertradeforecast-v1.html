function renderTradeForecast(list, sym){
    var card=document.getElementById("trFcCard");
    if(!card) return;
    if(!list.length){ card.style.display="none"; return; }
    var f=tradeForecast(list, trFcDays);
    if(!f.trades){ card.style.display="none"; return; }
    card.style.display="";
    var numEl=document.getElementById("trFcNum");
    numEl.textContent=(f.expected>=0?"+":"")+fmt(f.expected,sym);
    numEl.className="fc-num mono "+(f.expected>=0?"up":"down");
    document.getElementById("trFcCap").textContent="expected over "+trFcDays+" days · about "+f.trades+" trade"+(f.trades===1?"":"s");

    var sim=tradeSim(list, trFcDays, 1000);
    var band=document.getElementById("trFcBand");
    if(sim){
      var pct=Math.round(sim.chanceDown*100);
      band.innerHTML=mcBar(sim,sym)
        +'<div class="mc-legend">'
        +'<span>middle 80%: <b>'+fmt(sim.p10,sym)+'</b> to <b>'+fmt(sim.p90,sym)+'</b></span>'
        +'<span>typical: <b>'+fmt(sim.median,sym)+'</b></span>'
        +'<span>down after '+trFcDays+'d: <b>'+pct+'%</b></span>'
        +'</div>';
    } else band.innerHTML="";

    var warn=document.getElementById("trFcWarn");
    if(f.thin){
      warn.innerHTML='<div class="fc-alert warn"><span>◔</span><span>Only <b>'+list.length+'</b> trade'+(list.length===1?"":"s")+' to go on. Under ~'+TR_FC_MIN_TRADES+', one result moves all of this — treat it as a sketch.</span></div>';
    } else if(f.expectancy<0){
      warn.innerHTML='<div class="fc-alert bad"><span>⚠️</span><span>Your edge is <b>negative</b> ('+fmt(f.expectancy,sym)+' per trade). More trading makes this worse, not better — the projection gets more negative the longer you go.</span></div>';
    } else if(sim && sim.chanceDown>=0.35){
      warn.innerHTML='<div class="fc-alert warn"><span>⚡</span><span>Edge is positive, but <b>'+Math.round(sim.chanceDown*100)+'%</b> of runs still finish down. A real edge loses often — that\'s what surviving it costs.</span></div>';
    } else warn.innerHTML="";

    document.getElementById("trFcBasis").innerHTML="Your edge is <b>"+fmt(f.expectancy,sym)+"/trade</b> and you trade about <b>"+(Math.round(f.perDay*10)/10)+"×/day</b> (over "+f.histDays+" days). Expected = edge × trades. The range comes from replaying your own wins and losses <b>1,000</b> times — same numbers, reshuffled. It describes your record, it doesn\'t predict the market.";
  }
