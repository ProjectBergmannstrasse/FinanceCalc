function renderTradingSanity(){
    var box=document.getElementById("trNegWarn");
    if(!box) return;
    if(!sepTrading()){ box.style.display="none"; return; }
    var t=totals(), sym=state.currency;
    if(t.trading>=0){ box.style.display="none"; return; }
    var funded=tradingFunded();
    box.style.display="";
    document.getElementById("trNegMsg").innerHTML=
      "⚠️ Your <b>"+esc(L("trading"))+"</b> account shows <b>"+fmt(t.trading,sym)+"</b> — an account can't hold less than nothing."
      + (funded<=0
          ? " You've logged trades but never logged <b>moving money into trading</b>, so the losses are being charged to a pot that was never funded."
          : " You've funded <b>"+fmt(funded,sym)+"</b> and lost more than that.")
      + " That means <b>Money I have right now</b> is overstating your cash by <b>"+fmt(Math.abs(t.trading),sym)+"</b> — it's really "
      + fmt(t.balance-Math.abs(t.trading),sym) + ". Add a <b>Moved to trading</b> entry for what you actually put in, or turn the split off in Settings.";
  }
