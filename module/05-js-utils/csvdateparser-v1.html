function csvDateParser(samples){
    var iso=/^(\d{4})-(\d{2})-(\d{2})/, dmy=/^(\d{1,2})[\/.](\d{1,2})[\/.](\d{4})/, ymd=/^(\d{4})[\/.](\d{1,2})[\/.](\d{1,2})/;
    var sawBig=false, sawBigSecond=false;
    samples.forEach(function(v){
      var m=dmy.exec(String(v).trim());
      if(m){ if(+m[1]>12) sawBig=true; if(+m[2]>12) sawBigSecond=true; }
    });
    var dayFirst = sawBig || !sawBigSecond; // default to D/M (rest of world) unless proven otherwise
    return function(v){
      v=String(v||"").trim();
      var m=iso.exec(v); if(m) return m[1]+"-"+m[2]+"-"+m[3];
      m=ymd.exec(v);     if(m) return m[1]+"-"+two(+m[2])+"-"+two(+m[3]);
      m=dmy.exec(v);
      if(m){
        var a=+m[1], b=+m[2];
        var day=dayFirst?a:b, mon=dayFirst?b:a;
        if(mon>12){ var t=day; day=mon; mon=t; }  // impossible month => it was the other way
        return m[3]+"-"+two(mon)+"-"+two(day);
      }
      var d=new Date(v);
      if(!isNaN(d.getTime())) return d.getFullYear()+"-"+two(d.getMonth()+1)+"-"+two(d.getDate());
      return null;
    };
  }
