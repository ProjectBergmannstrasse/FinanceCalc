function checkBillNotifs(daysAhead){
    daysAhead = daysAhead || 3;
    if(typeof Notification==="undefined") return;   // browser doesn't support it
    loadNotifSent();
    var todayS = todayStr(), today = new Date(todayS);
    // build the window of dates to check
    var window_dates = [];
    for(var d=0; d<=daysAhead; d++){
      var dt = new Date(today); dt.setDate(dt.getDate()+d);
      window_dates.push(dt.getFullYear()+"-"+String(dt.getMonth()+1).padStart(2,"0")+"-"+String(dt.getDate()).padStart(2,"0"));
    }
    // which rules have something due in that window?
    var upcoming = [];
    recurList().forEach(function(r){
      if(r.paused) return;
      window_dates.forEach(function(ds){
        var dts = dueDates(r, ds);
        if(dts.indexOf(ds)>=0){
          var daysUntil = window_dates.indexOf(ds);
          upcoming.push({rule:r, date:ds, daysUntil:daysUntil});
        }
      });
    });
    if(!upcoming.length) return;
    // reset sent-list daily
    if((_notifSent.date||"") !== todayS){ _notifSent={date:todayS,ids:[]}; }
    // fire one notification per rule not yet fired today
    upcoming.forEach(function(item){
      var key = item.rule.id+"_"+item.date;
      if(_notifSent.ids.indexOf(key)>=0) return;   // already sent
      var label = item.rule.note || item.rule.cat || "Bill";
      var amt = item.rule.amount ? " — "+fmt(item.rule.amount, state.currency) : "";
      var when = item.daysUntil===0 ? "due today" : "due in "+item.daysUntil+" day"+(item.daysUntil===1?"":"s");
      try{
        new Notification("💳 "+label+amt, {
          body: when.charAt(0).toUpperCase()+when.slice(1),
          icon: "",
          tag: key,
          renotify: false
        });
        _notifSent.ids.push(key);
        saveNotifSent();
      }catch(e){}
    });
  }

  function requestBillNotifs(cb){
    if(typeof Notification==="undefined"){ if(cb) cb(false); return; }
    if(Notification.permission==="granted"){ checkBillNotifs(); if(cb) cb(true); return; }
    if(Notification.permission==="denied"){ if(cb) cb(false); return; }
    Notification.requestPermission(function(perm){
      if(perm==="granted"){ checkBillNotifs(); }
      if(cb) cb(perm==="granted");
    });
  }

  // ---- preprocessing: boost contrast, greyscale → helps OCR ----
  function preprocessReceiptCanvas(src, maxW){
    maxW=maxW||1200;
    var cv=document.createElement("canvas");
    var scale=Math.min(1,maxW/(src.naturalWidth||src.width||maxW));
    cv.width =(src.naturalWidth||src.width)*scale;
    cv.height=(src.naturalHeight||src.height)*scale;
    var ctx=cv.getContext("2d"); ctx.drawImage(src,0,0,cv.width,cv.height);
    // greyscale + contrast boost
    var id=ctx.getImageData(0,0,cv.width,cv.height), d=id.data;
    for(var i=0;i<d.length;i+=4){
      var g=0.299*d[i]+0.587*d[i+1]+0.114*d[i+2];
      // S-curve contrast: push mid-tones apart
      var c=(g/255-0.5)*1.6+0.5;
      var v=Math.min(255,Math.max(0,c*255));
      d[i]=d[i+1]=d[i+2]=v;
    }
    ctx.putImageData(id,0,0);
    return cv;
  }

  function parseReceiptText(lines){
    var text=lines.join("\n");
    // currency amounts: $123.45, 12.50, etc.
    var amounts=[];
    var amtRe=/[\$£€]?\s*(\d{1,6}[,.]\d{2})\b/g, m;
    while((m=amtRe.exec(text))!==null){
      var v=parseFloat(m[1].replace(",","."));
      if(isFinite(v)&&v>0&&v<99999) amounts.push(v);
    }
    // best guess at the total: the LARGEST amount (receipts put total last)
    amounts.sort(function(a,b){return b-a;});
    var total=amounts.length?amounts[0]:null;
    // date: ISO first (unambiguous), then DD/MM or MM/DD auto-detect
    var date=null;
    var isoM=text.match(/\b(\d{4})[\/\-\.](\d{1,2})[\/\-\.](\d{1,2})\b/);
    if(isoM){
      var iy=+isoM[1],im=+isoM[2],idd=+isoM[3];
      if(im>=1&&im<=12&&idd>=1&&idd<=31) date=iy+"-"+(im<10?"0":"")+im+"-"+(idd<10?"0":"")+idd;
    }
    if(!date){
      var dmyM=text.match(/\b(\d{1,2})[\/\-\.](\d{1,2})[\/\-\.](\d{2,4})\b/);
      if(dmyM){
        var da=+dmyM[1],db=+dmyM[2],dyr=+dmyM[3]; if(dyr<100)dyr+=2000;
        var rdd,rmm;
        if(da>12&&db<=12){rdd=da;rmm=db;}else if(db>12&&da<=12){rdd=db;rmm=da;}else{rdd=da;rmm=db;}
        if(rmm>=1&&rmm<=12&&rdd>=1&&rdd<=31) date=dyr+"-"+(rmm<10?"0":"")+rmm+"-"+(rdd<10?"0":"")+rdd;
      }
    }
    // merchant: first non-blank line that isn't all numbers/punctuation
    var merchant="";
    for(var i=0;i<lines.length;i++){
      var ln=lines[i].trim();
      if(ln.length>2&&!/^[\d\s\.\-\$£€*#:\/\\,]+$/.test(ln)){
        merchant=ln.slice(0,50); break;
      }
    }
    // note: build a clean one-liner
    var note=(merchant?" "+merchant:"")+(date?" "+date:"");
    return {total:total, date:date, merchant:merchant, note:note.trim(), amounts:amounts, raw:text};
  }
