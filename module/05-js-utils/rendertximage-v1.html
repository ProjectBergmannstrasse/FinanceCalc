function renderTxImage(t, withPhoto, cb){
    var C=themeColors(), sym=state.currency, DPR=2;
    var meta=TYPE_META[t.type]||{label:t.type};
    var d=String(t.date||"").split("-");
    var dateStr=(MONTHS[(+d[1])-1]||"")+" "+(+d[2])+", "+d[0];
    var W=440, pad=22;

    // gather the pot-move rows as {label, text, colorKey}
    var moves=[];
    var addMove=function(label,v,key){ if(Math.abs(v)<0.005) return;
      moves.push({label:label, text:(v>0?"+":"\u2212")+fmt(Math.abs(v),sym), key:key}); };
    addMove(L("trading")||"Money", balDelta(t), balDelta(t)>0?"up":"down");
    addMove(L("debt"), debtDelta(t), debtDelta(t)>0?"up":"down");
    addMove(L("savings"), savDelta(t), savDelta(t)>0?"up":"down");
    addMove("Tax set aside", taxDelta(t), "cool");
    if(sepTrading()) addMove("Trading pot", tradingDelta(t), tradingDelta(t)>0?"up":"down");
    if(!moves.length) moves.push({label:"Nothing moved", text:"\u2014", key:"mut"});

    var details=[];
    if(t.cat) details.push(["Category", t.cat]);
    if(t.inst) details.push(["Instrument", t.inst]);
    var note=(t.note||"").trim();

    var loadImg=function(next){
      if(!withPhoto){ next(null); return; }
      var ps=photosOf(t); if(!ps.length){ next(null); return; }
      loadPhoto(ps[0], function(url){
        if(!url){ next(null); return; }
        var im=new Image(); im.onload=function(){ next(im); }; im.onerror=function(){ next(null); };
        im.src=url;
      });
    };

    loadImg(function(photoImg){
      var mono='-apple-system,BlinkMacSystemFont,"Segoe UI",Roboto,sans-serif';

      // wrap the note once, up front, so both passes agree
      var noteLines=[];
      if(note){
        var words=note.split(/\s+/), line="";
        words.forEach(function(w){
          if((line+" "+w).trim().length>46){ noteLines.push(line.trim()); line=w; }
          else line+=" "+w;
        });
        if(line.trim()) noteLines.push(line.trim());
      }
      var photoDims=null;
      if(photoImg){
        var iw=W-pad*2, ih=Math.min(240, iw*photoImg.height/photoImg.width);
        photoDims={w:iw,h:ih};
      }

      /* One draw routine, run twice. First pass measures the true bottom (no clipping is
         possible because the height IS whatever the drawing reached). Second pass draws
         for real at that exact size. Beats hand-tuned constants that drift out of sync. */
      function paint(x, measureOnly){
        var cy=pad;
        if(!measureOnly){ x.fillStyle=C.bg; x.fillRect(0,0,W, x.canvas.height/DPR); }

        // badge
        var badge=meta.label;
        x.font="700 12px "+mono;
        var bw=(x.measureText(badge).width||badge.length*7)+24;
        if(!measureOnly){ x.fillStyle=C.card2; roundRect(x,(W-bw)/2,cy,bw,26,13); x.fill();
          x.fillStyle=C.mut; x.textAlign="center"; x.textBaseline="middle";
          x.fillText(badge, W/2, cy+13); }
        cy+=44;

        // big amount
        var bd=balDelta(t), amtColor = bd<0 ? C.down : (bd>0 ? C.up : C.tx);
        if(Math.abs(bd)<0.005) amtColor=C.tx;
        if(!measureOnly){ x.font="800 44px "+mono; x.fillStyle=amtColor; x.textAlign="center";
          x.textBaseline="alphabetic";
          x.fillText((bd<0?"\u2212":"")+sym+Math.abs(num(t.amount)).toFixed(2), W/2, cy+34); }
        cy+=58;

        // date
        if(!measureOnly){ x.font="500 15px "+mono; x.fillStyle=C.mut; x.textAlign="center";
          x.fillText(dateStr, W/2, cy); }
        cy+=30;

        function divider(){ if(!measureOnly){ x.strokeStyle=C.bd; x.lineWidth=1;
          x.beginPath(); x.moveTo(pad,cy); x.lineTo(W-pad,cy); x.stroke(); } cy+=18; }
        function section(title){ if(!measureOnly){ x.font="700 11px "+mono; x.fillStyle=C.mut;
          x.textAlign="left"; x.fillText(title.toUpperCase(), pad, cy); } cy+=24; }
        function rowLR(label,val,color){ if(!measureOnly){
          x.font="500 15px "+mono; x.textAlign="left"; x.fillStyle=C.tx; x.fillText(label, pad, cy);
          x.font="700 15px "+mono; x.textAlign="right"; x.fillStyle=color||C.tx; x.fillText(val, W-pad, cy); }
          cy+=26; }

        divider(); section("What it moved");
        moves.forEach(function(m){
          var col=m.key==="up"?C.up:m.key==="down"?C.down:m.key==="cool"?C.cool:m.key==="mut"?C.mut:C.tx;
          rowLR(m.label, m.text, col);
        });
        cy+=10;

        if(details.length){ divider(); section("Details");
          details.forEach(function(dd){ rowLR(dd[0], dd[1], C.tx); }); cy+=10; }

        if(noteLines.length){
          divider(); section("Note");
          var boxTop=cy-2, boxH=noteLines.length*22+18;
          if(!measureOnly){ x.fillStyle=C.card2; roundRect(x,pad,boxTop,W-pad*2,boxH,10); x.fill();
            x.strokeStyle=C.bd; x.lineWidth=1; roundRect(x,pad,boxTop,W-pad*2,boxH,10); x.stroke();
            x.font="500 15px "+mono; x.fillStyle=C.tx; x.textAlign="left";
            var ty=boxTop+22; noteLines.forEach(function(ln){ x.fillText(ln, pad+14, ty); ty+=22; }); }
          cy=boxTop+boxH+14;
        }

        if(photoDims){
          divider(); section(photoDims?"Photo":"File");
          if(!measureOnly){ roundRectClip(x, pad, cy, photoDims.w, photoDims.h, 10, function(){
            x.drawImage(photoImg, pad, cy, photoDims.w, photoDims.h); }); }
          cy+=photoDims.h+10;
        }

        return cy+pad;   // the true bottom
      }

      // pass 1: measure on a scratch context
      var scratch=document.createElement("canvas").getContext("2d");
      var H=Math.ceil(paint(scratch, true));

      // pass 2: draw for real
      var cv=document.createElement("canvas");
      cv.width=W*DPR; cv.height=H*DPR;
      var ctx2=cv.getContext("2d"); ctx2.scale(DPR,DPR);
      paint(ctx2, false);

      cv.toBlob(function(blob){ cb(blob, cv.toDataURL("image/png")); }, "image/png");
    });
  }
