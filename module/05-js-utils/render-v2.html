function render(){
    var t=totals(), sym=state.currency;

    // live date, top of page
    updateDateBar();

    // hero
    var hb=document.getElementById("heroBal");
    hb.innerHTML='<span class="cur">'+esc(sym)+'</span>'+ (t.balance<0?"-":"") + Math.abs(t.balance).toLocaleString("en-US",{minimumFractionDigits:2,maximumFractionDigits:2});
    hb.className="big mono "+(t.balance<0?"down":"");
    var curM=monthKey(todayStr());
    var sThis=summaryFor(curM);
    var capParts=[];
    if(sThis.net!==0) capParts.push((sThis.net>0?"up ":"down ")+fmt(Math.abs(sThis.net),sym)+" this month");
    else capParts.push("no change yet this month");
    document.getElementById("heroCap").textContent=capParts[0];

    // trio
    document.getElementById("stDebt").textContent=fmt(t.debt,sym);
    document.getElementById("stDebt").className="v mono "+(t.debt>0?"down":"up");
    document.getElementById("stSav").textContent=fmt(t.savings,sym);
    // trading account tile — only exists once you've split it out from your cash
    var trWrap=document.getElementById("stTrWrap");
    if(sepTrading()){ trWrap.style.display=""; document.getElementById("stTr").textContent=fmt(t.trading,sym); }
    else trWrap.style.display="none";

    // the tax tile stays hidden until tax is actually in play, so it doesn't clutter
    // the row for people who never use it
    var taxWrap=document.getElementById("stTaxWrap");
    var taxUsed = Math.abs(t.tax)>=0.01 || num(state.settings.taxPct)>0 || num(state.opening.tax)>0;
    taxWrap.style.display = taxUsed ? "" : "none";
    if(taxUsed) document.getElementById("stTax").textContent=fmt(t.tax,sym);

    // win rate lives in its own dashboard card now — the stat grid stays an even 3
    var netEl=document.getElementById("stNet");
    netEl.textContent=fmt(t.net,sym);
    netEl.className="v mono "+(t.net>=0?"up":"down");

    // chart
    buildChart();

    // automation summary
    var alw=num(state.settings.allowance), sd=num(state.settings.toDebt), stt=num(state.settings.toTrading), ss=num(state.settings.toSavings);
    var used=sd+stt+ss, left=alw-used;
    var au=document.getElementById("autoSummary");
    var msg;
    // Zero is a legitimate answer — plenty of people have no fixed monthly income.
    // Don't warn about "unassigned" money when there's no money to assign.
    if(alw<=0 && used<=0){
      msg="No fixed monthly amount set. That's fine — just add entries manually as money comes in.";
    } else {
      msg="Splitting <b>"+fmt(alw,sym)+"</b>: <b>"+fmt(sd,sym)+"</b> to "+esc(L("debt").toLowerCase())+", <b>"+fmt(stt,sym)+"</b> to "+esc(L("trading").toLowerCase())+", <b>"+fmt(ss,sym)+"</b> to "+esc(L("savings").toLowerCase())+".";
      if(Math.abs(left)>=0.01) msg+=" <span class='warn'>("+(left>0?fmt(left,sym)+" not assigned":"over by "+fmt(-left,sym))+")</span>";
    }
    au.innerHTML=msg;

    // allowance button state — shows next un-added month, allows up to 3 months advance
    var btn=document.getElementById("applyAlw");
    var amSet2=allowanceMonths();
    // find next unadded month starting from current month — walks forward with monthAdd()
    // so it rolls over Dec → Jan into next year automatically, instead of stopping dead
    // at December of whatever year it happens to be right now.
    var nextUnadded=null;
    var mk2=curM;
    for(var gi=0; gi<60; gi++){
      if(!amSet2[mk2]){ nextUnadded=mk2; break; }
      mk2=monthAdd(mk2,1);
    }
    // how many advance months already taken (months added AFTER current month)
    var advanceTaken=0;
    for(var ak in amSet2){
      if(ak > curM) advanceTaken++;
    }
    var canAdvance=advanceTaken<3;
    if(used<=0){ btn.disabled=true; btn.textContent=(alw<=0?"No monthly amount":"Set amounts first"); }
    else if(!nextUnadded){ btn.disabled=true; btn.textContent="All set for now ✓"; }
    else if(nextUnadded<=curM){
      // current or past month not yet added — always allow
      btn.disabled=false;
      btn.textContent="Add "+monthLong(nextUnadded);
    } else {
      // would be an advance month
      if(canAdvance){
        btn.disabled=false;
        btn.textContent="Advance: "+monthLong(nextUnadded)+" ("+(3-advanceTaken)+" left)";
        btn.style.borderColor="var(--amber)";
        btn.style.color="var(--amber)";
      } else {
        btn.disabled=true;
        btn.textContent="Max advance taken (3 months)";
      }
    }

    // year progress — 12 pips lit for months added, label shows current month / 12
    var yc=document.getElementById("yrCount");
    var amSet=allowanceMonths(), pips="";
    var curMonthNum=+todayStr().slice(5,7), yr=todayStr().slice(0,4);
    for(var pi=1;pi<=12;pi++){
      var mk=yr+"-"+two(pi);
      var on=!!amSet[mk];
      pips+='<span class="pip'+(on?" on":"")+(pi>curMonthNum?" future":"")+'"></span>';
    }
    yc.innerHTML='<span class="pips">'+pips+'</span><span>Month <b class="done-tag">'+curMonthNum+'</b> / 12 of '+yr+'</span>';

    // outstanding advances — one row per future month already funded, each removable
    var advL=document.getElementById("advList");
    if(advL){
      var outAdv=outstandingAdvances();
      if(!outAdv.length){ advL.innerHTML=""; }
      else{
        var ah='<div class="adv-list-ttl">Outstanding advances</div>';
        outAdv.forEach(function(a){
          ah+='<div class="adv-row"><span>'+monthLong(a.forMonth)+' <span class="mono">'+fmt(a.amount,sym)+'</span> <span class="mut" style="font-weight:400">· taken '+niceDate(a.date)+'</span></span>'
             +'<button class="adv-remove" data-fm="'+a.forMonth+'">✕ Remove</button></div>';
        });
        advL.innerHTML=ah;
        Array.prototype.forEach.call(advL.querySelectorAll(".adv-remove"), function(b){
          b.addEventListener("click", function(){ removeAdvance(b.getAttribute("data-fm")); });
        });
      }
    }

    // month picker — normalize selection (defaults to current month on first-ever load)
    var months=monthsInData();
    if(state.selectedMonths===null){
      state.selectedMonths=[curM];
    } else if(!isAllSelected()){
      state.selectedMonths=state.selectedMonths.filter(function(m){ return months.indexOf(m)>=0; });
    }
    updateMonthPickBtn();

    // metrics — combined across every ticked month (or everything, or nothing)
    var activeM=activeMonths(), allSel=isAllSelected();
    var sm = allSel ? summaryFor("all") : summaryFor(activeM);

    // advance notices — one set of banners per ticked month
    var advNotesEl=document.getElementById("advNotes");
    if(advNotesEl){
      var nh="";
      if(!allSel && activeM.length>0){
        activeM.slice().sort().forEach(function(mkey){
          var advIn=advanceReceivedFor(mkey);
          if(advIn){
            nh+='<div class="adv-note adv-in"><span>✓ '+monthLong(mkey)+'\'s allowance ('+fmt(advIn.amount,sym)+') was already taken in advance on '+niceDate(advIn.date)+'.</span>'
               +'<button class="adv-remove" data-fm="'+mkey+'">✕ Remove</button></div>';
          }
          advancesTakenIn(mkey).forEach(function(a){
            nh+='<div class="adv-note adv-out"><span>⚡ Advance taken (in '+monthLong(mkey)+') for '+monthLong(a.forMonth)+': '+fmt(a.amount,sym)+'</span>'
               +'<button class="adv-remove" data-fm="'+a.forMonth+'">✕ Remove</button></div>';
          });
        });
      }
      advNotesEl.innerHTML=nh;
      Array.prototype.forEach.call(advNotesEl.querySelectorAll(".adv-remove"), function(b){
        b.addEventListener("click", function(){ removeAdvance(b.getAttribute("data-fm")); });
      });
    }

    var m=document.getElementById("metrics");
    function metric(k,v,cls){ return '<div class="metric"><div class="k">'+k+'</div><div class="v mono '+(cls||"")+'">'+v+'</div></div>'; }
    var tradeCls=sm.tradePL>0?"up":(sm.tradePL<0?"down":"");
    var netCls=sm.net>0?"up":(sm.net<0?"down":"");
    m.innerHTML=
      metric("Spent", fmt(sm.spent,sym), sm.spent>0?"down":"") +
      metric("Trade P/L", (sm.tradePL>0?"+":"")+fmt(sm.tradePL,sym), tradeCls) +
      metric("Money in", fmt(sm.moneyIn,sym), sm.moneyIn>0?"up":"") +
      metric("Net change", (sm.net>0?"+":"")+fmt(sm.net,sym), netCls);

    // categories
    var cats=document.getElementById("cats");
    var keys=Object.keys(sm.cats).sort(function(a,b){ return sm.cats[b]-sm.cats[a]; });
    if(keys.length===0){ cats.innerHTML= (!allSel && activeM.length===0) ? '<div class="cttl">Tick a month above to see numbers here.</div>' : ""; }
    else{
      var max=sm.cats[keys[0]]||1;
      var html='<div class="cttl">Where the spending went'+(allSel?"":" · "+scopeLabel(activeM))+'</div>';
      keys.forEach(function(k){
        var v=sm.cats[k], pct=Math.max(4,Math.round(v/max*100));
        html+='<div class="catrow"><div class="cn">'+esc(k)+'</div><div class="catbar"><span style="width:'+pct+'%"></span></div><div class="cv mono">'+fmt(v,sym)+'</div></div>';
      });
      cats.innerHTML=html;
    }
    renderBudgets();
    renderForecast();
    renderHeatmap();
    renderGoals();
    checkRecurPrompt();

    // ledger
    renderLedger();
    checkAllowancePrompt();

    // sync inputs (only if not focused, to avoid disrupting typing)
    syncInput("sAlw", state.settings.allowance);
    syncInput("sTaxPct", state.settings.taxPct);
    syncInput("oTrading", state.opening.trading);
    var sepCb=document.getElementById("sSepTrading");
    if(sepCb && document.activeElement!==sepCb) sepCb.checked=sepTrading();
    var sepHint=document.getElementById("sepHint");
    if(sepHint){
      sepHint.innerHTML = sepTrading()
        ? "✅ Split. The big number is your <b>spendable cash</b>. Trade P&L and allowance-to-trading go to the trading pot, and you can move money between them with the two <b>Move</b> entry types."
        : "Currently mixed — trade wins and losses move the same number you'd spend from, so it isn't really “money you have”.";
      sepHint.style.color = sepTrading() ? "var(--up)" : "var(--mut)";
    }
    var otr=document.getElementById("openTrRow");
    if(otr) otr.style.display = sepTrading() ? "" : "none";
    // hero label must say what it's actually counting
    var hl=document.getElementById("heroLbl");
    if(hl) hl.textContent = sepTrading() ? "Spendable cash right now" : "Money I have right now";
    applyLabels();
    // the transfer types are meaningless while it's one pot
    Array.prototype.forEach.call(document.querySelectorAll("#fType .sep-only"), function(o){
      o.style.display = sepTrading() ? "" : "none";
      o.disabled = !sepTrading();
    });
    var tph=document.getElementById("taxPctHint");
    if(tph){
      var tp=num(state.settings.taxPct);
      tph.textContent = tp>0
        ? "When you add money in or a trade win, the app will offer to set aside "+tp+"% for tax."
        : "Set a percentage to get a nudge whenever money comes in. Leave at 0 to handle tax manually.";
    }
    syncInput("sDebt", state.settings.toDebt);
    syncInput("sTrade", state.settings.toTrading);
    syncInput("sSave", state.settings.toSavings);
    syncInput("oBal", state.opening.balance);
    syncInput("oDebt", state.opening.debt);
    syncInput("oSav", state.opening.savings);
    // sync currency select
    syncCurrencySelect();

    // chart seg active
    Array.prototype.forEach.call(document.querySelectorAll("#chartSeg button"), function(b){
      b.classList.toggle("on", b.getAttribute("data-t")===state.chartType);
    });
  }
