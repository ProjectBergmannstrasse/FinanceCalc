function renderTrash(){
    var box=document.getElementById("trashList");
    if(!box) return;
    var items=trashList(), sym=state.currency;
    document.getElementById("trashBtns").style.display=items.length?"":"none";
    if(!items.length){ box.innerHTML='<div class="tr-empty-msg">Nothing deleted recently.</div>'; return; }
    box.innerHTML=items.slice(0,60).map(function(item,i){
      var t=item.tx||{}, meta=TYPE_META[t.type]||{label:t.type||"?"};
      var nm=(t.note||t.cat||meta.label||"Entry");
      var left=trashDaysLeft(item);
      var when = left<=0 ? "going now" : (left===1 ? "1 day left" : left+" days left");
      return '<div class="tr-item" data-i="'+i+'">'
        +'<div class="tr-item-main"><div class="tr-item-nm">'+esc(nm)+'</div>'
        +'<div class="tr-item-sub">'+esc(t.date||"")+' · '+esc(meta.label)+(t.photo?" · 📎":"")+' · <span class="'+(left<=5?"tr-soon":"")+'">'+when+'</span></div></div>'
        +'<div class="tr-item-amt">'+fmt(num(t.amount),sym)+'</div>'
        +'<button type="button" class="tr-item-btn" data-i="'+i+'">Restore</button>'
        +'</div>';
    }).join("")+(items.length>60?'<div class="tr-empty-msg">…and '+(items.length-60)+' more</div>':"");
    Array.prototype.forEach.call(box.querySelectorAll(".tr-item-btn"), function(b){
      b.addEventListener("click", function(){
        var tx=restoreFromTrash(+b.getAttribute("data-i"));
        if(tx){
          resetLedgerView();
          render(); renderTrash();
          flashSaved("restored");
        }
      });
    });
  }
