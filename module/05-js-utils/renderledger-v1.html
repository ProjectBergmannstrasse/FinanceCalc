function renderLedger(){
    var sym=state.currency, sc=state.selectedMonth;
    var list=sortedTx().reverse().filter(function(t){ return sc==="all" || monthKey(t.date)===sc; });
    var led=document.getElementById("ledger");
    document.getElementById("ledCount").textContent = list.length? "· "+list.length : "";
    if(list.length===0){
      led.innerHTML='<div class="empty"><b>Nothing logged'+(sc==="all"?"":" for "+monthLong(sc))+' yet.</b><br>Add your first entry above.</div>';
      return;
    }
    var html="";
    list.forEach(function(t){
      var d=t.date.split("-"), dd=(+d[2]), mm=MONTHS[(+d[1])-1];
      var meta=TYPE_META[t.type]||{label:t.type};
      var db=balDelta(t), dd2=debtDelta(t), ds=savDelta(t);
      var amtHtml, tag="";
      if(db!==0){ amtHtml=(db>0?"+":"")+fmt(db,sym); amtHtml='<span class="'+(db>0?"up":"down")+'">'+amtHtml+'</span>'; }
      else if(dd2!==0){ // debt-only (allowance→debt). less debt = good = green
        amtHtml='<span class="'+(dd2<0?"up":"down")+'">'+(dd2<0?"−":"+")+fmt(Math.abs(dd2),sym)+'</span>'; tag="debt";
      } else if(ds!==0){ amtHtml='<span class="cool">'+(ds>0?"+":"−")+fmt(Math.abs(ds),sym)+'</span>'; tag="savings"; }
      else { amtHtml='<span class="neu">'+fmt(0,sym)+'</span>'; }
      var note = t.note? esc(t.note) : (t.type==="spend"&&t.cat? esc(t.cat) : "");
      var hasProof = !!(t.photo || t.screenshot);
      var eyeBtn = hasProof ? '<span class="eye-btn" title="Click row to view screenshot">\u{1F441}</span>' : '';
      html+='<div class="lrow'+(hasProof?" has-proof":"")+'" data-id="'+t.id+'">'
          +'<div class="ldate">'+mm+'<b>'+dd+'</b></div>'
          +'<div class="lmid"><span class="badge">'+esc(meta.label)+'</span>'
          +(note?'<div class="lnote">'+note+'</div>':'')+'</div>'
          +eyeBtn
          +'<div class="lamt">'+amtHtml+(tag?'<span class="tag">'+tag+'</span>':'')+'</div>'
          +'<button class="del" data-id="'+t.id+'" title="Delete">\xd7</button>'
          +'</div>';
    });
    led.innerHTML=html;
    Array.prototype.forEach.call(led.querySelectorAll(".del"), function(b){
      b.addEventListener("click", function(e){
        e.stopPropagation();
        var id=+b.getAttribute("data-id");
        var tx2=state.transactions.filter(function(x){ return x.id===id; })[0];
        if(tx2 && tx2.photo) delPhoto(tx2.photo);
        state.transactions=state.transactions.filter(function(x){ return x.id!==id; });
        persist(); render();
      });
    });
    Array.prototype.forEach.call(led.querySelectorAll(".lrow"), function(row){
      row.addEventListener("click", function(e){
        if(e.target.classList.contains("del")) return;
        var id=+row.getAttribute("data-id");
        var tx3=state.transactions.filter(function(x){ return x.id===id; })[0];
        if(tx3 && (tx3.photo || tx3.screenshot)) openViewer(id);
      });
    });
  }
