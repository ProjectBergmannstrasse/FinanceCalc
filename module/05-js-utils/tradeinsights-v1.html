function tradeInsights(list, sym){
    var out=[], s=tradeStats(list), n=list.length;
    function add(kind, text){ out.push({kind:kind, text:text}); }

    if(n<20) add("info","Only <b>"+n+" trade"+(n===1?"":"s")+"</b> in this period. Below ~20 the numbers below swing wildly on one result — treat them as rough.");

    // break-even win rate implied by the actual R:R
    if(s.avgWin>0 && s.avgLoss>0){
      var rr=s.avgWin/s.avgLoss;
      var be=1/(1+rr)*100;
      var gap=(s.winRate*100)-be;
      var line="Your average win is <b>"+fmt(s.avgWin,sym)+"</b> against an average loss of <b>"+fmt(s.avgLoss,sym)+"</b> — a reward:risk of <b>"+rr.toFixed(2)+"</b>. "
        +"At that ratio you need <b>"+be.toFixed(1)+"%</b> to break even and you're at <b>"+(s.winRate*100).toFixed(1)+"%</b>";
      if(gap>=0) add("good", line+", so you're <b>"+gap.toFixed(1)+" points clear</b>.");
      else add("bad", line+" — <b>"+Math.abs(gap).toFixed(1)+" points short</b>. Either the win rate rises or the winners get bigger relative to the losers; nothing else closes that gap.");
    }
    // the classic trap
    if(s.winRate>=0.5 && s.expectancy<0)
      add("bad","You win <b>"+(s.winRate*100).toFixed(1)+"%</b> of the time and still lose money: expectancy is <b>"+fmt(s.expectancy,sym)+"</b> per trade. The win rate is flattering this.");
    if(s.winRate<0.5 && s.expectancy>0)
      add("good","You lose more often than you win, but expectancy is <b>+"+fmt(s.expectancy,sym).replace(/^./,"")+"</b> per trade — the winners are carrying it. That's a working edge.");

    // concentration: is one trade doing all the work?
    if(n>=5 && s.grossWin>0){
      var biggest=s.best/s.grossWin;
      if(biggest>0.4) add("warn","Your single best trade (<b>"+fmt(s.best,sym)+"</b>) is <b>"+(biggest*100).toFixed(0)+"%</b> of all your winnings. Strip it out and the picture changes — this record leans on one result.");
    }
    if(n>=5 && s.grossLoss>0){
      var worst=s.worst/s.grossLoss;
      if(worst>0.4) add("warn","One loss (<b>"+fmt(s.worst,sym)+"</b>) is <b>"+(worst*100).toFixed(0)+"%</b> of everything you've lost. Your damage is concentrated in single events, not slow bleed.");
    }
    // drawdown vs typical win
    if(s.maxDD>0 && s.avgWin>0){
      var back=s.maxDD/Math.max(s.expectancy>0?s.expectancy:s.avgWin,0.0001);
      if(s.expectancy>0 && back>=10) add("warn","Worst drawdown was <b>"+fmt(s.maxDD,sym)+"</b>. At <b>"+fmt(s.expectancy,sym)+"</b>/trade that takes about <b>"+Math.ceil(back)+" trades</b> to climb back.");
    }
    // streaks — a real one, not folklore
    if(s.bestLossStreak>=4) add("info","Longest losing run: <b>"+s.bestLossStreak+" trades</b>. Worth knowing before it happens again — it will.");

    // after a loss, do you do worse? (revenge-trading, measured not assumed)
    if(n>=12){
      var after=[], rest=[];
      for(var i=1;i<list.length;i++){
        (list[i-1].type==="trade_loss"?after:rest).push(balDelta(list[i]));
      }
      if(after.length>=5 && rest.length>=5){
        var avgA=after.reduce(function(a,b){return a+b;},0)/after.length;
        var avgR=rest.reduce(function(a,b){return a+b;},0)/rest.length;
        if(avgA<avgR-Math.abs(avgR*0.25)&&avgA<0)
          add("bad","Trades taken straight after a loss average <b>"+fmt(avgA,sym)+"</b>, versus <b>"+fmt(avgR,sym)+"</b> otherwise (<b>"+after.length+"</b> vs <b>"+rest.length+"</b> trades). Your worst trades cluster right after your losses.");
        else if(avgA>avgR)
          add("good","Trades right after a loss average <b>"+fmt(avgA,sym)+"</b> vs <b>"+fmt(avgR,sym)+"</b> otherwise — no sign of tilt in the data.");
      }
    }
    // stake discipline
    var sizes=list.map(function(t){ return Math.abs(num(t.amount)); }).filter(function(x){ return x>0; });
    if(sizes.length>=8){
      var mean=sizes.reduce(function(a,b){return a+b;},0)/sizes.length;
      var sd=Math.sqrt(sizes.reduce(function(a,b){return a+Math.pow(b-mean,2);},0)/sizes.length);
      var cv=sd/mean;
      if(cv>1) add("warn","Your trade sizes are all over the place — average <b>"+fmt(mean,sym)+"</b>, swing of <b>"+fmt(sd,sym)+"</b>. When size is inconsistent, one oversized trade decides your month rather than your edge.");
      else if(cv<0.35) add("good","Trade sizing is consistent (average <b>"+fmt(mean,sym)+"</b>, swing <b>"+fmt(sd,sym)+"</b>) — results reflect your edge rather than your bet size.");
    }
    // instruments: best vs worst, only when each has a real sample
    var byI=tradeByInstrument(list).filter(function(r){ return r.stats.n>=5 && r.inst!=="— untagged —"; });
    if(byI.length>=2){
      var best=byI[0], worst=byI[byI.length-1];
      if(best.stats.net>0 && worst.stats.net<0)
        add("info","<b>"+esc(best.inst)+"</b> is your earner (<b>"+fmt(best.stats.net,sym)+"</b> over "+best.stats.n+" trades). <b>"+esc(worst.inst)+"</b> is where it goes (<b>"+fmt(worst.stats.net,sym)+"</b> over "+worst.stats.n+"). Same you, different results.");
    }
    // recent form vs the whole record
    if(trPeriod==="all" && tradeList().length>=25){
      var all=tradeList(), recent=all.slice(-10), older=all.slice(0,-10);
      if(older.length>=10){
        var rs=tradeStats(recent), os=tradeStats(older);
        var d=(rs.winRate-os.winRate)*100;
        if(Math.abs(d)>=15)
          add(d>0?"good":"warn","Last 10 trades: <b>"+(rs.winRate*100).toFixed(0)+"%</b> win rate against <b>"+(os.winRate*100).toFixed(0)+"%</b> before that — "+(d>0?"improving":"a real drop")+" of <b>"+Math.abs(d).toFixed(0)+" points</b>.");
      }
    }
    if(!out.length) add("info","Nothing stands out in this period — no lopsided ratios, no concentration, no tilt pattern. That's usually a good sign.");
    return out;
  }
