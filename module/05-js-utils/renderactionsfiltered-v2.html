function renderActionsFiltered(){
    renderActions();
    var box=document.getElementById("actList");
    if(!box) return;
    var secs=(ACT_FILTERS.filter(function(f){return f[0]===actFilter;})[0]||[null,0])[1];
    if(secs>0){
      var rows=box.querySelectorAll(".act-row");
      ACTIONS.forEach(function(a,i){
        if(rows[i]&&(Date.now()-a.at)/1000>secs) rows[i].style.display="none";
      });
    }
    if(actFilter==="all"){
      actArchiveList(function(arch){
        if(!arch.length) return;
        arch.sort(function(a,b){return b.at-a.at;});
        var live={}; ACTIONS.forEach(function(a){live[a.at]=1;});
        var older=arch.filter(function(a){return !live[a.at];}).slice(0,100);
        if(!older.length) return;
        var div=document.createElement("div");
        div.innerHTML='<div class="hint" style="margin:10px 0 6px">Older (archived \u2014 view only, undo window has passed)</div>'
          +older.map(function(a){
            return '<div class="act-row"><div class="act-t"><b>'+esc(a.label)+'</b><span class="act-w">'+new Date(a.at).toLocaleString()+'</span></div></div>';
          }).join("");
        box.appendChild(div);
      });
    }
  }
