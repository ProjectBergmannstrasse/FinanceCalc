function drawRing(){
      var n=60, out="";
      var per=n/SWEEP_BUCKETS.length;
      for(var i=0;i<n;i++){
        var seg=Math.floor(i/per);
        var b=SWEEP_BUCKETS[seg];
        var has=!!best[b.k];
        var ang=(i/n)*360-90;
        var r1=has?95:99, r2=has?113:107;
        var x1=120+r1*Math.cos(ang*Math.PI/180), y1=120+r1*Math.sin(ang*Math.PI/180);
        var x2=120+r2*Math.cos(ang*Math.PI/180), y2=120+r2*Math.sin(ang*Math.PI/180);
        out+='<line x1="'+x1.toFixed(1)+'" y1="'+y1.toFixed(1)+'" x2="'+x2.toFixed(1)+'" y2="'+y2.toFixed(1)
          +'" stroke="'+(has?"var(--up)":"var(--bd2)")+'" stroke-width="'+(has?3:2)
          +'" stroke-linecap="round"'+(has?'':' opacity="0.45"')+'/>';
      }
      ticks.innerHTML=out;
      badge.textContent=got()+"/"+SWEEP_BUCKETS.length;
      chips.innerHTML=SWEEP_BUCKETS.map(function(b){
        return '<span class="en-chip'+(best[b.k]?" on":"")+'">'+(best[b.k]?"\u2713 ":"")+esc(b.label)+'</span>';
      }).join("");
      doneBtn.style.display = got()>0 ? "" : "none";
    }
