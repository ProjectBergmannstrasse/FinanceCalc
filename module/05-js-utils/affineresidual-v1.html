function affineResidual(p0, p1){
    var n=Math.min(p0.length, p1.length);
    if(n<12) return null;
    var Sx=0,Sy=0,Sxx=0,Sxy=0,Syy=0,Sux=0,Suy=0,Su=0,Svx=0,Svy=0,Sv=0,i;
    for(i=0;i<n;i++){
      var x=p0[i].x, y=p0[i].y, u=p1[i].x, v=p1[i].y;
      Sx+=x; Sy+=y; Sxx+=x*x; Sxy+=x*y; Syy+=y*y;
      Sux+=u*x; Suy+=u*y; Su+=u; Svx+=v*x; Svy+=v*y; Sv+=v;
    }
    function solve(Sax,Say,Sa){
      var M=[[Sxx,Sxy,Sx],[Sxy,Syy,Sy],[Sx,Sy,n]], b=[Sax,Say,Sa], a,k,j,p,f,tmp;
      for(a=0;a<3;a++){
        p=a;
        for(j=a+1;j<3;j++) if(Math.abs(M[j][a])>Math.abs(M[p][a])) p=j;
        tmp=M[a]; M[a]=M[p]; M[p]=tmp;
        tmp=b[a]; b[a]=b[p]; b[p]=tmp;
        if(Math.abs(M[a][a])<1e-9) return null;
        for(j=a+1;j<3;j++){
          f=M[j][a]/M[a][a];
          for(k=a;k<3;k++) M[j][k]-=f*M[a][k];
          b[j]-=f*b[a];
        }
      }
      var s=[0,0,0];
      for(a=2;a>=0;a--){
        var vv=b[a];
        for(k=a+1;k<3;k++) vv-=M[a][k]*s[k];
        s[a]=vv/M[a][a];
      }
      return s;
    }
    var A=solve(Sux,Suy,Su), C=solve(Svx,Svy,Sv);
    if(!A||!C) return null;
    var res=0, motion=0;
    for(i=0;i<n;i++){
      var xx=p0[i].x, yy=p0[i].y;
      res+=Math.hypot(p1[i].x-(A[0]*xx+A[1]*yy+A[2]), p1[i].y-(C[0]*xx+C[1]*yy+C[2]));
      motion+=Math.hypot(p1[i].x-xx, p1[i].y-yy);
    }
    res/=n; motion/=n;
    if(motion<0.6) return null;          // too still to judge
    return res/motion;
  }
