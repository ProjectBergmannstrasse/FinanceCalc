function drawForecast(){
    var box=document.getElementById("fcBody"); if(!box) return;
    var f=liquidityForecast(12);
    var sym=state.currency;
    if(!f.ok){
      box.innerHTML='<div class="hs-empty"><div class="he-ic">\uD83D\uDCC8</div>'
        +'<div class="he-t">Not enough history yet</div>'
        +'<div class="he-d">A forecast needs about 6 months of entries \u2014 you have '
        +f.months+'.<br>Yearly patterns need around 24.</div></div>';
      return;
    }
    var MONTHS_S=["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
    var now=new Date();
    var nameAt=function(h){ var d=new Date(now.getFullYear(), now.getMonth()+h, 1);
      return MONTHS_S[d.getMonth()]+" "+String(d.getFullYear()).slice(2); };

    var html="";
    // headline verdict
    var dry=f.dryMonth, dryW=f.dryWorst;
    var level = dry&&dry<=3 ? "bad" : dry&&dry<=6 ? "warn" : dryW&&dryW<=6 ? "warn" : "good";
    html+='<div class="dura-card '+level+'" style="margin-bottom:16px">'
      +'<div class="dura-h">'+(dry
          ? "\u26A0\uFE0F Money runs out around "+nameAt(dry)
          : dryW ? "\u26A0\uFE0F Tight if things go badly \u2014 "+nameAt(dryW)
                 : "\u2713 Cash flow looks sustainable for the next year")+'</div>'
      +'<div class="dura-d">'
      + (dry ? "On current trends, spending overtakes what you hold in about "+dry+" month"+(dry===1?"":"s")+"."
             : dryW ? "The central forecast holds up, but a run of low income and high spending would empty it by "+nameAt(dryW)+"."
                    : "Both the expected path and a pessimistic one stay positive across 12 months.")
      +'</div></div>';

    // the fan chart
    html+='<div class="fc-chart" id="fcChart"></div>';
    html+='<div class="fc-legend"><span><i class="l-mid"></i>expected</span>'
       +'<span><i class="l-band"></i>if things go badly</span>'
       +'<span><i class="l-zero"></i>empty</span></div>';

    // seasonal findings
    if(f.outModel.n>=12){
      html+='<div class="rc-sec">Seasonal pattern</div>';
      html+='<div class="fc-row"><span>Most expensive month</span><b>'
        +MONTHS_S[f.peakIdx]+' \u00b7 +'+fmt(Math.abs(f.peakAmt),sym)+' above normal</b></div>';
      html+='<div class="fc-row"><span>Cheapest month</span><b>'
        +MONTHS_S[f.troughIdx]+' \u00b7 \u2212'+fmt(Math.abs(f.troughAmt),sym)+' below normal</b></div>';
    }

    // month-by-month
    html+='<div class="rc-sec">Next 12 months</div><div class="fc-list">';
    f.path.forEach(function(p,i){
      var w=f.worstPath[i];
      var cls=p.bal<0?"bad":w.bal<0?"warn":"";
      html+='<div class="fc-item '+cls+'"><span class="fc-m">'+nameAt(p.h)+'</span>'
        +'<span class="fc-n '+(p.net>=0?"up":"down")+'">'+(p.net>=0?"+":"\u2212")+fmt(Math.abs(p.net),sym)+'</span>'
        +'<span class="fc-b">'+fmt(p.bal,sym)+'</span></div>';
    });
    html+='</div>';
    html+='<div class="rc-secd" style="margin-top:14px">Built from '+f.series.months.length
      +' months of your own entries \u2014 trend, repeating seasonal pattern, and the spread of past '
      +'variation. Further out is less certain, which is why the band widens. Not financial advice.</div>';
    box.innerHTML=html;
    drawFanChart(f);
  }
