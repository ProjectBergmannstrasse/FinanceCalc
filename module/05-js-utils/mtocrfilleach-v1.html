function mtOcrFillEach(trades){
    var rows=trades.filter(function(t){return typeof t.profit==="number"&&t.profit!==0;});
    if(!rows.length){flashSaved("no P/L amounts to import");return;}
    var noSplits=!!(state.settings&&state.settings.noSplits);
    var splitN=noSplits?1:(parseInt(document.getElementById("fSplit").value,10)||1);
    if(splitN<1) splitN=1;
    var total=mtOcrNet(rows);
    appModal({
      title:"Add "+rows.length+" separate trades?",
      message:"One entry per trade from the photo — <b>"+rows.length+"</b> entries, net <b>"
        +(total>=0?"+":"")+total.toFixed(2)+"</b>"
        +(splitN>1?"<br>Each amount split "+splitN+" ways (your share recorded).":"")
        +"<br><span class='mut'>The photo attaches to the first entry.</span>",
      confirmText:"Add them", cancelText:"Cancel",
      onConfirm:function(){
        var date=document.getElementById("fDate").value||todayStr();
        var added=0;
        rows.forEach(function(t,i){
          var amtFull=Math.abs(t.profit);
          var amt=splitN>1?Math.round((amtFull/splitN)*100)/100:amtFull;
          if(!amt) return;
          var tx={id:freshIdBase(),date:date,
                  type:(t.profit>=0?"trade_win":"trade_loss"),amount:amt};
          if(t.symbol) tx.inst=t.symbol.replace(/_+$/,"");
          var noteBits=[];
          if(t.date) noteBits.push(t.date);
          if(splitN>1) noteBits.push("Split "+splitN+" ways — "+fmt(amtFull)+" total, your share "+fmt(amt));
          if(noteBits.length) tx.note=noteBits.join(" · ");
          if(i===0&&pendingPhotos.length){
            var pids=[];
            pendingPhotos.forEach(function(pp){
              var nid=newPhotoId(tx.id,pids);pids.push(nid);
              savePhoto(nid,pp.dataUrl,null,pp.name);
            });
            setPhotos(tx,pids);
          }
          state.transactions.push(tx);
          mirrorNewTransaction(tx);
          added++;
        });
        recordAction("Imported "+added+" trades from a statement photo");
        state.selectedMonths=[monthKey(date)];
        persist();render();
        document.getElementById("fAmt").value="";document.getElementById("fNote").value="";
        document.getElementById("fInst").value="";
        clearPhotoPick();
        mtOcr.state="idle";mtOcr.result=null;mtOcrUI();
        flashSaved("added "+added+" trades");
      }
    });
  }
