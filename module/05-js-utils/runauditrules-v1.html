function runAuditRules(){
    var txs=sortedTx(), t=totals(), sym=state.currency;
    var now=new Date(), nowM=monthKey(todayStr());
    var findings=[];

    function add(level, icon, title, body){
      findings.push({level:level, icon:icon, title:title, body:body});
    }
    function fmtA(v){ return sym+Math.abs(v).toFixed(2); }

    // ── 1. Net worth health ────────────────────────────────────
    if(t.net<0) add("flag","🚨","Negative net worth",
      "Your net worth is "+fmtA(t.net)+" in the red. Debt ("+fmtA(t.debt)+") exceeds your assets.");
    else if(t.net>0) add("ok","✅","Net worth positive",
      "Net worth "+fmtA(t.net)+". Balance "+fmtA(t.balance)+" · savings "+fmtA(t.savings)+(t.debt>0?" · debt "+fmtA(t.debt):"")+".");

    // ── 2. Duplicate detection (same amount same day) ──────────
    var seen={};
    txs.forEach(function(tx){ var k=tx.date+"_"+num(tx.amount).toFixed(2)+"_"+(tx.type||"");
      seen[k]=(seen[k]||0)+1; });
    var dupes=Object.entries(seen).filter(function(e){ return e[1]>1; });
    if(dupes.length) add("warn","⚠️","Possible duplicate entries",
      dupes.length+" set"+(dupes.length>1?"s":"")+" of transactions share the same date and amount. "
      +"Check: "+dupes.map(function(d){ var p=d[0].split("_"); return sym+p[1]+" on "+p[0]; }).slice(0,3).join(", ")+".");

    // ── 3. Spending velocity (this month vs prior 3-month avg) ─
    var byMonth={};
    txs.forEach(function(tx){
      if(balDelta(tx)>=0) return;  // only outflows
      var m=tx.date?tx.date.slice(0,7):""; if(!m) return;
      byMonth[m]=(byMonth[m]||0)+Math.abs(balDelta(tx));
    });
    var months=Object.keys(byMonth).sort();
    var thisMSpend=byMonth[nowM]||0;
    var priorMonths=months.filter(function(m){ return m<nowM; }).slice(-3);
    if(priorMonths.length>=2){
      var avg=priorMonths.reduce(function(a,m){ return a+(byMonth[m]||0); },0)/priorMonths.length;
      if(avg>0&&thisMSpend>avg*1.4) add("warn","📈","Spending spike this month",
        "You've spent "+fmtA(thisMSpend)+" so far this month, "
        +Math.round((thisMSpend/avg-1)*100)+"% above your recent average of "+fmtA(avg)+"/mo.");
      else if(avg>0&&thisMSpend<avg*0.6&&new Date().getDate()>15) add("info","💡","Spending well below average",
        "Only "+fmtA(thisMSpend)+" spent this month vs an average of "+fmtA(avg)+".");
    }

    // ── 4. Category concentration ──────────────────────────────
    var byCat={};
    txs.filter(function(tx){ return tx.cat&&balDelta(tx)<0; }).forEach(function(tx){
      byCat[tx.cat]=(byCat[tx.cat]||0)+Math.abs(balDelta(tx)); });
    var catArr=Object.entries(byCat).sort(function(a,b){ return b[1]-a[1]; });
    var totalSpend=catArr.reduce(function(a,e){ return a+e[1]; },0);
    if(catArr.length&&totalSpend>0){
      var top=catArr[0], pct=Math.round(top[1]/totalSpend*100);
      if(pct>=40) add("warn","🏷️","One category dominates spending",
        top[0]+" is "+pct+"% of all categorised spending ("+fmtA(top[1])+"). "
        +(pct>=60?"Consider whether that's intentional.":"Worth watching."));
      add("info","📊","Top spending categories",
        catArr.slice(0,4).map(function(e){
          return e[0]+": "+fmtA(e[1])+" ("+Math.round(e[1]/totalSpend*100)+"%)";
        }).join("  ·  "));
    }

    // ── 5. Debt age — has any debt been sitting a long time? ───
    if(t.debt>0){
      var debtTxs=txs.filter(function(tx){ return debtDelta(tx)>0; });
      if(debtTxs.length){
        var oldest=debtTxs[0].date;
        var daysOld=Math.floor((now-new Date(oldest))/864e5);
        if(daysOld>60) add("warn","💳","Debt outstanding "+daysOld+" days",
          "Oldest unpaid debt entry is from "+oldest+". Current debt balance: "+fmtA(t.debt)+".");
      }
    }

    // ── 6. Savings rate ────────────────────────────────────────
    var totalIn=txs.reduce(function(a,tx){ return a+(balDelta(tx)>0?balDelta(tx):0); },0);
    var totalSaved=txs.reduce(function(a,tx){ return a+(savDelta(tx)>0?savDelta(tx):0); },0);
    if(totalIn>0){
      var sRate=Math.round(totalSaved/totalIn*100);
      var level=sRate>=20?"ok":sRate>=10?"info":"warn";
      add(level, sRate>=20?"🏦":sRate>=10?"💰":"💸", "Savings rate: "+sRate+"%",
        "You've saved "+fmtA(totalSaved)+" out of "+fmtA(totalIn)+" income — a "+sRate+"% rate. "
        +(sRate>=20?"Strong. Keep it up.":sRate>=10?"Decent. Aim for 20%+.":"Low. Try automating a recurring save."));
    }

    // ── 7. Round-number test (fraud / estimation signal) ───────
    var rounds=txs.filter(function(tx){ return num(tx.amount)>0&&num(tx.amount)%50===0; });
    if(rounds.length>txs.length*0.4&&txs.length>10)
      add("info","🔢","Many round-number transactions",
        rounds.length+" of "+txs.length+" entries are multiples of 50. "
        +"This can indicate estimated rather than exact figures.");

    // ── 8. Large single outflow ────────────────────────────────
    var outflows=txs.filter(function(tx){ return balDelta(tx)<0; })
      .sort(function(a,b){ return balDelta(a)-balDelta(b); });
    if(outflows.length){
      var big=outflows[0], bigAmt=Math.abs(balDelta(big));
      var avgOut=outflows.reduce(function(a,t){ return a+Math.abs(balDelta(t)); },0)/outflows.length;
      if(bigAmt>avgOut*5) add("info","🔎","Largest single outflow",
        fmtA(bigAmt)+" on "+big.date+(big.note?" ("+big.note+")":"")
        +" — "+Math.round(bigAmt/avgOut)+"× your average transaction.");
    }

    // ── 9. Positive cash-flow check ────────────────────────────
    var netFlow=txs.reduce(function(a,tx){ return a+balDelta(tx); },0);
    if(netFlow<0) add("flag","⬇️","Negative cash flow overall",
      "Total outflows exceed inflows by "+fmtA(Math.abs(netFlow))+". "
      +"Your recorded spending has consumed more than your income.");
    else if(netFlow>0&&txs.length>3) add("ok","⬆️","Positive cash flow",
      "Net cash flow is +"+fmtA(netFlow)+" across all recorded entries. Income exceeds spending.");

    if(!findings.length) add("ok","✅","Nothing flagged","Your financial data looks clean. Add more transactions for deeper analysis.");
    return findings;
  }
