function renderTrash(){
    var box=document.getElementById("trashList");
    if(!box) return;
    var items=trashList(), sym=state.currency;
    var btns=document.getElementById("trashBtns");
    if(btns) btns.style.display=items.length?"":"none";
    if(!items.length){
      box.innerHTML='<div class="hs-empty"><div class="he-ic">\uD83D\uDDD1\uFE0F</div>'
        +'<div class="he-t">Nothing deleted recently</div>'
        +'<div class="he-d">Deleted entries wait here for '+TRASH_DAYS+' days<br>before they\u2019re cleared.</div></div>';
      return;
    }
    /* Same card shape as the action list, grouped by the day it was deleted,
       showing the amount and how long is left — the old version buried all
       of that in one grey line. */
    var html="", lastDay="";
    items.slice(0,60).forEach(function(item,i){
      var t=item.tx||{}, meta=TYPE_META[t.type]||{label:t.type||"?"};
      var nm=(t.note||t.cat||meta.label||"Entry");
      var left=trashDaysLeft(item);
      var when = left<=0 ? "going now" : (left===1 ? "1 day left" : left+" days left");
      var day=dayHeading(item.at||Date.now());
      if(day!==lastDay){ html+='<div class="hs-day">deleted '+esc(day)+'</div>'; lastDay=day; }
      var delta=balDelta(t);
      var pics=photosOf(t).length;
      html+='<div class="hs-card" style="animation-delay:'+Math.min(i*22,220)+'ms">'
        +'<div class="hs-ic del">'+(meta.label||"?").charAt(0)+'</div>'
        +'<div class="hs-main"><div class="hs-t">'+esc(nm)+'</div>'
        +'<div class="hs-sub"><span>'+esc(t.date||"no date")+'</span>'
        +'<span class="hs-dot"></span><span>'+esc(meta.label)+'</span>'
        + (pics?'<span class="hs-dot"></span><span>\uD83D\uDCCE'+(pics>1?pics:"")+'</span>':"")
        +'<span class="hs-pill'+(left<=5?" soon":"")+'">'+esc(when)+'</span>'
        +'</div></div>'
        +'<div class="hs-amt '+(delta>0?"in":delta<0?"out":"")+'">'
        + (delta>0?"+":delta<0?"\u2212":"")+fmt(Math.abs(num(t.amount)),sym)+'</div>'
        +'<button type="button" class="hs-btn tr-item-btn" data-i="'+i+'">Restore</button>'
        +'</div>';
    });
    if(items.length>60) html+='<div class="hs-empty" style="padding:14px"><div class="he-d">\u2026and '+(items.length-60)+' more</div></div>';
    box.innerHTML=html;
    Array.prototype.forEach.call(box.querySelectorAll(".tr-item-btn"), function(b){
      b.addEventListener("click", function(){
        var tx=restoreFromTrash(+b.getAttribute("data-i"));
        if(tx){
          resetLedgerView();
          render(); renderTrash();
          flashSaved("restored");
        }
      });
    });
  }
