function aiPrompt(opts){
    var cats=aiCats().join(", ");
    var p=[
"You are reading images of receipts, bank/trading screenshots or handwritten notes, and turning them into ledger entries.",
"",
"Return ONLY a JSON object. No prose, no markdown, no code fences. Exact shape:",
'{"transactions":[{"date":"YYYY-MM-DD","type":"<type>","amount":<positive number>,"cat":"<category or empty>","note":"<short note>","confidence":<0..1>}]}',
"",
"Allowed type values (use EXACTLY one of these):",
AI_TYPES.join(", "),
"",
"What each type means:",
"- income: money received (not an allowance)",
"- spend: money spent on something",
"- trade_win / trade_loss: profit or loss on a trade",
"- pay_debt_balance: paying down debt out of your own money",
"- borrow: debt went up",
"- save / withdraw: moving money into / out of savings",
"- tax_set: money parked aside for tax",
"- tax_pay: paying a tax bill out of money already set aside",
"- adjust: a correction; only this type may be negative",
"",
"Rules:",
"- amount is ALWAYS a positive number, except for 'adjust' which may be negative.",
"- Never include a currency symbol or thousands separators in amount. 1,234.50 -> 1234.50",
"- date must be YYYY-MM-DD. If the image shows no year, assume "+todayStr().slice(0,4)+". If no date at all, use "+todayStr()+".",
"- cat only applies to 'spend'. Choose from: "+cats+". Otherwise use an empty string.",
"- note: short and factual (merchant, instrument, what it was). No invented detail.",
"- confidence: your honest 0..1 read of how sure you are of THIS entry.",
"- Read every image. One image may hold several transactions, or none.",
"- If an image contains nothing financial, contribute no entries for it.",
"- Do not merge separate transactions, and do not split one transaction into several."
    ];
    if(opts.expected>0){
      p.push("- The user states there are EXACTLY "+opts.expected+" transactions across these images. Return exactly "+opts.expected+". If you can only find fewer, still return "+opts.expected+" and mark the uncertain ones with low confidence.");
    }
    if(opts.conflict){
      p.push("",
"IMPORTANT — CROSS-CHECK CONTEXT:",
"A second, independent AI was given these exact images and the exact same instructions, and returned a DIFFERENT answer. That means at least one of you misread something. This is attempt "+opts.attempt+".",
"",
"Your previous answer:", opts.mine,
"",
"The other model's answer:", opts.theirs,
"",
"Re-examine the images from scratch. Do not simply repeat your previous answer, and do not simply defer to the other model — either of you may be wrong. Look specifically at the fields that differ (dates, amounts, and types). Return your best, carefully re-checked answer in the same JSON format.");
    }
    return p.join("\n");
  }
