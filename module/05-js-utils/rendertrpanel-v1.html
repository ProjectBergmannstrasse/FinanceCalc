function renderTrPanel(){
    var sym=state.currency;
    // instrument filter options rebuild from the full list, not the filtered one
    var sel=document.getElementById("trInst");
    var prev=sel.value;
    var insts=tradeByInstrument(tradeList()).map(function(r){ return r.inst; });
    sel.innerHTML='<option value="__all">All instruments</option>'
      + insts.map(function(i){ return '<option value="'+esc(i)+'">'+esc(i)+'</option>'; }).join("");
    sel.value = insts.indexOf(prev)>=0 ? prev : "__all";

    var list=trScopedList();
    var box=document.getElementById("trBody");
    if(!list.length){
      box.innerHTML='<div class="empty"><b>No trades yet.</b><br>Log one with <b>Trade — made money</b> or <b>Trade — lost money</b>. Add an instrument (like XAUUSD) and you\'ll get a per-instrument breakdown here.</div>';
      return;
    }
    var s=tradeStats(list);
    var pfTxt = s.profitFactor===Infinity ? "∞" : s.profitFactor.toFixed(2);
    var wrCls = s.winRate>=0.5?"up":"down";
    var html='<div class="tr-hero">'
      +'<div class="tr-wr '+wrCls+'">'+(s.winRate*100).toFixed(1)+'%</div>'
      +'<div class="tr-wr-sub">win rate · '+s.wins+'W / '+s.losses+'L over '+s.n+' trade'+(s.n===1?"":"s")+'</div>'
      +'</div>';
    html+=equityCurveSvg(s.curve, sym);
    html+='<div class="tr-grid">'
      +trStat("Net P&amp;L", fmt(s.net,sym), s.net>=0?"up":"down")
      +trStat("Expectancy / trade", fmt(s.expectancy,sym), s.expectancy>=0?"up":"down")
      +trStat("Profit factor", pfTxt, s.profitFactor>=1?"up":"down")
      +trStat("Max drawdown", fmt(s.maxDD,sym), s.maxDD>0?"down":"")
      +trStat("Avg win", fmt(s.avgWin,sym), "up")
      +trStat("Avg loss", fmt(s.avgLoss,sym), "down")
      +trStat("Best trade", fmt(s.best,sym), "up")
      +trStat("Worst trade", fmt(s.worst,sym), "down")
      +trStat("Longest win streak", s.bestWinStreak)
      +trStat("Longest loss streak", s.bestLossStreak)
      +'</div>';

    // Expectancy is the honest headline — say so when win rate flatters a losing system.
    if(s.winRate>=0.5 && s.expectancy<0){
      html+='<div class="ai-banner warn" style="margin-top:10px">⚠️ You win more often than you lose, but expectancy is <b>negative</b> — the losses are bigger than the wins. Win rate is flattering this.</div>';
    } else if(s.winRate<0.5 && s.expectancy>0){
      html+='<div class="ai-banner good" style="margin-top:10px">You lose more often than you win, but expectancy is <b>positive</b> — the winners are big enough to carry it.</div>';
    }

    if(sel.value==="__all"){
      var rows=tradeByInstrument(list);
      if(rows.length>1){
        html+='<div class="tr-sec">By instrument</div><div class="tr-tbl">';
        rows.forEach(function(r){
          var st=r.stats;
          html+='<div class="tr-row">'
            +'<div class="tr-i">'+esc(r.inst)+'<span class="tr-n">'+st.n+' trade'+(st.n===1?"":"s")+'</span></div>'
            +'<div class="tr-wr2 '+(st.winRate>=0.5?"up":"down")+'">'+(st.winRate*100).toFixed(0)+'%</div>'
            +'<div class="tr-pl '+(st.net>=0?"up":"down")+'">'+fmt(st.net,sym)+'</div>'
            +'</div>';
        });
        html+='</div>';
      }
    }
    box.innerHTML=html;
  }
