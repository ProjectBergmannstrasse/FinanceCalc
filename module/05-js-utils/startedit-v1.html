    if(idx<0) return;

    state.transactions.splice(idx,1);

  function finalizePendingDelete(){
    if(!pendingDelete) return;
    if(pendingDelete.timer) clearTimeout(pendingDelete.timer);
    // the 5s toast expiring means "they didn't undo", not "destroy it" — it goes to the
    // bin, photo intact, and only leaves for good after TRASH_DAYS
    toTrash(pendingDelete.tx);
    pendingDelete=null;
    persistNow();
    hideUndoToast();
  }
  function undoDelete(){
    if(!pendingDelete) return;
    if(pendingDelete.timer) clearTimeout(pendingDelete.timer);
    state.transactions.push(pendingDelete.tx);
    pendingDelete=null;
    persist(); render();
    hideUndoToast();
  }

  /* ---- ledger search: matches note, category, and the type's display label ---- */
  var searchQuery="";
  var sortMode="latest";
  /* Words for things that aren't written anywhere on the entry.
     "untagged" is the obvious one: the app SAYS "— untagged —" on the Trading page, so
     it's the natural thing to search — but an untagged trade has no instrument, so
     there's literally no text to match. Same for "trade": you'd have to know the type
     label happens to read "TRADE +".
     These match IN ADDITION to the normal text search, so a note that genuinely says
     "trade" still comes back. And they're whole-word, or "tagged" would match
     "untagged" by substring and mean the opposite of what you asked. */
  var SEARCH_WORDS={
    trade:      function(t){ return isTradeTx(t); },
    trades:     function(t){ return isTradeTx(t); },
    untagged:   function(t){ return isTradeTx(t) && !t.inst; },
    tagged:     function(t){ return isTradeTx(t) && !!t.inst; },
    win:        function(t){ return t.type==="trade_win"; },
    wins:       function(t){ return t.type==="trade_win"; },
    loss:       function(t){ return t.type==="trade_loss"; },
    losses:     function(t){ return t.type==="trade_loss"; },
    photo:      function(t){ return photosOf(t).length>0; },
    note:       function(t){ return !!(t.note && t.note.trim()); }
  };
  function isTradeTx(t){ return t.type==="trade_win" || t.type==="trade_loss"; }
  function matchesSearch(t){
    if(!searchQuery) return true;
    var q=searchQuery.trim();
    if(SEARCH_WORDS[q] && SEARCH_WORDS[q](t)) return true;
    var meta=TYPE_META[t.type]||{label:t.type};
    var hay=((t.note||"")+" "+(t.cat||"")+" "+(t.inst||"")+" "+(meta.label||"")+" "+(t.date||"")).toLowerCase();
    return hay.indexOf(searchQuery)>=0;
  }
  // Profit/loss here means effect on your balance — money in is profit, money out is a
  // loss. Debt- and savings-only entries don't move the balance, so they sort as 0 and
  // settle at the boundary between the two rather than being dropped.
  function applySort(list){
    var l=list.slice();
    if(sortMode==="latest") return l; // already newest-first from the caller
    if(sortMode==="oldest") return l.reverse();
    function pl(t){ return balDelta(t); }
    function amt(t){ var d=balDelta(t); return d!==0 ? Math.abs(d) : Math.max(Math.abs(debtDelta(t)), Math.abs(savDelta(t))); }
    if(sortMode==="profit") l.sort(function(a,b){ return pl(b)-pl(a); });
    else if(sortMode==="loss") l.sort(function(a,b){ return pl(a)-pl(b); });
    else if(sortMode==="biggest") l.sort(function(a,b){ return amt(b)-amt(a); });
    else if(sortMode==="smallest") l.sort(function(a,b){ return amt(a)-amt(b); });
    return l;
  }

  /* ================= RECENT ACTIONS =================
     A snapshot of the whole portfolio taken BEFORE each change, so any of them can be
     put back. Snapshots rather than inverse operations: an inverse has to be written
     per action type and is wrong exactly when it matters, whereas a snapshot restores
     whatever was actually there.

     In memory only, and deliberately: persisting a dozen copies of the ledger risks the
     storage quota that holds the real data. Undo covers this session — the trash bin and
     the automatic backups cover the rest.

     Undoing an older change necessarily undoes the newer ones too, because state goes
     back to how it was before it. The confirm says so rather than surprising anyone. */
  var ACTIONS=[], ACTIONS_MAX=12;
  function recordAction(label){
    try{
      ACTIONS.unshift({ label:label, at:Date.now(), snap:JSON.stringify(state) });
      if(ACTIONS.length>ACTIONS_MAX) ACTIONS.pop();
    }catch(e){ return; }   // a snapshot that can't be taken must never block the action
    renderActions();
  }
