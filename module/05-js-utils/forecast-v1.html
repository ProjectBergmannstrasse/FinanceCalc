function forecast(days){
    var start=totals().balance;
    var burn=dailyBurn();
    var events=knownEvents(days);
    var byDate={};
    events.forEach(function(e){ byDate[e.date]=(byDate[e.date]||0)+e.amount; });
    var series=[], bal=start, low={bal:start, date:todayStr()}, zeroOn=null;
    var d=new Date();
    for(var i=0;i<=days;i++){
      var ds=d.getFullYear()+"-"+two(d.getMonth()+1)+"-"+two(d.getDate());
      if(i>0){
        bal-=burn.perDay;
        if(byDate[ds]) bal+=byDate[ds];
      }
      series.push({date:ds, bal:bal, event:byDate[ds]||0});
      if(bal<low.bal){ low={bal:bal, date:ds}; }
      if(bal<0 && !zeroOn) zeroOn=ds;
      d.setDate(d.getDate()+1);
    }
    return {start:start, series:series, events:events, burn:burn, low:low, zeroOn:zeroOn, end:series[series.length-1].bal};
  }
