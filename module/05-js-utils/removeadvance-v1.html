function removeAdvance(forMonth){
    var txs=state.transactions.filter(function(t){ return t.advance && t.forMonth===forMonth; });
    if(!txs.length) return;
    var total=0; txs.forEach(function(t){ total+=num(t.amount); });
    var n=txs.length;
    appModal({
      title:"Remove the advance for "+monthLong(forMonth)+"?",
      message:"That deletes <b>"+n+"</b> linked "+(n===1?"entry":"entries")+" totalling <b>"+fmt(total)+"</b> and frees the advance slot again.\n\nThey go to Recently deleted, so this is reversible.",
      confirmText:"Remove", danger:true,
      onConfirm:function(){
        var ids=txs.map(function(t){ return t.id; });
        txs.forEach(toTrash);   // recoverable, like any other delete
        state.transactions=state.transactions.filter(function(t){ return ids.indexOf(t.id)<0; });
        ids.forEach(function(id){ delete selectedIds[id]; });
        persistNow(); render(); flashSaved("advance removed");
      }
    });
  }
