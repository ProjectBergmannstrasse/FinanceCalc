function tradeSim(list, days, runs){
    var s=tradeStats(list), rate=tradeRate(list);
    var n=Math.round(rate.perDay*days);
    if(n<1 || !list.length) return null;
    var wins=[], losses=[];
    list.forEach(function(t){
      var a=Math.abs(num(t.amount));
      if(t.type==="trade_win") wins.push(a); else losses.push(a);
    });
    if(!wins.length && !losses.length) return null;
    runs=runs||1000;
    var outs=[], down=0, worstRun=0;
    for(var r=0;r<runs;r++){
      var tot=0, eq=0, peak=0, dd=0;
      for(var i=0;i<n;i++){
        var win = Math.random() < s.winRate;
        var pool = win ? wins : losses;
        if(!pool.length) continue;
        // resample real results rather than assume a bell curve — his tail IS the point
        var v = pool[Math.floor(Math.random()*pool.length)] * (win?1:-1);
        tot+=v; eq+=v;
        if(eq>peak) peak=eq;
        if(peak-eq>dd) dd=peak-eq;
      }
      if(tot<0) down++;
      if(dd>worstRun) worstRun=dd;
      outs.push(tot);
    }
    outs.sort(function(a,b){ return a-b; });
    var q=function(p){ return outs[Math.min(outs.length-1, Math.floor(p*outs.length))]; };
    return {
      n:n, runs:runs, median:q(0.5), p10:q(0.1), p90:q(0.9),
      best:outs[outs.length-1], worst:outs[0],
      chanceDown:down/runs, typicalDD:worstRun, outs:outs
    };
  }
