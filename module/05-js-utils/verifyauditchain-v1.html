async function verifyAuditChain(){
    var stored=state.auditChain;
    if(!Array.isArray(stored)||!stored.length){
      return {status:"none", message:"No audit chain recorded yet. Seal your records to start one."};
    }
    var txs=sortedTx();
    var prev="0".repeat(64);
    var firstBreak=null, checked=0;
    for(var i=0;i<stored.length;i++){
      var tx=txs.filter(function(x){ return String(x.id)===String(stored[i].id); })[0];
      if(!tx){ firstBreak={idx:i, reason:"entry deleted", id:stored[i].id, date:stored[i].date}; break; }
      var h=await sha256Hex(prev+canonicalTx(tx));
      checked++;
      if(h!==stored[i].hash){ firstBreak={idx:i, reason:"entry modified", id:stored[i].id, date:stored[i].date}; break; }
      prev=h;
    }
    var added=txs.length-stored.length;
    if(firstBreak){
      return {status:"broken", message:"Chain breaks at entry "+(firstBreak.idx+1)+" of "+stored.length
        +" ("+firstBreak.reason+", dated "+(firstBreak.date||"?")+"). Everything before it is still verified.",
        verified:firstBreak.idx, total:stored.length};
    }
    return {status:"ok", verified:checked, total:stored.length, added:added>0?added:0,
      message:checked+" sealed entr"+(checked===1?"y":"ies")+" verified intact"
        +(added>0?" · "+added+" new entr"+(added===1?"y":"ies")+" not yet sealed":"")+"."};
  }
