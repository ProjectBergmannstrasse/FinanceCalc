function maybeSuggestTax(tx){
    var pct=num(state.settings.taxPct);
    if(pct<=0) return;
    if(tx.type!=="income" && tx.type!=="trade_win") return;
    var amt=Math.round(num(tx.amount)*pct)/100;
    if(amt<0.01) return;
    appModal({
      title:"Set aside for tax?",
      message:"You've set a tax rate of <b>"+pct+"%</b>. On "+fmt(num(tx.amount))+" that's <b>"+fmt(amt)+"</b>.\n\nSetting it aside moves it out of your spendable balance into the tax pot. It won't count toward net worth.",
      confirmText:"Set aside "+fmt(amt),
      cancelText:"Skip",
      onConfirm:function(){
        var t={id:freshIdBase(), date:tx.date, type:"tax_set", amount:amt, note:"Tax "+pct+"% of "+fmt(num(tx.amount))};
        state.transactions.push(t);
        mirrorNewTransaction(t);
        persistNow(); render();
        flashSaved("tax set aside");
      }
    });
  }
