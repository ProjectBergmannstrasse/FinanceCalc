function openTxDetail(id){
    var t=state.transactions.filter(function(x){ return x.id===id; })[0];
    if(!t) return;
    var sym=state.currency, meta=TYPE_META[t.type]||{label:t.type};
    var d=String(t.date||"").split("-");
    var full=(MONTHS[(+d[1])-1]||"")+" "+(+d[2])+", "+d[0];
    var ov=document.createElement("div");
    ov.className="xsel-ov td-ov";
    var bits="";
    if(t.cat) bits+='<div class="td-row"><span>Category</span><b>'+esc(t.cat)+'</b></div>';
    if(t.inst) bits+='<div class="td-row"><span>Instrument</span><b>'+esc(t.inst)+'</b></div>';
    else if(t.type==="trade_win"||t.type==="trade_loss")
      bits+='<div class="td-row"><span>Instrument</span><b class="mut">\u2014 untagged \u2014</b></div>';
    if(t.auto) bits+='<div class="td-row"><span>Added by</span><b class="mut">Automatically</b></div>';
    ov.innerHTML='<div class="xsel-sheet td-sheet">'
      +'<div class="td-head">'
        +'<div class="td-badge">'+esc(meta.label)+'</div>'
        +'<div class="td-amt mono">'+fmt(Math.abs(num(t.amount)),sym)+'</div>'
        +'<div class="td-date">'+esc(full)+'</div>'
      +'</div>'
      +'<div class="td-body">'
        +'<div class="td-sec">What it moved</div>'+txPotMoves(t,sym)
        +(bits?'<div class="td-sec">Details</div>'+bits:"")
        +(t.note?'<div class="td-sec">Note</div><div class="td-note">'+esc(t.note)+'</div>':"")
        +(photosOf(t).length?'<div class="td-sec">'+(photosOf(t).length>1?photosOf(t).length+" files":"File")
            +'</div><div class="td-photo" id="tdPhoto">loading\u2026</div>':"")
      +'</div>'
      +'<div class="td-foot">'
        +'<button type="button" class="btn ghost sm td-copy" id="tdCopy" title="Copy transaction">📋 Copy</button>'
        +(t.auto?'<div class="hint" style="margin:0;flex:1">Managed by the allowance tools \u2014 edit it there.</div>'
                :'<button type="button" class="btn ghost sm" id="tdDel">Delete</button>'
                 +'<button type="button" class="btn sm" id="tdEdit">Edit</button>')
        +'<button type="button" class="btn ghost sm" id="tdClose">Close</button>'
      +'</div></div>';
    document.body.appendChild(ov);
    requestAnimationFrame(function(){ ov.classList.add("in"); });
    var close=function(){
      ov.classList.remove("in");
      document.removeEventListener("keydown", onKey);
      setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); },140);
    };
    var onKey=function(e){ if(e.key==="Escape"){ e.preventDefault(); close(); } };
    document.addEventListener("keydown", onKey);
    ov.addEventListener("click", function(e){ if(e.target===ov) close(); });
    ov.querySelector("#tdClose").addEventListener("click", close);
    var cp=ov.querySelector("#tdCopy");
    if(cp) cp.addEventListener("click", function(){ openCopyMenu(t); });
    var ed=ov.querySelector("#tdEdit");
    if(ed) ed.addEventListener("click", function(){ close(); startEdit(t.id); });
    var dl=ov.querySelector("#tdDel");
    if(dl) dl.addEventListener("click", function(){
      close();
      if(editingId===t.id) cancelEdit();   // don't leave a deleted entry loaded in the form
      deleteWithUndo(t.id);                // same path as the row's bin icon, undo included
    });
    var pics=photosOf(t);
    if(pics.length){
      var box=ov.querySelector("#tdPhoto");
      if(box){
        box.innerHTML="";
        box.classList.toggle("td-photo-many", pics.length>1);
        pics.forEach(function(pid,i){
          var cell=document.createElement("div");
          cell.className="td-pcell";
          box.appendChild(cell);
          loadPhoto(pid, function(url){
            if(!url){ cell.innerHTML='<div style="color:var(--mut);padding:8px;font-size:11px">file missing — re-attach</div>'; return; }
            var nm=attNames[pid]||"", kind=attKind(nm,url);
            if(kind==="image"){
              cell.innerHTML='<img src="'+url+'" alt="Attachment '+(i+1)+'">';
              cell.querySelector("img").addEventListener("click", function(){ close(); openViewer(t.id); });
            } else if(kind==="audio"){
              cell.classList.add("td-media");
              cell.innerHTML='<div class="td-fname">'+attIcon(nm,url)+' '+esc(nm||"audio file")+'</div><audio src="'+url+'" controls></audio>';
            } else if(kind==="video"){
              cell.classList.add("td-media");
              cell.innerHTML='<div class="td-fname">'+attIcon(nm,url)+' '+esc(nm||"video file")+'</div><video src="'+url+'" controls playsinline></video>';
            } else if(kind==="pdf"){
              cell.classList.add("td-media");
              cell.innerHTML='<embed type="application/pdf" src="'+url+'" class="td-pdf">'
                +'<button type="button" class="btn ghost sm td-open">Open '+esc(nm||"PDF")+'</button>';
              cell.querySelector(".td-open").addEventListener("click", function(){ attDownload(url,nm||"file.pdf"); });
            } else {
              cell.classList.add("td-file");
              cell.innerHTML='<span class="td-fic">'+attIcon(nm,url)+'</span>'
                +'<span class="td-fnm">'+esc(nm||"file")+'</span>'
                +'<button type="button" class="btn ghost sm td-open">Download</button>';
              cell.querySelector(".td-open").addEventListener("click", function(){ attDownload(url,nm||"file"); });
            }
          });
        });
      }
    }
  }
