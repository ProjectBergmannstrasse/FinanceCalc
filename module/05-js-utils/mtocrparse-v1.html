function mtOcrParse(words){
    if(!words.length) return {trades:[],isStatement:false};
    var rowsData=mtOcrRows(words)
      .map(function(r){return r.slice().sort(function(a,b){return a.x-b.x;});})
      .sort(function(a,b){
        var ay=Math.min.apply(null,a.map(function(w){return w.y;}));
        var by=Math.min.apply(null,b.map(function(w){return w.y;}));
        return ay-by;
      })
      .map(function(r){return{words:r,cells:mtOcrCells(r)};})
      .filter(function(rd){return rd.cells.length;});

    var headerRow=null, headerMap=[];
    rowsData.forEach(function(rd){
      var m=rd.cells.map(function(c){
        var f=MTOCR_HEADERS.filter(function(p){return p.re.test(c.text);});
        return f.length?f[0].key:null;
      });
      if(m.filter(Boolean).length>=3 && !headerRow){headerRow=rd.cells;headerMap=m;}
    });
    var trades=[];
    if(headerRow){
      rowsData.forEach(function(rd){
        var cells=rd.cells;
        if(cells===headerRow) return;
        if(!cells.some(function(c){return MTOCR_NUM.test(c.text.replace(/,/g,""));})) return;
        var t={symbol:null,type:null,volume:null,date:null,profit:null};
        var seenTime=0;
        cells.forEach(function(cell){
          var best=null,bd=Infinity;
          for(var i=0;i<headerRow.length;i++){
            var d=Math.abs(cell.x-headerRow[i].x);
            if(d<bd){bd=d;best=headerMap[i];}
          }
          if(!best) return;
          var raw=cell.text.trim(), num=parseFloat(raw.replace(/,/g,""));
          if(best==="symbol"&&!/^\d+$/.test(raw)) t.symbol=t.symbol||raw;
          else if(best==="type"&&MTOCR_DIR.test(raw)) t.type=raw.toLowerCase();
          else if(best==="volume"&&!isNaN(num)) t.volume=num;
          else if(best==="time"){seenTime++;if(seenTime===1)t.date=raw;}
          else if(best==="profit"&&!isNaN(num)) t.profit=num;
          else if(best==="id"&&/^\d{4,}$/.test(raw)) t.ticket=raw;   // v1.99 Stage 2
        });
        if(t.symbol||t.type||t.profit!=null) trades.push(t);
      });
    } else {
      rowsData.forEach(function(rd){
        var cells=rd.cells, dirIdx=-1;
        for(var i=0;i<cells.length;i++){if(MTOCR_DIR.test(cells[i].text.trim())){dirIdx=i;break;}}
        if(dirIdx===-1) return;
        var t={symbol:null,type:cells[dirIdx].text.trim().toLowerCase(),volume:null,date:null,profit:null};
        var timeCells=cells.filter(function(c){return MTOCR_TIME.test(c.text)&&!/^\d+$/.test(c.text.trim());});
        if(timeCells[0]) t.date=timeCells[0].text.trim();
        for(var j=dirIdx+1;j<cells.length;j++){
          if(mtOcrInstrument(cells[j].text.trim())){t.symbol=cells[j].text.trim().replace(/[,._]+$/,"").toUpperCase();break;}
        }
        var volCell=null;
        for(var k=dirIdx+1;k<cells.length;k++){
          var r2=cells[k].text.trim().replace(/,/g,"");
          if(MTOCR_NUM.test(r2)&&!MTOCR_INT.test(r2)){var v=parseFloat(r2);if(v<1000){t.volume=v;volCell=cells[k];}break;}
        }
        var nums=cells.filter(function(c){
          var raw=c.text.trim().replace(/,/g,"");
          return MTOCR_NUM.test(raw)&&!MTOCR_INT.test(raw)&&c!==volCell;
        });
        if(nums.length) t.profit=parseFloat(nums[nums.length-1].text.trim().replace(/,/g,""));
        // v1.98 STRICT WHITELIST: a header-less row only counts as a trade if it has a
        // REAL instrument (mtOcrInstrument) AND a buy/sell direction. A bare number grid
        // (e.g. an Apple Notes table) has neither -> "not MT4/MT5, ignored". This
        // deliberately replaces the old looser test that accepted 3 numbers alone.
        var looks=(!!t.symbol)&&(!!t.type);
        if(looks) trades.push(t);
      });
      if(!trades.length){
        var used={};
        for(var ri=0;ri<rowsData.length;ri++){
          if(used[ri]) continue;
          var wa=rowsData[ri].words;
          var ta=wa.map(function(w){return w.text;}).join(" ");
          if(!/\b(buy|sell)\b/i.test(ta)) continue;
          if(!wa.some(function(w){return mtOcrInstrument(w.text.trim());})) continue;
          var rb=rowsData[ri+1];
          if(!rb) continue;
          if(/\b(buy|sell)\b/i.test(rb.words.map(function(w){return w.text;}).join(" "))) continue;
          var sa=mtOcrStitch(wa), sb=mtOcrStitch(rb.words);
          var taJ=sa.map(function(w){return w.text;}).join(" ");
          var tbJ=sb.map(function(w){return w.text;}).join(" ");
          var sym=wa.filter(function(w){return mtOcrInstrument(w.text.trim());})[0];
          var dir=taJ.match(/\b(buy|sell)\b/i);
          var dm=taJ.match(/\d{4}[.\-]\d{2}[.\-]\d{2}(\s+\d{1,2}:\d{2}(:\d{2})?)?/);
          var da=taJ.match(/-?\d+\.\d+/g)||[];
          var db=tbJ.match(/-?\d+\.\d+/g)||[];
          if(!db.length) continue;
          var t2={symbol:sym?sym.text.replace(/[,._]+$/,"").toUpperCase():null,
                  type:dir?dir[1].toLowerCase():null,
                  volume:da.length?parseFloat(da[0]):null,
                  date:dm?dm[0]:null,
                  profit:db.length>2?parseFloat(db[2]):null};
          if(t2.symbol&&t2.type){trades.push(t2);used[ri+1]=true;}
        }
      }
    }
    return {trades:trades,isStatement:trades.length>0};
  }
