function renderHeatmap(){
    var card=document.getElementById("hmCard");
    if(!card) return;
    var cells=hmGrid(HM_WEEKS), sc=hmScale(cells);
    if(!sc.spentDays){ card.style.display="none"; return; }   // nothing spent = nothing to show
    card.style.display="";
    document.getElementById("hmChart").innerHTML=hmSvg(cells, sc);
    var first=cells[0].date, last=cells[cells.length-1].date;
    document.getElementById("hmSpan").textContent=fcShortDate(first)+" – "+fcShortDate(last);

    Array.prototype.forEach.call(document.querySelectorAll("#hmChart .hm-cell"), function(r){
      r.addEventListener("click", function(){
        var ds=r.getAttribute("data-d");
        hmSel=(hmSel===ds)?null:ds;
        renderHeatmap();
      });
    });

    var pick=document.getElementById("hmPick");
    if(hmSel){
      var day=state.transactions.filter(function(t){ return t.date===hmSel && balDelta(t)<0; });
      var tot=day.reduce(function(n,t){ return n+(-balDelta(t)); },0);
      pick.style.display="";
      pick.innerHTML='<button type="button" class="hm-pick-x" id="hmPickX" aria-label="Close">✕</button>'
        +'<b>'+esc(niceDate(hmSel))+'</b> — '+(day.length?('<b>'+fmt(tot,state.currency)+'</b> across '+day.length+' entr'+(day.length===1?"y":"ies")+'<br>'
          +day.slice(0,4).map(function(t){ return '· '+esc((t.note||t.cat||TYPE_META[t.type].label))+' — '+fmt(Math.abs(num(t.amount)),state.currency); }).join("<br>")
          +(day.length>4?"<br>· …and "+(day.length-4)+" more":"")):"nothing went out.");
      document.getElementById("hmPickX").addEventListener("click", function(){ hmSel=null; renderHeatmap(); });
    } else pick.style.display="none";

    var by=hmByWeekday(cells);
    var el=document.getElementById("hmInsight");
    if(by.length>=3){
      var top=by[0];
      var days=["Sundays","Mondays","Tuesdays","Wednesdays","Thursdays","Fridays","Saturdays"];
      el.innerHTML="<b>"+days[top.dow]+"</b> cost you the most — <b>"+fmt(top.total,state.currency)+"</b> across "+top.n+" of them, averaging <b>"+fmt(top.avg,state.currency)+"</b> each. "
        +"You spent something on <b>"+sc.spentDays+"</b> of the last "+cells.length+" days.";
    } else el.innerHTML="You spent something on <b>"+sc.spentDays+"</b> of the last "+cells.length+" days.";
  }
