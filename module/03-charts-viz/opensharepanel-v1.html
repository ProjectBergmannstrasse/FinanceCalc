function openSharePanel(){
    openOverlay(
      '<div class="sheet"><div class="sheet-hd"><h3>👨‍👩‍👦 Family sharing</h3><button class="x-btn" id="spX">✕</button></div>'
      +'<div class="sheet-bd">'
        +'<div class="mut" style="font-size:12px;margin-bottom:12px;line-height:1.65">'
          +'Generate a cryptographically signed access token. The recipient '
          +'pastes it below or opens the shared link — their role is enforced locally.</div>'
        +'<div class="fld"><label>Role</label><select class="inp" id="spRole">'
          +'<option value="editor">Editor — can add entries, not delete or change settings</option>'
          +'<option value="viewer">Viewer — read only, no changes</option></select></div>'
        +'<div class="fld"><label>Label (who is this for?)</label>'
          +'<input class="inp" id="spLabel" placeholder="e.g. Spouse, Accountant"></div>'
        +'<div class="fld"><label>Expires (days)</label>'
          +'<input class="inp" id="spExp" type="number" value="30" min="1" max="365"></div>'
        +'<div id="spOut" style="display:none;margin-top:10px">'
          +'<label style="font-size:11px;color:var(--mut);font-weight:700;display:block;margin-bottom:6px">Access token (copy and send)</label>'
          +'<textarea class="inp" id="spToken" rows="3" readonly style="font-size:11px;font-family:monospace;word-break:break-all"></textarea>'
          +'<button class="btn sm" id="spCopy" style="margin-top:8px;width:100%">📋 Copy token</button></div>'
        +'<div style="margin-top:14px;border-top:1px solid var(--bd);padding-top:12px">'
          +'<label style="font-size:11px;color:var(--mut);font-weight:700;display:block;margin-bottom:6px">Or paste a received token</label>'
          +'<div style="display:flex;gap:8px"><input class="inp" id="spPaste" placeholder="paste token here">'
          +'<button class="btn sm" id="spActivate">Activate</button></div></div>'
      +'</div>'
      +'<div class="sheet-ft"><button class="btn sm" id="spCancel">Close</button>'
        +'<button class="btn sm solid" id="spGen">Generate token</button></div></div>',
      {ready:function(ov,close){
        ov.querySelector("#spX").onclick=close;
        ov.querySelector("#spCancel").onclick=close;
        ov.querySelector("#spGen").onclick=async function(){
          var btn=ov.querySelector("#spGen"); btn.disabled=true; btn.textContent="Signing…";
          var role=ov.querySelector("#spRole").value;
          var label=ov.querySelector("#spLabel").value.trim();
          var exp=parseInt(ov.querySelector("#spExp").value,10)||30;
          var token=await createShareToken(role,label,exp);
          ov.querySelector("#spToken").value=token;
          ov.querySelector("#spOut").style.display="";
          btn.disabled=false; btn.textContent="Generate token";
        };
        ov.querySelector("#spCopy").onclick=function(){
          var t=ov.querySelector("#spToken").value; if(!t) return;
          if(navigator.clipboard&&navigator.clipboard.writeText) navigator.clipboard.writeText(t).then(function(){ flashSaved("token copied"); });
          else{ ov.querySelector("#spToken").select(); document.execCommand("copy"); flashSaved("copied"); }
        };
        ov.querySelector("#spActivate").onclick=async function(){
          var token=ov.querySelector("#spPaste").value.trim();
          if(!token) return;
          var payload=await verifyToken(token);
          if(payload){ applyRole(payload.role); close(); flashSaved("Activated as "+payload.role); }
          else flashSaved("Invalid or expired token");
        };
      }}
    );
  }
