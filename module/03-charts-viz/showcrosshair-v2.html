function showCrosshair(e){
      var pt=getSVGPoint(e);
      var mx=pt.x, my=pt.y;
      if(mx<padL||mx>W-padR||my<padT||my>padT+plotH){ hideCrosshair(); return; }
      // find nearest candle index
      var idx=Math.round((mx-padL)/step - 0.5);
      idx=Math.max(0,Math.min(n-1,idx));
      var cx=Xc(idx);
      var c=candles[idx];
      var valY=isLine?Yv(c.c):my;
      var priceAtCursor=isLine?c.c:(lo + (1-(my-padT)/plotH)*range);

      // vertical line snapped to candle center
      vLine.setAttribute("x1",cx.toFixed(1)); vLine.setAttribute("x2",cx.toFixed(1));
      vLine.style.display="";
      // horizontal line at cursor y
      hLine.setAttribute("y1",valY.toFixed(1)); hLine.setAttribute("y2",valY.toFixed(1));
      hLine.style.display="";

      // y-axis price label
      var dispPrice=isLine?c.c:priceAtCursor;
      var yLabelY=(valY-6).toFixed(1);
      yBg.setAttribute("y",yLabelY); yBg.setAttribute("width",padL); yBg.style.display="";
      yTxt.setAttribute("y",(parseFloat(yLabelY)+10).toFixed(1));
      yTxt.textContent=sym+(dispPrice>=0?"":"-")+Math.abs(dispPrice).toFixed(2);
      yTxt.style.display="";

      // x-axis month label
      var tipW=56;
      xBg.setAttribute("x",(cx-tipW/2).toFixed(1)); xBg.setAttribute("width",tipW);
      xBg.style.display="";
      xTxt.setAttribute("x",cx.toFixed(1));
      xTxt.textContent=seriesLong(c);
      xTxt.style.display="";

      // highlight dot (line mode)
      if(isLine){
        dotEl.setAttribute("cx",cx.toFixed(1));
        dotEl.setAttribute("cy",Yv(c.c).toFixed(1));
        dotEl.style.display="";
      } else { dotEl.style.display="none"; }

      // tooltip box
      var lines2, maxLen;
      if(isLine){
        lines2=[seriesLong(c)+"  "+sym+c.c.toFixed(2)];
        maxLen=lines2[0].length;
      } else {
        lines2=[
          seriesLong(c),
          "O "+sym+c.o.toFixed(2)+"  C "+sym+c.c.toFixed(2),
          "H "+sym+c.h.toFixed(2)+"  L "+sym+c.l.toFixed(2)
        ];
        maxLen=Math.max(lines2[0].length,lines2[1].length,lines2[2].length);
      }
      var tw=maxLen*5.8+16, th=lines2.length*14+8;
      var tx=cx+10; if(tx+tw>W-padR) tx=cx-tw-10;
      var ty=padT+4; if(isLine) ty=Math.max(padT+4,Yv(c.c)-th-8);
      tipBg.setAttribute("x",tx.toFixed(1)); tipBg.setAttribute("y",ty.toFixed(1));
      tipBg.setAttribute("width",tw.toFixed(1)); tipBg.setAttribute("height",th.toFixed(1));
      tipBg.style.display="";
      // clear old tspans
      while(tipTxt.firstChild) tipTxt.removeChild(tipTxt.firstChild);
      lines2.forEach(function(ln,li){
        var ts=document.createElementNS("http://www.w3.org/2000/svg","tspan");
        ts.setAttribute("x",(tx+8).toFixed(1));
        ts.setAttribute("dy",li===0?((ty+12).toFixed(1)):"13");
        if(li===0){ ts.setAttribute("y",(ty+12).toFixed(1)); ts.removeAttribute("dy"); }
        else ts.setAttribute("dy","13");
        ts.setAttribute("fill",li===0?AMBER:"#e6ebf2");
        ts.textContent=ln;
        tipTxt.appendChild(ts);
      });
      tipTxt.style.display="";
    }
