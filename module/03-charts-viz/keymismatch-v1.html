function keyMismatch(prov, key){
    if(!key) return "";
    var shape=keyLooksLike(key);
    if(!shape) return "That doesn't match the shape of any provider's key. Check it copied whole.";
    if(shape.prov===prov) return "";
    var want=AI_PROVIDERS[prov];
    return "This looks like "+shape.looks+", but the engine is set to "+(want?want.label:prov)
      +". Sending it there will always come back “invalid”. Either switch the engine to "
      +(AI_PROVIDERS[shape.prov]?AI_PROVIDERS[shape.prov].label:shape.prov)+", or paste a "+(want?want.label:prov)+" key.";
  }
