function equityCurveSvg(curve){
    if(curve.length<2) return "";
    var w=680,h=110,pad=6;
    var min=Math.min.apply(null,curve.concat([0])), max=Math.max.apply(null,curve.concat([0]));
    var span=(max-min)||1;
    var pts=curve.map(function(v,i){
      return (pad+(i/(curve.length-1))*(w-pad*2)).toFixed(1)+","+(h-pad-((v-min)/span)*(h-pad*2)).toFixed(1);
    });
    var zeroY=(h-pad-((0-min)/span)*(h-pad*2)).toFixed(1);
    var col=curve[curve.length-1]>=0?"var(--up)":"var(--down)";
    return '<svg viewBox="0 0 '+w+' '+h+'" preserveAspectRatio="none" style="width:100%;height:110px;display:block">'
      +'<line x1="0" y1="'+zeroY+'" x2="'+w+'" y2="'+zeroY+'" stroke="var(--bd2)" stroke-dasharray="3 3"/>'
      +'<polyline points="'+pts.join(" ")+'" fill="none" stroke="'+col+'" stroke-width="2" stroke-linejoin="round"/></svg>';
  }
