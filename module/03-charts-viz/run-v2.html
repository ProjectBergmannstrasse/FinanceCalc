var run=function(){
        try{
          var cv=preprocessReceiptCanvas(img);
          var detector=new window.TextDetector();
          detector.detect(cv).then(function(blocks){
            var lines=blocks.map(function(b){ return b.rawValue; })
              .sort(function(a,b){
                // sort roughly top-to-bottom by bounding box Y
                return (a._y||0)-(b._y||0);
              });
            // preserve vertical order via blocks directly
            var sorted=blocks.slice().sort(function(a,b){
              return (a.boundingBox.top)-(b.boundingBox.top);
            });
            var sortedLines=sorted.map(function(b){ return b.rawValue; });
            var parsed=parseReceiptText(sortedLines.length?sortedLines:lines);
            // fill in whatever we found
            if(parsed.total){
              document.getElementById("scAmt").value=parsed.total.toFixed(2);
            }
            if(parsed.date){
              document.getElementById("scDate").value=parsed.date;
            }
            if(parsed.merchant||parsed.note){
              document.getElementById("scNote").value=parsed.merchant||parsed.note;
            }
            var found=[];
            if(parsed.total) found.push(sym+parsed.total.toFixed(2));
            if(parsed.date)  found.push(parsed.date);
            if(parsed.merchant) found.push('"'+parsed.merchant+'"');
            status.textContent=found.length
              ? "Found: "+found.join(" · ")+" — check and adjust if needed."
              : "Couldn't read any amounts — fill in manually.";
            status.style.color=found.length?"var(--up)":"var(--amber)";
          }).catch(function(err){
            status.textContent="OCR error — fill in manually. ("+String(err).slice(0,60)+")";
            status.style.color="var(--amber)";
          });
        }catch(err){
          status.textContent="Couldn't run OCR — fill in manually.";
          status.style.color="var(--amber)";
        }
      };
