function preprocessReceiptCanvas(src, maxW){
    maxW=maxW||1200;
    var cv=document.createElement("canvas");
    var scale=Math.min(1,maxW/(src.naturalWidth||src.width||maxW));
    cv.width =(src.naturalWidth||src.width)*scale;
    cv.height=(src.naturalHeight||src.height)*scale;
    var ctx=cv.getContext("2d"); ctx.drawImage(src,0,0,cv.width,cv.height);
    // greyscale + contrast boost
    var id=ctx.getImageData(0,0,cv.width,cv.height), d=id.data;
    for(var i=0;i<d.length;i+=4){
      var g=0.299*d[i]+0.587*d[i+1]+0.114*d[i+2];
      // S-curve contrast: push mid-tones apart
      var c=(g/255-0.5)*1.6+0.5;
      var v=Math.min(255,Math.max(0,c*255));
      d[i]=d[i+1]=d[i+2]=v;
    }
    ctx.putImageData(id,0,0);
    return cv;
  }
