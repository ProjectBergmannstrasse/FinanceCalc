function equityCurveSvg(curve, sym){
    if(!curve||curve.length<2) return '<div class="hint" style="margin:0">One trade so far — the curve appears once there are two.</div>';
    var w=680,h=132,padL=6,padR=6,padT=10,padB=10;
    var peaks=[], pk=-Infinity;
    curve.forEach(function(v){ if(v>pk)pk=v; peaks.push(pk); });
    var min=Math.min.apply(null,curve.concat([0]));
    var max=Math.max.apply(null,peaks.concat([0]));
    var span=(max-min)||1;
    var n=curve.length;
    function X(i){ return padL+(i/(n-1))*(w-padL-padR); }
    function Y(v){ return h-padB-((v-min)/span)*(h-padT-padB); }
    var line="", peakLine="", area="";
    for(var i=0;i<n;i++){
      line+=(i?" L ":"M ")+X(i).toFixed(1)+" "+Y(curve[i]).toFixed(1);
      peakLine+=(i?" L ":"M ")+X(i).toFixed(1)+" "+Y(peaks[i]).toFixed(1);
    }
    // drawdown band: down along the peak, back along the equity
    area="M "+X(0).toFixed(1)+" "+Y(peaks[0]).toFixed(1);
    for(i=1;i<n;i++) area+=" L "+X(i).toFixed(1)+" "+Y(peaks[i]).toFixed(1);
    for(i=n-1;i>=0;i--) area+=" L "+X(i).toFixed(1)+" "+Y(curve[i]).toFixed(1);
    area+=" Z";
    var end=curve[n-1];
    var col=end>=0?"var(--up)":"var(--down)";
    var zeroY=Y(0).toFixed(1);
    return '<svg viewBox="0 0 '+w+' '+h+'" preserveAspectRatio="none" id="eqSvg" style="width:100%;height:132px;display:block">'
      +'<line x1="0" y1="'+zeroY+'" x2="'+w+'" y2="'+zeroY+'" stroke="var(--bd2)" stroke-dasharray="3 3"/>'
      +'<path d="'+area+'" fill="var(--down)" opacity="0.13"/>'
      +'<path d="'+peakLine+'" fill="none" stroke="var(--mut)" stroke-width="1" stroke-dasharray="4 4" opacity="0.75"/>'
      +'<path d="'+line+'" fill="none" stroke="'+col+'" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/>'
      +'<circle id="eqDot" r="3.2" fill="'+col+'" opacity="0"/>'
      +'<line id="eqCross" y1="0" y2="'+h+'" stroke="var(--bd2)" stroke-width="1" opacity="0"/>'
      +'</svg>'
      +'<div class="eq-read" id="eqRead"></div>';
  }
