function candleSvg(candles, sym){
    if(!candles.length) return "";
    if(candles.length===1){
      var c0=candles[0];
      return '<div class="hint" style="margin:0">Only one '+(CANDLE_LABELS[CANDLE_TF]||CANDLE_TF+"d")
        +' period so far — '+c0.n+' trade'+(c0.n===1?"":"s")+', closing at '+esc(fmt(c0.close,sym))
        +'. Pick a shorter candle to see more.</div>';
    }
    var w=680,h=150,padT=10,padB=18,padX=8;
    var his=candles.map(function(c){return c.high;}).concat([0]);
    var los=candles.map(function(c){return c.low;}).concat([0]);
    var max=Math.max.apply(null,his), min=Math.min.apply(null,los);
    var span=(max-min)||1;
    var n=candles.length;
    var slot=(w-padX*2)/n;
    var bodyW=Math.max(2,Math.min(18,slot*0.62));
    function Y(v){ return (h-padB-((v-min)/span)*(h-padT-padB)).toFixed(1); }
    var zeroY=Y(0);
    var parts=candles.map(function(c,i){
      var cx=padX+slot*i+slot/2;
      var up=c.close>=c.open;
      var yO=+Y(c.open), yC=+Y(c.close);
      var top=Math.min(yO,yC), bot=Math.max(yO,yC);
      var bh=Math.max(1.5,bot-top);
      var stroke=up?"var(--up)":"var(--down)";
      // hollow body for up, filled for down — direction survives greyscale
      var fill=up?"var(--bg)":"var(--down)";
      return '<g class="cndl" data-i="'+i+'">'
        +'<line x1="'+cx.toFixed(1)+'" y1="'+Y(c.high)+'" x2="'+cx.toFixed(1)+'" y2="'+Y(c.low)+'" stroke="'+stroke+'" stroke-width="1.2"/>'
        +'<rect x="'+(cx-bodyW/2).toFixed(1)+'" y="'+top.toFixed(1)+'" width="'+bodyW.toFixed(1)+'" height="'+bh.toFixed(1)+'" '
          +'fill="'+fill+'" stroke="'+stroke+'" stroke-width="1.4" rx="1"/>'
        +'<rect class="cndl-hit" x="'+(padX+slot*i).toFixed(1)+'" y="0" width="'+slot.toFixed(1)+'" height="'+h+'" fill="transparent"/>'
        +'</g>';
    }).join("");
    return '<svg viewBox="0 0 '+w+' '+h+'" preserveAspectRatio="none" id="cndlSvg" style="width:100%;height:150px;display:block">'
      +'<line x1="0" y1="'+zeroY+'" x2="'+w+'" y2="'+zeroY+'" stroke="var(--bd2)" stroke-dasharray="3 3"/>'
      +parts+'</svg>'
      +'<div class="cndl-read" id="cndlRead"><b>'+n+'</b> '+(CANDLE_LABELS[CANDLE_TF]||CANDLE_TF+"d")
      +' candles · each one is '+(CANDLE_TF===1?"a day":CANDLE_TF===7?"a week":CANDLE_TF+" days")
      +' of trading · hollow = up, filled = down</div>';
  }
