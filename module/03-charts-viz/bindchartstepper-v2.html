function bindChartStepper(){
    var seg=document.getElementById("trPeriod");
    if(!seg||seg.getAttribute("data-stepped")) return;
    seg.setAttribute("data-stepped","1");
    // v2.08: these set the CANDLE PERIOD (what one candlestick covers), not a filter
    ["7","4","3","1"].forEach(function(d){
      var b=document.createElement("button");
      b.type="button";b.setAttribute("data-p",d);b.setAttribute("data-tf","1");
      b.textContent=CANDLE_LABELS[d]||(d+"d");
      b.title="Each candle = "+(d==="1"?"one day":d==="7"?"one week":d+" days");
      seg.insertBefore(b,seg.firstChild);
      b.addEventListener("click",function(){
        CANDLE_TF=parseInt(d,10);
        Array.prototype.forEach.call(seg.querySelectorAll("button"),function(x){x.classList.toggle("on",x===b);});
        renderTradingPage();
      });
    });
    var wrap=document.createElement("div");
    wrap.className="tr-step";
    seg.parentNode.insertBefore(wrap,seg);
    var L=document.createElement("button");L.type="button";L.className="arrow";L.textContent="\u2039";
    var R=document.createElement("button");R.type="button";R.className="arrow";R.textContent="\u203a";
    wrap.appendChild(L);wrap.appendChild(seg);wrap.appendChild(R);
    L.addEventListener("click",function(){trStep(-1);});
    R.addEventListener("click",function(){trStep(1);});
    document.addEventListener("keydown",function(e){
      var pg=document.getElementById("pageTrading");
      if(!pg||pg.style.display==="none") return;
      if(e.target&&/INPUT|SELECT|TEXTAREA/.test(e.target.tagName)) return;
      if(e.key==="ArrowLeft"){trStep(-1);e.preventDefault();}
      if(e.key==="ArrowRight"){trStep(1);e.preventDefault();}
    });
    applyDeviceType();
  }
