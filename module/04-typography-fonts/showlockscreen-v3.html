function showLockScreen(onUnlock){
    var ls=lockState();
    var hasBio=ls.credId&&webAuthnSupported();
    var el=document.getElementById("lockScreen");
    if(el) el.remove();
    var div=document.createElement("div");
    div.id="lockScreen";
    div.innerHTML='<div class="lock-inner">'
      +'<div class="lock-icon">🔐</div>'
      +'<div class="lock-title">My Money</div>'
      +'<div class="lock-sub">Enter your PIN to continue</div>'
      +'<div class="lock-dots" id="lockDots">○ ○ ○ ○</div>'
      +'<input type="password" id="lockPinInput" inputmode="numeric" maxlength="6" autocomplete="off" '
        +'style="position:absolute;opacity:0;pointer-events:none" aria-label="PIN">'
      +'<div class="lock-pad" id="lockPad"></div>'
      +(hasBio?'<button class="btn ghost sm" id="lockBioBtn" style="margin-top:14px;width:200px">Use Face ID / Fingerprint</button>':'')
      +'<div class="lock-err" id="lockErr"></div></div>';
    document.body.appendChild(div);
    requestAnimationFrame(function(){ div.classList.add("in"); });

    var entered="";
    function updateDots(){
      var d=document.getElementById("lockDots"); if(!d) return;
      d.textContent=Array(4).fill(0).map((_,i)=>entered.length>i?"●":"○").join(" ");
    }
    function showErr(msg){ var e=document.getElementById("lockErr"); if(e){e.textContent=msg;e.style.color="var(--down)";} }
    async function tryPin(){
      var stored=ls.pin, salt=ls.salt||"mymoney";
      var h=await hashPin(entered, salt);
      if(h===stored){ unlock(); }
      else{ showErr("Wrong PIN — try again"); entered=""; updateDots(); }
    }
    function unlock(){
      _lockSession=true;
      div.classList.remove("in");
      setTimeout(function(){ if(div.parentNode) div.parentNode.removeChild(div); },250);
      resetLockTimer();
      if(onUnlock) onUnlock();
    }
    // number pad
    var pad=document.getElementById("lockPad");
    [1,2,3,4,5,6,7,8,9,"",0,"⌫"].forEach(function(k){
      var b=document.createElement("button");
      b.className="lock-key"+(k===""?" lock-key-blank":"");
      b.textContent=k===""?"":String(k);
      if(k!=="") b.addEventListener("click",function(){
        if(k==="⌫"){ entered=entered.slice(0,-1); updateDots(); return; }
        if(entered.length>=6) return;
        entered+=String(k); updateDots();
        if(entered.length===4||entered.length===6) tryPin();
      });
      pad.appendChild(b);
    });
    // biometric
    var bioBtn=document.getElementById("lockBioBtn");
    if(bioBtn) bioBtn.addEventListener("click",async function(){
      bioBtn.disabled=true; bioBtn.textContent="Waiting…";
      var ok=await verifyBiometric(ls.credId);
      if(ok) unlock(); else{ showErr("Biometric failed — use PIN"); bioBtn.disabled=false; bioBtn.textContent="Use Face ID / Fingerprint"; }
    });
    // keyboard
    document.getElementById("lockPinInput").focus();
    div.addEventListener("keydown",function(e){
      if(e.key>="0"&&e.key<="9"){ if(entered.length<6){entered+=e.key;updateDots();} if(entered.length===4||entered.length===6) tryPin(); }
      else if(e.key==="Backspace"){ entered=entered.slice(0,-1); updateDots(); }
    });
    // auto-attempt biometric on first show
    if(hasBio) setTimeout(function(){ if(bioBtn&&!bioBtn.disabled) bioBtn.click(); },400);
  }
