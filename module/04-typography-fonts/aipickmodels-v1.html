function aiPickModels(slot){
    var isB=(slot==="B");
    var prov=AI_PROVIDERS[isB?aiCfg.provB:aiCfg.provA];
    var key=isB?aiCfg.keyB:aiCfg.keyA;
    if(!prov || !key){
      appModal({title:"Add a key first", message:"Paste the API key for engine "+slot+", then tap Find models.", confirmText:"OK", cancelText:"Close"});
      return;
    }
    var out=document.getElementById("aiTestOut");
    out.style.color="var(--mut)"; out.textContent="Asking "+prov.label+" which models this key can use…";
    prov.listModels(key).then(function(models){
      if(!models.length){ out.style.color="var(--down)"; out.textContent="No usable models came back for this key."; return; }
      models.sort(function(a,b){ return aiRankModel(b.id)-aiRankModel(a.id); });
      out.textContent="";
      // Styled with its own classes rather than inheriting .mdl-bd, which is muted
      // prose text — on the Mono theme that rendered the model names near-invisible.
      var body=models.slice(0,40).map(function(m,i){
        var nice=(m.display && m.display!==m.id) ? m.display : "";
        return '<label class="mdl-pick'+(i===0?" mdl-pick-rec":"")+'">'
          +'<input type="radio" name="mdlPick" value="'+esc(m.id)+'" data-ver="'+esc(m.ver||"")+'"'+(i===0?" checked":"")+'>'
          +'<span class="mdl-pick-main">'
            +'<span class="mdl-pick-id">'+esc(m.id)+'</span>'
            +(nice?'<span class="mdl-pick-name">'+esc(nice)+'</span>':'')
          +'</span>'
          +(i===0?'<span class="mdl-pick-tag">recommended</span>':'')
          +'</label>';
      }).join("");
      // Track the choice as it's made, rather than reading the DOM at confirm time:
      // appModal's own handler closes (and removes) the modal before ours would run,
      // so any read-on-confirm returns null and silently does nothing.
      var chosen={id:models[0].id, ver:models[0].ver||""};
      appModal({
        title:prov.label+" models",
        message:'<div class="mdl-picklist">'+body+'</div>',
        confirmText:"Use this one",
        onConfirm:function(){
          if(isB){ aiCfg.modelB=chosen.id; aiCfg.verB=chosen.ver; }
          else   { aiCfg.modelA=chosen.id; aiCfg.verA=chosen.ver; }
          saveAiCfg();
          document.getElementById(isB?"aiModelB":"aiModelA").value=chosen.id;
          out.style.color="var(--up)";
          out.textContent="Engine "+slot+" set to "+chosen.id;
          refreshAiVisibility();
        }
      });
      setTimeout(function(){
        Array.prototype.forEach.call(document.querySelectorAll('input[name="mdlPick"]'), function(r){
          r.addEventListener("change", function(){
            chosen={id:r.value, ver:r.getAttribute("data-ver")||""};
          });
        });
      }, 30);
    }).catch(function(e){
      out.style.color="var(--down)";
      out.innerHTML="<b>Couldn't list models.</b><br>"+esc(e.message)
        +"<br><br>If this says the key was rejected, the key itself is the problem. If it lists fine but reading images fails, it's the model name.";
    });
  }
