function pwaIcon(){
    // a plain generated icon so we don't ship a binary asset
    var c=document.createElement("canvas"); c.width=c.height=512;
    var x=c.getContext("2d");
    x.fillStyle="#0b0e14"; x.fillRect(0,0,512,512);
    x.fillStyle="#3ddc84"; x.font="bold 300px ui-monospace,monospace";
    x.textAlign="center"; x.textBaseline="middle"; x.fillText("$",256,270);
    return c.toDataURL("image/png");
  }
  function triggerInstall(){
    if(_deferredPrompt){
      _deferredPrompt.prompt();
      _deferredPrompt.userChoice.then(function(){ _deferredPrompt=null; });
      return;
    }
    /* Chrome only fires beforeinstallprompt over https with a service worker,
       and Safari never fires it. Rather than a toast that vanishes, say exactly
       what to tap on the browser they're actually using. */
    var ua=navigator.userAgent||"";
    var isIOS=/iPad|iPhone|iPod/.test(ua);
    var isSafari=/^((?!chrome|android).)*safari/i.test(ua);
    var installed=window.matchMedia&&window.matchMedia("(display-mode: standalone)").matches;
    var framed=false; try{ framed=(window.self!==window.top); }catch(e){ framed=true; }
    var msg;
    if(installed) msg="It's already installed \u2014 you're running the installed copy now.";
    else if(isIOS) msg="On iPhone or iPad: tap Share, then \u201cAdd to Home Screen\u201d.";
    else if(isSafari) msg="In Safari: File \u2192 Add to Dock, or Share \u2192 Add to Home Screen.";
    else if(location.protocol==="file:")
      msg="Installing needs the app served over https \u2014 a file opened straight from disk can't be installed. Put it on GitHub Pages or any host and this will work.";
    else msg="Open your browser menu and choose \u201cInstall app\u201d. If it's missing, the page needs https.";
    appModal({title:"Install as an app", message:msg, confirmText:"OK"});
  }
