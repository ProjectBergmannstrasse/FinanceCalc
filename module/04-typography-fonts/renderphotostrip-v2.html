function renderPhotoStrip(){
    var strip=document.getElementById("fPhotoStrip");
    if(!strip) return;
    var total=keepPhotos.length+pendingPhotos.length;
    var nm=document.getElementById("fPhotoName");
    if(nm) nm.textContent = total ? total+" file"+(total===1?"":"s")
      +(total>=PHOTO_MAX?" (max)":"") : "";
    var clr=document.getElementById("fPhotoClear");
    if(clr) clr.style.display = total ? "inline-block" : "none";
    var btn=document.getElementById("fPhotoBtn");
    if(btn) btn.disabled = total>=PHOTO_MAX;
    strip.innerHTML="";
    keepPhotos.forEach(function(id,i){
      var c=document.createElement("div"); c.className="ph-thumb"; c.setAttribute("data-keep",id);
      var nm=(attNames[id]||"");
      c.innerHTML='<button type="button" class="ph-x" title="Remove">×</button>';
      strip.appendChild(c);
      loadPhoto(id, function(url){
        if(!url) return;
        if(attIsImage(nm,url)){ var im=new Image(); im.alt="Attachment "+(i+1); im.src=url; c.insertBefore(im,c.firstChild); }
        else { var d=document.createElement("div"); d.className="ph-doc";
          d.innerHTML='<span class="ph-ic">'+attIcon(nm,url)+'</span>'; c.insertBefore(d,c.firstChild); }
      });
      c.querySelector(".ph-x").addEventListener("click", function(){
        keepPhotos=keepPhotos.filter(function(x){ return x!==id; });
        photoCleared=true; renderPhotoStrip();
      });
    });
    pendingPhotos.forEach(function(pp,i){
      var c=document.createElement("div"); c.className="ph-thumb ph-new";
      var note="";
      if(pp.wasBytes && pp.nowBytes && pp.nowBytes < pp.wasBytes*0.9)
        note=kbTxt(pp.wasBytes)+" → "+kbTxt(pp.nowBytes);
      else if(pp.nowBytes||pp.wasBytes) note=kbTxt(pp.nowBytes||pp.wasBytes);
      c.title=(pp.name||"photo")+(note?" · "+note:"");
      var inner = attIsImage(pp.name,pp.dataUrl)
        ? '<img src="'+pp.dataUrl+'" alt="New attachment '+(i+1)+'">'
        : '<div class="ph-doc"><span class="ph-ic">'+attIcon(pp.name,pp.dataUrl)+'</span><span class="ph-nm">'+esc((pp.name||"file").slice(0,14))+'</span></div>';
      c.innerHTML=inner+'<button type="button" class="ph-x" title="Remove">×</button>'
        +(note?'<span class="ph-kb">'+esc(note)+'</span>':'');
      strip.appendChild(c);
      c.querySelector(".ph-x").addEventListener("click", function(){
        pendingPhotos.splice(pendingPhotos.indexOf(pp),1); renderPhotoStrip();
        mtOcr.token++; mtOcr.state="idle"; mtOcr.result=null; mtOcr.wantFill=false;  // v1.96
        if(mtOcrImages().length) mtOcrStart(); else mtOcrUI();                       // v1.96
      });
    });
  }
