function duplicatePortfolio(pid){
    var src=findPortfolio(pid); if(!src) return;
    if(pid===ACTIVE_PID) flushActiveIntoRegistry();
    appModal({
      title:"Duplicate portfolio",
      message:"Creates an independent copy of <b>"+esc(src.name)+"</b>, including its own copies of any photos.",
      input:{value:src.name+" copy"},
      confirmText:"Duplicate",
      onConfirm:function(name){
        var dataCopy=clone(src.data||{});
        var txs=Array.isArray(dataCopy.transactions)?dataCopy.transactions:[];
        var newPid=genPid();
        var photoJobs=[];
        txs.forEach(function(t){
          var old=photosOf(t);
          t.id=freshIdBase();
          if(old.length){
            var fresh=[];
            old.forEach(function(oldId){
              var nid=newPhotoId(t.id, fresh);
              fresh.push(nid);
              photoJobs.push({oldId:oldId, newId:nid});
            });
            setPhotos(t, fresh);   // every photo re-keyed, not just the first
          }
        });
        delete dataCopy.mirrorFrom;
        PORTFOLIOS.push({id:newPid, name:name, createdAt:Date.now(), data:dataCopy});
        var remaining=photoJobs.length;
        function finish(){ switchPortfolio(newPid); }
        if(remaining===0){ finish(); return; }
        photoJobs.forEach(function(job){
          loadPhoto(job.oldId, function(dataUrl){
            if(dataUrl){ savePhoto(job.newId, dataUrl, function(){ remaining--; if(remaining<=0) finish(); }); }
            else { remaining--; if(remaining<=0) finish(); }
          });
        });
      }
    });
  }
