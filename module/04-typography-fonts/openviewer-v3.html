function openViewer(txId){
    var tx=state.transactions.filter(function(x){ return x.id===txId; })[0];
    if(!tx) return;
    viewingId=txId;
    var body=document.getElementById("viewerBody");
    document.getElementById("viewerTitle").textContent=(tx.note||tx.cat||"Photo").slice(0,55);
    body.innerHTML='<div class="viewer-note">Loading\u2026</div>';
    document.getElementById("viewer").style.display="flex";
    if(tx.photo){
      loadPhoto(tx.photo, function(data){
        if(!data){ body.innerHTML='<div class="viewer-note">Photo not found. Re-attach it with the \ud83d\udcce button.</div>'; return; }
        var img=new Image();
        img.style.cssText="max-width:100%;border-radius:8px";
        img.onload=function(){ body.innerHTML=""; body.appendChild(img); };
        img.onerror=function(){ body.innerHTML='<div class="viewer-note">Can\'t preview this format in browser.<br><a href="'+data+'" download="receipt">Download to view</a></div>'; };
        img.src=data; img.alt="receipt";
      });
    } else {
      body.innerHTML='<div class="viewer-note">No photo attached.</div>';
    }
  }
