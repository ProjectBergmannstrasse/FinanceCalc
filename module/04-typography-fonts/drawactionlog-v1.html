function drawActionLog(){
    if(!actionLog.length)
      return '<div class="hs-empty"><div class="he-ic">\u21BA</div>'
        +'<div class="he-t">Nothing logged yet</div>'
        +'<div class="he-d">Adds, edits and deletes appear here<br>with a one-tap undo.</div></div>';
    var html="", lastDay="";
    actionLog.slice(0,80).forEach(function(a,i){
      var day=dayHeading(a.at);
      if(day!==lastDay){ html+='<div class="hs-day">'+esc(day)+'</div>'; lastDay=day; }
      var cls = a.kind==="delete"||a.kind==="mass" ? "del"
              : a.kind==="add" ? "add"
              : a.kind==="edit" ? "edit" : "";
      html+='<div class="hs-card'+(a.reverted?" done":"")+'" style="animation-delay:'+Math.min(i*22,220)+'ms">'
        +'<div class="hs-ic '+cls+'">'+(ACTION_ICON[a.kind]||"\u2022")+'</div>'
        +'<div class="hs-main"><div class="hs-t">'+esc(a.label)+'</div>'
        +'<div class="hs-sub"><span>'+esc(agoText(a.at))+'</span>'
        +'<span class="hs-dot"></span><span>'+esc(fmtLogTime(a.at))+'</span></div></div>'
        +(a.reverted
          ? '<span class="hs-pill">reverted</span>'
          : (a.undo?'<button class="hs-btn" data-revert="'+a.id+'">Undo</button>':''))
        +'</div>';
    });
    return html;
  }
