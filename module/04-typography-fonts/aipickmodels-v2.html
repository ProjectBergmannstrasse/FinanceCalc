function aiPickModels(slot){
    var isB=(slot==="B");
    var prov=AI_PROVIDERS[isB?aiCfg.provB:aiCfg.provA];
    var key=isB?aiCfg.keyB:aiCfg.keyA;
    if(!prov || !key){
      appModal({title:"Add a key first", message:"Paste the API key for engine "+slot+", then tap Find models.", confirmText:"OK", cancelText:"Close"});
      return;
    }
    var out=document.getElementById("aiTestOut");
    out.style.color="var(--mut)"; out.textContent="Asking "+prov.label+" which models this key can use…";
    prov.listModels(key).then(function(models){
      if(!models.length){ out.style.color="var(--down)"; out.textContent="No usable models came back for this key."; return; }
      models.sort(function(a,b){ return aiRankModel(b.id)-aiRankModel(a.id); });
      out.textContent="";
      var body=models.slice(0,40).map(function(m,i){
        return '<label style="display:flex;gap:8px;align-items:center;padding:7px 2px;border-bottom:1px solid var(--bd);cursor:pointer">'
          +'<input type="radio" name="mdlPick" value="'+esc(m.id)+'" data-ver="'+esc(m.ver||"")+'"'+(i===0?" checked":"")+'>'
          +'<span style="font-family:var(--mono);font-size:11.5px">'+esc(m.id)+(i===0?' <span style="color:var(--up)">· recommended</span>':'')+'</span></label>';
      }).join("");
      appModal({
        title:prov.label+" models",
        message:'<div style="max-height:280px;overflow-y:auto;margin-top:6px">'+body+'</div>',
        confirmText:"Use this one",
        onConfirm:function(){}
      });
      // wire after the modal exists
      setTimeout(function(){
        var ok=document.getElementById("mdlOk");
        if(!ok) return;
        ok.onclick=function(){
          var sel=document.querySelector('input[name="mdlPick"]:checked');
          if(!sel) return;
          if(isB){ aiCfg.modelB=sel.value; aiCfg.verB=sel.getAttribute("data-ver")||""; }
          else   { aiCfg.modelA=sel.value; aiCfg.verA=sel.getAttribute("data-ver")||""; }
          saveAiCfg();
          document.getElementById(isB?"aiModelB":"aiModelA").value=sel.value;
          closeModal();
          out.style.color="var(--up)";
          out.textContent="Engine "+slot+" set to "+sel.value;
          refreshAiVisibility();
        };
      }, 40);
    }).catch(function(e){
      out.style.color="var(--down)";
      out.innerHTML="<b>Couldn't list models.</b><br>"+esc(e.message)
        +"<br><br>If this says the key was rejected, the key itself is the problem. If it lists fine but reading images fails, it's the model name.";
    });
  }
