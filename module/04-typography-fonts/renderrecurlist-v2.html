function renderRecurList(){
    var box=document.getElementById("recList");
    if(!box) return;
    var list=recurList(), sym=state.currency;
    if(!list.length){ box.innerHTML='<div class="rec-empty">Nothing repeats yet.</div>'; return; }
    box.innerHTML=list.map(function(r){
      var meta=TYPE_META[r.type]||{label:r.type};
      var title=r.note ? esc(r.note) : esc(meta.label)+(r.cat?" · "+esc(r.cat):"");
      return '<div class="rec-item'+(r.paused?" paused":"")+'" data-id="'+esc(r.id)+'">'
        +'<div class="rec-main"><div class="rec-t">'+title+'</div><div class="rec-sub">'+esc(recurDescribe(r))+'</div></div>'
        +'<div class="rec-amt mono">'+fmt(Math.abs(num(r.amount)),sym)+'</div>'
        +'<button type="button" class="rec-btn" data-act="pause">'+(r.paused?"Resume":"Pause")+'</button>'
        +'<button type="button" class="rec-btn" data-act="del">Delete</button></div>';
    }).join("");
    Array.prototype.forEach.call(box.querySelectorAll(".rec-btn"), function(b){
      b.addEventListener("click", function(){
        var id=b.parentNode.getAttribute("data-id");
        var r=recurList().filter(function(x){ return String(x.id)===String(id); })[0];
        if(!r) return;
        if(b.getAttribute("data-act")==="pause"){
          r.paused=!r.paused; persistNow(); renderRecurList(); checkRecurPrompt();
        } else {
          appModal({ title:"Delete this repeat?",
            message:"It stops repeating. Entries it already created stay in your ledger — deleting the rule doesn't erase history.",
            confirmText:"Delete", danger:true,
            onConfirm:function(){
              state.recurring=recurList().filter(function(x){ return String(x.id)!==String(id); });
              persistNow(); renderRecurList(); checkRecurPrompt();
            }});
        }
      });
    });
  }
