function persEnsure(){
    if(document.getElementById("persOv")) return;
    var ov=document.createElement("div");
    ov.className="pers-ov"; ov.id="persOv";
    ov.innerHTML=
      '<div class="pers-hd"><button type="button" class="pers-back" id="persClose" aria-label="Close">←</button><h2>Personalize</h2></div>'
      +'<div class="pers-pages" id="persPages">'

      +'<div class="pers-page" data-pg="app">'
        +'<div class="pers-card"><div class="pers-themes" id="persThemes">'
          +PERS_THEME_KEYS.map(function(t){return '<button type="button" class="pers-th" data-th="'+t[0]+'"><div class="sw2" style="background:'+t[1]+'"></div><div class="nm">'+t[0]+'</div></button>';}).join('')
        +'</div></div>'
      +'</div>'

      +'<div class="pers-page" data-pg="trade">'
        +'<div class="pers-card">'
          +persSw('persTrader','Trader mode','Unlocks the Trading page, instruments and trade entry types.')
          +persSeg('persMtOcr','Statement reading','Photos of MT4/MT5 statements read on this device — nothing uploaded.',[["off","Off"],["auto","Auto"],["silent","Silent"]])
          +persSw('persNoSplits','No splits for trades','Hides the split field on trade entries. Existing entries never change.')
        +'</div>'
      +'</div>'

      +'<div class="pers-page" data-pg="behav">'
        +'<div class="pers-card">'
          +persSw('persDraft','Draft safety','If the app closes mid-entry, what you typed is offered back next time.')
        +'</div>'
      +'</div>'

      +'<div class="pers-page" data-pg="device">'
        +'<div class="pers-card">'
          +persSeg('persDevType','Device type','Touch shows tap-arrows on the chart stepper.',[["auto","Auto"],["touch","Touch"],["desktop","Desktop"]])
        +'</div>'
      +'</div>'

      +'<div class="pers-page" data-pg="priv">'
        +'<div class="pers-card">'
          +persSeg('persConsent','Share flagged images','Helps improve statement reading. Items only ever wait in a local outbox on this device — this version never transmits.',[["never","Never"],["wifi","Wi-Fi"],["any","Wi-Fi+data"]])
          +'<div class="pers-row"><div><div class="lab">Terms &amp; privacy</div><div class="sub">Plain-language, on-device-only policy.</div></div>'
          +'<button type="button" class="btn ghost sm" id="persTos">Read</button></div>'
        +'</div>'
      +'</div>'

      +'</div>'
      +'<div class="pers-bar2" id="persBar" role="tablist"><div class="pill" id="persPill"></div>'
        +PERS_TABS.map(function(t,i){return '<button type="button" class="pers-tab'+(i===0?' on':'')+'" role="tab" aria-selected="'+(i===0)+'" data-pg="'+t[0]+'"><span class="ic">'+t[1]+'</span><span class="lb">'+t[2]+'</span></button>';}).join('')
      +'</div>';
    document.body.appendChild(ov);

    // tab bar: sliding pill + page switch with retriggered fade/rise
    function persAlignPill(){
      var on=ov.querySelector('.pers-tab.on'), pill=document.getElementById('persPill');
      if(!on||!pill) return;
      pill.style.left=on.offsetLeft+'px'; pill.style.width=on.offsetWidth+'px';
    }
    function persGo(pg){
      Array.prototype.forEach.call(ov.querySelectorAll('.pers-tab'),function(t){
        var on=t.getAttribute('data-pg')===pg;
        t.classList.toggle('on',on); t.setAttribute('aria-selected',on?'true':'false');
      });
      Array.prototype.forEach.call(ov.querySelectorAll('.pers-page'),function(p){
        var on=p.getAttribute('data-pg')===pg;
        if(on){p.classList.remove('on');void p.offsetWidth;p.classList.add('on');}
        else p.classList.remove('on');
      });
      persAlignPill();
    }
    Array.prototype.forEach.call(ov.querySelectorAll('.pers-tab'),function(t){
      t.addEventListener('click',function(){persGo(t.getAttribute('data-pg'));});
    });
    window.addEventListener('resize',persAlignPill);
    ov.__go=persGo; ov.__align=persAlignPill;

    // theme grid
    Array.prototype.forEach.call(ov.querySelectorAll('.pers-th'),function(b){
      b.addEventListener('click',function(){
        var k=b.getAttribute('data-th');
        Array.prototype.forEach.call(ov.querySelectorAll('.pers-th'),function(x){x.classList.toggle('on',x===b);});
        var real=document.querySelector('.theme-opt[data-theme-key="'+k+'"]');
        if(real) real.click();
        else{document.documentElement.setAttribute('data-theme',k==='dark'?'':k);try{localStorage.setItem('mymoney_theme',k);}catch(e){}}
        flashSaved('theme applied');
      });
    });

    // live-apply wiring
    persSwBind('persTrader',function(on){
      var c=document.getElementById('sIsTrader');
      if(c){c.checked=on;c.dispatchEvent(new Event('change'));}
    });
    persSegBind('persMtOcr',function(v){
      if(!state.settings)state.settings={};
      state.settings.mtOcr=v;persistNow();mtOcrUI();flashSaved('saved');
    });
    persSwBind('persNoSplits',function(on){
      if(!state.settings)state.settings={};
      state.settings.noSplits=on;persistNow();applyNoSplits();flashSaved('saved');
    });
    persSwBind('persDraft',function(on){
      if(!state.settings)state.settings={};
      state.settings.draftSafety=on;persistNow();flashSaved('saved');
    });
    persSegBind('persDevType',function(v){
      if(!state.settings)state.settings={};
      state.settings.deviceType=v;persistNow();applyDeviceType();flashSaved('saved');
    });
    persSegBind('persConsent',function(v){
      if(!state.settings)state.settings={};
      state.settings.shareConsent=v;persistNow();flashSaved('saved');
    });
    document.getElementById('persTos').addEventListener('click',showTosModal);
    document.getElementById('persClose').addEventListener('click',function(){
      ov.classList.remove('vis');
      setTimeout(function(){ov.classList.remove('on');},220);
    });
  }
