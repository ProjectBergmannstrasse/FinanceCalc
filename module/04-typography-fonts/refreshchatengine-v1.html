function refreshChatEngine(){
    var el=document.getElementById("chatEngine");
    if(!el) return;
    var set=function(txt,cls){ el.textContent=txt; el.className="chat-engine"+(cls?" "+cls:""); };
    if(window._wllmEngine){ set("WebLLM \u00b7 on device","ready"); return; }
    if(window._wllmLoading){ set("loading model\u2026","busy"); return; }
    set("checking\u2026","busy");
    initBuiltinAi().then(function(ready){
      if(ready){ set("Chrome AI \u00b7 on device","ready"); return; }
      if(wllmRemembered()){
        set("reconnecting\u2026","busy");
        loadWebllm({silent:true});
        return;
      }
      set("no model \u2014 tap to set up","");
      el.style.cursor="pointer";
      el.onclick=function(){ openAiSettings(); };
    });
  }
