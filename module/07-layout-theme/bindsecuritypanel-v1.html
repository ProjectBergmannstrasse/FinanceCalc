function bindSecurityPanel(){
    loadLock();
    loadFace();
    // The panel now lives inside a settings <details> box, so it has no collapse
    // header of its own — a second layer of clicking looked broken when the
    // outer box was shut. Refresh when the box it sits in is opened instead.
    var body=document.getElementById("secBody");
    if(!body) return;
    var outer=body.closest("details.sect");
    if(outer) outer.addEventListener("toggle", function(){
      if(outer.open){ refreshSecUI(); }
    });

    // biometric
    /* App lock: PIN first (works anywhere), Face ID as an optional extra
       where the page is on https and a sensor exists. */
    var bioBtn=document.getElementById("bioBtn");
    if(bioBtn) bioBtn.addEventListener("click", function(){
      if(lockIsOn()){
        appModal({title:"Turn off the app lock?",
          message:"The app will open without asking for your PIN"+(faceCfg.on?" or your face":"")+".",
          confirmText:"Turn off", cancelText:"Keep it on",
          onConfirm:function(){ clearLock(); faceDisable(); refreshSecUI(); flashSaved("app lock off"); }});
        return;
      }
      openPinSetup();      // proper keypad, and it always says what went wrong
    });

    // separate control for face unlock, so it can be added or removed on its own
    var afChk=document.getElementById("autoFaceChk");
    if(afChk && !afChk._bound){
      afChk._bound=true;
      afChk.checked=autoFaceOn();
      afChk.addEventListener("change", function(){
        setAutoFace(afChk.checked);
        var n=document.getElementById("autoFaceNote");
        if(n) n.textContent = afChk.checked
          ? "Look for your face as soon as the lock screen opens. The camera runs only on that screen."
          : "You'll tap “Use face unlock” yourself. The camera stays off until you do.";
        flashSaved(afChk.checked?"auto unlock on":"auto unlock off");
      });
    }
    var faNote=document.getElementById("faLogNote");
    if(faNote){
      var n=loadShots().length;
      faNote.textContent = n
        ? n+" attempt"+(n===1?"":"s")+" recorded with images. Review or download them any time."
        : "Unrecognised attempts, with images. Nothing recorded yet.";
    }
    var capChk=document.getElementById("capChk");
    if(capChk && !capChk._bound){
      capChk._bound=true;
      capChk.checked=captureOn();
      capChk.addEventListener("change", function(){
        setCaptureOn(capChk.checked);
        var n=document.getElementById("capNote");
        if(n) n.textContent = capChk.checked
          ? "Takes a few photos when someone isn't recognised, and shows them a notice while it does. Never silent."
          : "Off. Attempts are still logged with a timestamp, just without images.";
        flashSaved(capChk.checked?"attempt photos on":"attempt photos off");
      });
    }
    var faBtn=document.getElementById("faLogBtn");
    if(faBtn && !faBtn._bound){
      faBtn._bound=true;
      faBtn.addEventListener("click", function(){ showIntrusionReview(true); });
    }
    var recBtn=document.getElementById("recBtn");
    if(recBtn) recBtn.addEventListener("click", function(){
      if(recoveryOn()){
        appModal({title:"Remove the recovery question?",
          message:"A forgotten PIN would then mean clearing the app's data and restoring a backup.",
          confirmText:"Remove", cancelText:"Keep",
          onConfirm:function(){ clearRecovery(); refreshSecUI(); flashSaved("recovery removed"); }});
        return;
      }
      openRecoverySetup();
    });

    var faceBtn=document.getElementById("faceBtn");
    if(faceBtn) faceBtn.addEventListener("click", function(){
      if(!lockIsOn()){
        appModal({title:"Set a PIN first",
          message:"Face unlock sits on top of the PIN — the PIN is the fallback whenever your face can't be read.",
          confirmText:"OK"});
        return;
      }
      if(faceCfg.on){
        appModal({title:"Remove face unlock?", message:"You'll unlock with your PIN only.",
          confirmText:"Remove", cancelText:"Keep",
          onConfirm:function(){ faceDisable(); refreshSecUI(); flashSaved("face unlock removed"); }});
        return;
      }
      enrolFace(function(){ refreshSecUI(); });
    });

    // PWA install
    var pwaBtn=document.getElementById("pwaInstallBtn");
    if(pwaBtn) pwaBtn.addEventListener("click", triggerInstall);

    // role
    var adChk=document.getElementById("autoDetectChk");
    if(adChk){
      adChk.checked=!(state.settings&&state.settings.autoDetect===false);
      adChk.addEventListener("change",function(){
        if(!state.settings) state.settings={};
        state.settings.autoDetect=adChk.checked; persistNow();
        flashSaved(adChk.checked?"auto-detect on":"auto-detect off");
      });
    }
    refreshSecUI();
  }
