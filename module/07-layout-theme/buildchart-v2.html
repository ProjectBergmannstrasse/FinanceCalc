function buildChart(){
    var CC=chartColors();   // resolved per build, so a theme switch repaints correctly
    var UP=CC.up, DOWN=CC.down, GRID=CC.grid, ZERO=CC.zero, AXTX=CC.ax, DOJI=CC.doji, AMBER=CC.amber, COOL=CC.cool;
    var wrap=document.getElementById("chartWrap");
    if(state.chartCollapsed){ wrap.style.display="none"; document.getElementById("chartToggle").textContent="▸"; return; }
    wrap.style.display="block"; document.getElementById("chartToggle").textContent="▾";

    var candles=computeMonthly();
    var sym=state.currency;
    var W=720, H=210, padL=52, padR=14, padT=16, padB=28;
    var n=candles.length;
    var plotW=W-padL-padR, plotH=H-padT-padB;

    var lo=Infinity, hi=-Infinity, i;
    for(i=0;i<n;i++){
      if(candles[i].l<lo) lo=candles[i].l;
      if(candles[i].h>hi) hi=candles[i].h;
    }
    if(!isFinite(lo)||!isFinite(hi)||hi===lo){ lo=lo-1; hi=hi+1; }
    var range=hi-lo, pad=range*0.12; hi+=pad; lo-=pad; range=hi-lo;
    function Yv(v){ return padT + plotH*(1-(v-lo)/range); }
    var step=plotW/n;
    function Xc(idx){ return padL + step*idx + step/2; }

    // Build static SVG parts (background, grid, candles/line, x-labels)
    var svg='';

    // gridlines
    var gridVals=[hi, lo, (hi+lo)/2];
    if(lo<0 && hi>0) gridVals.push(0);
    gridVals.forEach(function(gv){
      var y=Yv(gv), isZero=(Math.abs(gv)<0.001);
      svg+='<line x1="'+padL+'" y1="'+y.toFixed(1)+'" x2="'+(W-padR)+'" y2="'+y.toFixed(1)+
           '" stroke="'+(isZero?ZERO:GRID)+'" stroke-width="'+(isZero?1.5:1)+'"'+
           (isZero?'':' stroke-dasharray="3 5"')+' opacity="0.7"/>';
      svg+='<text x="'+(padL-5)+'" y="'+(y+3.5).toFixed(1)+'" text-anchor="end" '+
           'font-size="9" font-family="'+MFONT+'" fill="'+AXTX+'">'+fmtShort(gv,sym)+'</text>';
    });

    var labelEvery = n>18?3:(n>10?2:1);
    var isLine = state.chartType==="line";

    if(isLine){
      // gradient area
      var apts="M "+Xc(0).toFixed(1)+" "+Yv(candles[0].c).toFixed(1)+" ";
      for(i=1;i<n;i++) apts+="L "+Xc(i).toFixed(1)+" "+Yv(candles[i].c).toFixed(1)+" ";
      apts+="L "+Xc(n-1).toFixed(1)+" "+(padT+plotH)+" L "+Xc(0).toFixed(1)+" "+(padT+plotH)+" Z";
      svg+='<defs><linearGradient id="lgrad" x1="0" y1="0" x2="0" y2="1">'+
           '<stop offset="0%" stop-color="'+AMBER+'" stop-opacity="0.18"/>'+
           '<stop offset="100%" stop-color="'+AMBER+'" stop-opacity="0.01"/></linearGradient></defs>';
      svg+='<path d="'+apts+'" fill="url(#lgrad)"/>';
      // line
      var lpts="";
      for(i=0;i<n;i++) lpts+=Xc(i).toFixed(1)+","+Yv(candles[i].c).toFixed(1)+" ";
      svg+='<polyline points="'+lpts.trim()+'" fill="none" stroke="'+AMBER+'" stroke-width="2.2" '+
           'stroke-linejoin="round" stroke-linecap="round"/>';
      // dots (invisible large hit areas + visible small dots)
      for(i=0;i<n;i++){
        var cx2=Xc(i).toFixed(1), cy2=Yv(candles[i].c).toFixed(1);
        svg+='<circle class="cv-dot" cx="'+cx2+'" cy="'+cy2+'" r="3" fill="'+AMBER+'" data-i="'+i+'"/>';
      }
    } else {
      var bw=Math.max(6, Math.min(step*0.6, 26));
      for(i=0;i<n;i++){
        var c=candles[i], cx=Xc(i);
        var isUp=c.c>=c.o;
        var col=(Math.abs(c.c-c.o)<0.001)?DOJI:(isUp?UP:DOWN);
        var yH=Yv(c.h), yL=Yv(c.l), yO=Yv(c.o), yC=Yv(c.c);
        var top=Math.min(yO,yC), bh=Math.max(Math.abs(yO-yC),3);
        svg+='<g class="cv-candle" data-i="'+i+'">';
        svg+='<rect x="'+(cx-step/2).toFixed(1)+'" y="'+padT+'" width="'+step.toFixed(1)+'" height="'+plotH+'" fill="transparent"/>';
        svg+='<line x1="'+cx.toFixed(1)+'" y1="'+yH.toFixed(1)+'" x2="'+cx.toFixed(1)+'" y2="'+yL.toFixed(1)+'" stroke="'+col+'" stroke-width="2"/>';
        svg+='<rect x="'+(cx-bw/2).toFixed(1)+'" y="'+top.toFixed(1)+'" width="'+bw.toFixed(1)+'" height="'+bh.toFixed(1)+'" rx="2" fill="'+col+'" stroke="'+col+'" stroke-width="0.5"/>';
        svg+='</g>';
      }
    }

    // x-axis labels
    for(i=0;i<n;i++){
      if(i%labelEvery!==0 && i!==n-1) continue;
      svg+='<text x="'+Xc(i).toFixed(1)+'" y="'+(H-8)+'" text-anchor="middle" '+
           'font-size="9" font-family="'+MFONT+'" fill="'+AXTX+'">'+monthLabel(candles[i].key)+'</text>';
    }

    // Crosshair elements (hidden by default, shown on hover via JS)
    svg+='<line id="cvVLine" x1="0" y1="'+padT+'" x2="0" y2="'+(padT+plotH)+'" '+
         'stroke="rgba(255,255,255,0.25)" stroke-width="1" stroke-dasharray="3 3" style="display:none"/>';
    svg+='<line id="cvHLine" x1="'+padL+'" y1="0" x2="'+(W-padR)+'" y2="0" '+
         'stroke="rgba(255,255,255,0.18)" stroke-width="1" stroke-dasharray="3 3" style="display:none"/>';
    // price label on y-axis
    svg+='<rect id="cvYBg" x="0" y="0" width="'+padL+'" height="15" rx="3" fill="'+AMBER+'" style="display:none"/>';
    svg+='<text id="cvYTxt" x="'+(padL-4)+'" y="10" text-anchor="end" font-size="9" font-family="'+MFONT+'" font-weight="700" fill="#20160a" style="display:none"></text>';
    // month label on x-axis
    svg+='<rect id="cvXBg" x="0" y="'+(H-padB+1)+'" width="54" height="14" rx="3" fill="'+AMBER+'" style="display:none"/>';
    svg+='<text id="cvXTxt" x="0" y="'+(H-padB+10)+'" text-anchor="middle" font-size="9" font-family="'+MFONT+'" font-weight="700" fill="#20160a" style="display:none"></text>';
    // Tooltip box (OHLC for candles, value for line)
    svg+='<rect id="cvTipBg" x="60" y="6" rx="5" ry="5" fill="#1b2230" stroke="#2c3648" stroke-width="1" style="display:none"/>';
    svg+='<text id="cvTipTxt" x="68" y="18" font-size="9.5" font-family="'+MFONT+'" fill="#e6ebf2" style="display:none"></text>';
    // Highlight circle (line mode)
    svg+='<circle id="cvDot" r="5" fill="'+AMBER+'" stroke="#0b0e14" stroke-width="2" style="display:none"/>';

    // Invisible mouse capture overlay
    svg+='<rect id="cvOverlay" x="'+padL+'" y="'+padT+'" width="'+plotW+'" height="'+plotH+'" fill="transparent" style="cursor:crosshair"/>';

    svg='<svg id="chartSvg" class="chart" viewBox="0 0 '+W+' '+H+'" preserveAspectRatio="xMidYMid meet">'+svg+'</svg>';
    wrap.innerHTML=svg;

    // ---- crosshair interaction ----
    var svgEl=document.getElementById("chartSvg");
    var overlay=document.getElementById("cvOverlay");
    var vLine=document.getElementById("cvVLine");
    var hLine=document.getElementById("cvHLine");
    var yBg=document.getElementById("cvYBg");
    var yTxt=document.getElementById("cvYTxt");
    var xBg=document.getElementById("cvXBg");
    var xTxt=document.getElementById("cvXTxt");
    var tipBg=document.getElementById("cvTipBg");
    var tipTxt=document.getElementById("cvTipTxt");
    var dotEl=document.getElementById("cvDot");

    function getSVGPoint(e){
      var rect=svgEl.getBoundingClientRect();
      var clientX=e.touches?e.touches[0].clientX:e.clientX;
      var clientY=e.touches?e.touches[0].clientY:e.clientY;
      var sx=(clientX-rect.left)/rect.width*W;
      var sy=(clientY-rect.top)/rect.height*H;
      return {x:sx, y:sy};
    }

    function showCrosshair(e){
      var pt=getSVGPoint(e);
      var mx=pt.x, my=pt.y;
      if(mx<padL||mx>W-padR||my<padT||my>padT+plotH){ hideCrosshair(); return; }
      // find nearest candle index
      var idx=Math.round((mx-padL)/step - 0.5);
      idx=Math.max(0,Math.min(n-1,idx));
      var cx=Xc(idx);
      var c=candles[idx];
      var valY=isLine?Yv(c.c):my;
      var priceAtCursor=isLine?c.c:(lo + (1-(my-padT)/plotH)*range);

      // vertical line snapped to candle center
      vLine.setAttribute("x1",cx.toFixed(1)); vLine.setAttribute("x2",cx.toFixed(1));
      vLine.style.display="";
      // horizontal line at cursor y
      hLine.setAttribute("y1",valY.toFixed(1)); hLine.setAttribute("y2",valY.toFixed(1));
      hLine.style.display="";

      // y-axis price label
      var dispPrice=isLine?c.c:priceAtCursor;
      var yLabelY=(valY-6).toFixed(1);
      yBg.setAttribute("y",yLabelY); yBg.setAttribute("width",padL); yBg.style.display="";
      yTxt.setAttribute("y",(parseFloat(yLabelY)+10).toFixed(1));
      yTxt.textContent=sym+(dispPrice>=0?"":"-")+Math.abs(dispPrice).toFixed(2);
      yTxt.style.display="";

      // x-axis month label
      var tipW=56;
      xBg.setAttribute("x",(cx-tipW/2).toFixed(1)); xBg.setAttribute("width",tipW);
      xBg.style.display="";
      xTxt.setAttribute("x",cx.toFixed(1));
      xTxt.textContent=monthLong(c.key);
      xTxt.style.display="";

      // highlight dot (line mode)
      if(isLine){
        dotEl.setAttribute("cx",cx.toFixed(1));
        dotEl.setAttribute("cy",Yv(c.c).toFixed(1));
        dotEl.style.display="";
      } else { dotEl.style.display="none"; }

      // tooltip box
      var lines2, maxLen;
      if(isLine){
        lines2=[monthLong(c.key)+"  "+sym+c.c.toFixed(2)];
        maxLen=lines2[0].length;
      } else {
        lines2=[
          monthLong(c.key),
          "O "+sym+c.o.toFixed(2)+"  C "+sym+c.c.toFixed(2),
          "H "+sym+c.h.toFixed(2)+"  L "+sym+c.l.toFixed(2)
        ];
        maxLen=Math.max(lines2[0].length,lines2[1].length,lines2[2].length);
      }
      var tw=maxLen*5.8+16, th=lines2.length*14+8;
      var tx=cx+10; if(tx+tw>W-padR) tx=cx-tw-10;
      var ty=padT+4; if(isLine) ty=Math.max(padT+4,Yv(c.c)-th-8);
      tipBg.setAttribute("x",tx.toFixed(1)); tipBg.setAttribute("y",ty.toFixed(1));
      tipBg.setAttribute("width",tw.toFixed(1)); tipBg.setAttribute("height",th.toFixed(1));
      tipBg.style.display="";
      // clear old tspans
      while(tipTxt.firstChild) tipTxt.removeChild(tipTxt.firstChild);
      lines2.forEach(function(ln,li){
        var ts=document.createElementNS("http://www.w3.org/2000/svg","tspan");
        ts.setAttribute("x",(tx+8).toFixed(1));
        ts.setAttribute("dy",li===0?((ty+12).toFixed(1)):"13");
        if(li===0){ ts.setAttribute("y",(ty+12).toFixed(1)); ts.removeAttribute("dy"); }
        else ts.setAttribute("dy","13");
        ts.setAttribute("fill",li===0?AMBER:"#e6ebf2");
        ts.textContent=ln;
        tipTxt.appendChild(ts);
      });
      tipTxt.style.display="";
    }

    function hideCrosshair(){
      [vLine,hLine,yBg,yTxt,xBg,xTxt,tipBg,tipTxt,dotEl].forEach(function(el){ el.style.display="none"; });
    }

    overlay.addEventListener("mousemove", showCrosshair);
    overlay.addEventListener("touchmove", function(e){ e.preventDefault(); showCrosshair(e); }, {passive:false});
    overlay.addEventListener("mouseleave", hideCrosshair);
    overlay.addEventListener("touchend", hideCrosshair);
    // also attach to candle/dot hit areas
    svgEl.querySelectorAll(".cv-candle,.cv-dot").forEach(function(el){
      el.addEventListener("mousemove", showCrosshair);
      el.addEventListener("mouseleave", hideCrosshair);
    });
  }
