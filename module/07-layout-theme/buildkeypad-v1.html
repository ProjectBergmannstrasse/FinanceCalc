function buildKeypad(opts){
    opts=opts||{};
    var len=opts.len||PIN_LEN_DEFAULT;
    var wrap=document.createElement("div");
    wrap.className="kp";
    wrap.innerHTML=
       '<div class="kp-title"></div>'
      +'<div class="kp-sub"></div>'
      +'<div class="kp-dots"></div>'
      +'<div class="kp-err"></div>'
      +'<div class="kp-grid">'
      +   [1,2,3,4,5,6,7,8,9].map(function(n){
            return '<button type="button" class="kp-key" data-k="'+n+'">'+n+'</button>'; }).join("")
      +  '<button type="button" class="kp-key kp-ghost" data-act="cancel">Cancel</button>'
      +  '<button type="button" class="kp-key" data-k="0">0</button>'
      +  '<button type="button" class="kp-key kp-ghost" data-act="back">\u232B</button>'
      +'</div>'
      +'<div class="kp-extra"></div>';

    var titleEl=wrap.querySelector(".kp-title");
    var subEl=wrap.querySelector(".kp-sub");
    var dotsEl=wrap.querySelector(".kp-dots");
    var errEl=wrap.querySelector(".kp-err");
    var gridEl=wrap.querySelector(".kp-grid");
    var extraEl=wrap.querySelector(".kp-extra");
    var digits="", busy=false;

    titleEl.textContent=opts.title||"Enter your PIN";
    subEl.textContent=opts.sub||"";
    if(opts.extra) extraEl.innerHTML=opts.extra;

    function drawDots(){
      var n=Math.max(len, digits.length);
      if(dotsEl.children.length!==n){
        dotsEl.innerHTML="";
        for(var i=0;i<n;i++){
          var s=document.createElement("span"); s.className="kp-dot"; dotsEl.appendChild(s);
        }
      }
      Array.prototype.forEach.call(dotsEl.children, function(d,i){
        d.classList.toggle("on", i<digits.length);
      });
    }
    function shake(msg){
      errEl.textContent=msg||"";
      Array.prototype.forEach.call(dotsEl.children,function(d){ d.classList.add("shake"); });
      setTimeout(function(){
        Array.prototype.forEach.call(dotsEl.children,function(d){ d.classList.remove("shake"); });
      },420);
    }
    function say(msg, kind){
      errEl.textContent=msg||"";
      errEl.className="kp-err"+(kind?" "+kind:"");
    }
    function press(k){
      if(busy) return;
      say("");
      if(digits.length>=(opts.max||len)) return;
      digits+=k; drawDots();
      if(digits.length>=len) submit();
    }
    function back(){ if(busy) return; digits=digits.slice(0,-1); drawDots(); say(""); }
    function submit(){
      if(busy) return;
      if(!opts.validate){ opts.onDone&&opts.onDone(digits); return; }
      busy=true;
      say("Checking\u2026");
      Promise.resolve(opts.validate(digits, function(p){
        if(p<1) say("Checking\u2026 "+Math.round(p*100)+"%");
      })).then(function(ok){
        busy=false;
        if(ok){ say(""); opts.onDone&&opts.onDone(digits); return; }
        digits=""; drawDots();
        shake(opts.wrongText||"Wrong PIN \u2014 try again");
      }).catch(function(){ busy=false; digits=""; drawDots(); shake("Something went wrong"); });
    }

    gridEl.addEventListener("click", function(e){
      var b=e.target.closest("button"); if(!b) return;
      var act=b.getAttribute("data-act");
      if(act==="cancel"){ opts.onCancel&&opts.onCancel(); return; }
      if(act==="back"){ back(); return; }
      var k=b.getAttribute("data-k"); if(k) press(k);
    });
    // physical keyboard, for desktop
    var onKey=function(e){
      if(!wrap.isConnected){ document.removeEventListener("keydown", onKey); return; }
      if(/^[0-9]$/.test(e.key)){ e.preventDefault(); press(e.key); }
      else if(e.key==="Backspace"){ e.preventDefault(); back(); }
      else if(e.key==="Enter" && digits.length>=len){ e.preventDefault(); submit(); }
      else if(e.key==="Escape" && opts.onCancel){ e.preventDefault(); opts.onCancel(); }
    };
    document.addEventListener("keydown", onKey);

    drawDots();
    return {el:wrap, say:say, shake:shake, reset:function(){ digits=""; drawDots(); say(""); },
            setTitle:function(t){ titleEl.textContent=t; },
            setSub:function(t){ subEl.textContent=t; },
            extra:extraEl, dispose:function(){ document.removeEventListener("keydown", onKey); }};
  }
