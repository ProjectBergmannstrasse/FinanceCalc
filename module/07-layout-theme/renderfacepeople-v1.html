function renderFacePeople(){
    var box=document.getElementById("facePeopleList"); if(!box) return;
    var list=facePeople();
    if(!list.length){
      box.innerHTML='<div class="subhint" style="padding:4px 2px">Nobody enrolled yet.</div>';
      return;
    }
    box.innerHTML=list.map(function(p){
      var n=(p.angles&&p.angles.length)||1;
      return '<div class="fp-row"><span>\uD83D\uDC64</span><span class="fp-n">'+esc(p.name)
        +' <span class="mut" style="font-weight:400;font-size:11px">\u00b7 '+n+' angle'+(n===1?'':'s')+'</span></span>'
        +'<button data-id="'+p.id+'" title="Remove">\u2715</button></div>';
    }).join("");
    box.querySelectorAll("button").forEach(function(b){
      b.addEventListener("click", function(){
        var id=b.getAttribute("data-id");
        var who=(facePeople().filter(function(x){return x.id===id;})[0]||{}).name||"this face";
        appModal({title:"Remove "+esc(who)+"?", message:"They won't be able to unlock with their face.",
          confirmText:"Remove", cancelText:"Keep",
          onConfirm:function(){ removeFacePerson(id); renderFacePeople(); refreshSecUI(); flashSaved("removed"); }});
      });
    });
  }
