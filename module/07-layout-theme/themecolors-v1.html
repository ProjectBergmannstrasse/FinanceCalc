function themeColors(){
    var cs=getComputedStyle(document.documentElement);
    var g=function(n,f){ var v=cs.getPropertyValue(n).trim(); return v||f; };
    return {
      bg:g("--bg","#0b0e14"), panel:g("--panel","#121722"), card:g("--card","#161c28"),
      card2:g("--card2","#1b2230"), bd:g("--bd","#232c3d"),
      tx:g("--tx","#e6ebf2"), mut:g("--mut","#8b94a7"),
      up:g("--up","#3ddc84"), down:g("--down","#ff5d5d"), cool:g("--cool","#5aa9e6"),
      amber:g("--amber","#f4b73f")
    };
  }
