function showLockScreen(){
    var ov=document.createElement("div");
    ov.id="lockScreen";
    ov.style.cssText="position:fixed;inset:0;z-index:9999;background:var(--bg);display:flex;"
      +"align-items:center;justify-content:center;flex-direction:column;gap:18px;padding:24px;text-align:center";
    ov.innerHTML='<div style="font-size:44px">🔒</div>'
      +'<div style="font-size:16px;font-weight:800">My Money Tracker</div>'
      +'<div style="font-size:13px;color:var(--mut);max-width:280px;line-height:1.6" id="lockMsg">'
        +'Unlock with Face ID or your fingerprint to see your finances.</div>'
      +'<button class="btn primary" id="lockBtn" style="min-width:180px">Unlock</button>'
      +'<button class="btn ghost sm" id="lockBypass" style="display:none;font-size:11px">'
        +'Biometrics unavailable — continue anyway</button>';
    document.body.appendChild(ov);
    var attempts=0;
    var tryUnlock=function(){
      var msg=document.getElementById("lockMsg");
      msg.textContent="Waiting for verification…";
      bioVerify().then(function(ok){
        if(ok){ ov.style.opacity="0"; ov.style.transition="opacity .25s";
          setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); },260); }
        else { msg.textContent="Verification failed. Try again."; }
      }).catch(function(e){
        attempts++;
        msg.textContent="Couldn't verify: "+String(e.message||e).slice(0,70);
        // after two failures, offer the escape hatch — a locked-out user with no
        // recovery path would simply lose access to their own records
        if(attempts>=2) document.getElementById("lockBypass").style.display="";
      });
    };
    document.getElementById("lockBtn").addEventListener("click", tryUnlock);
    document.getElementById("lockBypass").addEventListener("click", function(){
      if(ov.parentNode) ov.parentNode.removeChild(ov);
    });
    tryUnlock();   // prompt immediately
  }
