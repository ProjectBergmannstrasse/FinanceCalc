function slotArt(kind){
    var eye=function(cx,cy){ return '<circle cx="'+cx+'" cy="'+cy+'" r="2.4" fill="currentColor"/>'; };
    var head='<ellipse cx="24" cy="24" rx="14" ry="17" fill="none" stroke="currentColor" stroke-width="1.6" opacity=".5"/>';
    var m={
      straight: head+eye(19,21)+eye(29,21)+'<path d="M19 30 Q24 33 29 30" stroke="currentColor" stroke-width="1.6" fill="none"/>',
      right:    head+eye(25,21)+eye(33,21)+'<path d="M25 30 Q30 33 34 30" stroke="currentColor" stroke-width="1.6" fill="none"/><path d="M12 24 l-5 0 M9 21 l-3 3 3 3" stroke="currentColor" stroke-width="1.4" fill="none" opacity=".6"/>',
      left:     head+eye(15,21)+eye(23,21)+'<path d="M14 30 Q18 33 23 30" stroke="currentColor" stroke-width="1.6" fill="none"/><path d="M36 24 l5 0 M39 21 l3 3 -3 3" stroke="currentColor" stroke-width="1.4" fill="none" opacity=".6"/>',
      down:     head+eye(19,25)+eye(29,25)+'<path d="M19 33 Q24 36 29 33" stroke="currentColor" stroke-width="1.6" fill="none"/><path d="M24 6 l0 5 M21 9 l3 3 3 -3" stroke="currentColor" stroke-width="1.4" fill="none" opacity=".6"/>',
      up:       head+eye(19,18)+eye(29,18)+'<path d="M19 27 Q24 30 29 27" stroke="currentColor" stroke-width="1.6" fill="none"/><path d="M24 42 l0 -5 M21 39 l3 -3 3 3" stroke="currentColor" stroke-width="1.4" fill="none" opacity=".6"/>'
    };
    return '<svg viewBox="0 0 48 48" class="ps-art">'+(m[kind]||m.straight)+'</svg>';
  }
