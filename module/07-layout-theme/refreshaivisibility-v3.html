function refreshAiVisibility(){
    var on=aiOnline();
    var sec=document.getElementById("aiSection");
    var sep=document.getElementById("aiSectionSep");
    var btn=document.getElementById("aiBtn");
    if(sec) sec.style.display = on ? "" : "none";
    if(sep) sep.style.display = on ? "" : "none";
    if(btn) btn.style.display = on ? "" : "none";
    if(!on) closeAiPanel();
    var pill=document.getElementById("aiStatusPill");
    if(pill){
      pill.textContent = !aiConfigured() ? "· not set up"
        : (aiCrossChecking() ? "· cross-checking on" : "· single engine");
      pill.style.color = !aiConfigured() ? "var(--mut)" : (aiCrossChecking() ? "var(--up)" : "var(--amber)");
    }
    var ch=document.getElementById("aiCrossHint");
    if(ch){
      ch.innerHTML = aiCrossChecking()
        ? "✅ Two engines set. Every read is checked by both, and you're only interrupted when they can't agree."
        : "⚠️ With one engine there's nothing to check its answer against, so mistakes can pass silently. Adding a second is what makes this trustworthy.";
      ch.style.color = aiCrossChecking() ? "var(--up)" : "var(--amber)";
    }
    var ow=document.getElementById("aiOriginWarn");
    if(ow){
      if(location.protocol==="file:"){
        ow.style.display="block";
        ow.innerHTML="⚠️ <b>This file is open straight from disk (file://).</b> Browsers send no origin for local files, and OpenAI and Claude reject those requests before your key is ever read — that's what “Load failed” means. <b>Gemini generally still works</b> because its key travels in the URL. To use the others, serve the file over http (or host it) rather than double-clicking it.";
      } else ow.style.display="none";
    }
  }
