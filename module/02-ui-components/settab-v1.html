function setTab(page){
    var btns=document.querySelectorAll("#tabbar .tab");
    Array.prototype.forEach.call(btns, function(b){
      var on=b.getAttribute("data-page")===page;
      b.classList.toggle("on", on);
      b.setAttribute("aria-selected", on?"true":"false");
      if(on) movePill(b);
    });
    lastContentPage=page; currentPage=page;
    ["pageDash","pageLedger","pageTrading","pageSettings"].forEach(function(id){
      var el=document.getElementById(id); if(!el) return;
      if(id===page){
        el.style.display="";
        // retrigger the entrance animation on every switch
        el.style.animation="none"; void el.offsetWidth; el.style.animation="";
      } else el.style.display="none";
    });
    try{ window.scrollTo(0,0); }catch(e){} // older/embedded webviews don't all have it
    if(page==="pageSettings"){ render(); fillLabelInputs(); fillPhotoQ(); fillChartMode(); fillTraderToggle(); renderLayoutUI(); renderBudgetSettings(); renderRecurList(); renderGoalSettings(); renderTrash(); renderActions(); }
    if(page==="pageTrading") renderTradingPage();
  }
  function bindTabs(){
    Array.prototype.forEach.call(document.querySelectorAll("#tabbar .tab"), function(b){
      b.addEventListener("click", function(){ setTab(b.getAttribute("data-page")); });
    });
    window.addEventListener("resize", function(){
      var on=document.querySelector("#tabbar .tab.on"); if(on) movePill(on);
    });
    setTimeout(function(){ var on=document.querySelector("#tabbar .tab.on"); if(on) movePill(on); }, 60);
  }

  /* ================= CSV IMPORT =================
     Built for real exports (MT5 statements, bank CSVs), which means: quoted fields with
     commas inside, escaped quotes, CRLF, semicolon/tab delimiters, and several date
     formats. A naive text.split(",") corrupts all of those silently. */
  function csvSplitLine(line, delim){
    var out=[], cur="", q=false;
    for(var i=0;i<line.length;i++){
      var c=line[i];
      if(q){
        if(c==='"'){
          if(line[i+1]==='"'){ cur+='"'; i++; }   // "" is a literal quote
          else q=false;
        } else cur+=c;
      } else {
        if(c==='"') q=true;
        else if(c===delim){ out.push(cur); cur=""; }
        else cur+=c;
      }
    }
    out.push(cur);
    return out.map(function(x){ return x.trim(); });
  }
  function csvDetectDelim(text){
    var head=text.split(/\r?\n/).slice(0,5).join("\n");
    var best=",", bestN=0;
    [",",";","\t","|"].forEach(function(d){
      var n=(head.split(d).length-1);
      if(n>bestN){ bestN=n; best=d; }
    });
    return best;
  }
  function csvParse(text){
    text=text.replace(/^\uFEFF/,"");                       // Excel's BOM
    var delim=csvDetectDelim(text);
    var lines=text.split(/\r?\n/).filter(function(l){ return l.trim()!==""; });
    if(!lines.length) return {rows:[], delim:delim};
    var rows=lines.map(function(l){ return csvSplitLine(l, delim); });
    // drop ragged trailing junk some exporters append
    var width=rows[0].length;
    rows=rows.filter(function(r){ return r.length>=Math.min(2,width); });
    return {rows:rows, delim:delim};
  }
