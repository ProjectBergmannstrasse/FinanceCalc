function bindTagCard(){
    var scope=document.getElementById("tagScope"), only=document.getElementById("tagOnly"),
        inst=document.getElementById("tagInst"), go=document.getElementById("tagGo");
    if(!go) return;
    scope.addEventListener("change", updateTagPreview);
    only.addEventListener("input", updateTagPreview);
    inst.addEventListener("input", updateTagPreview);
    go.addEventListener("click", function(){
      var sc=scope.value, w=only.value, sym=inst.value.trim().toUpperCase();
      var hits=tagTargets(sc, w);
      if(!hits.length) return;
      appModal({
        title:(sym ? "Tag " : "Untag ")+hits.length+" trade"+(hits.length===1?"":"s")+"?",
        message: sym
          ? "They'll all be marked <b>"+esc(sym)+"</b>, replacing whatever they say now. You can change any of them again afterwards."
          : "Their instrument will be <b>removed</b>, putting them back in the untagged group. Nothing else about them changes.",
        confirmText: sym ? "Tag them" : "Untag them",
        danger: !sym,
        onConfirm:function(){
          recordAction(sym ? ("Tagged "+hits.length+" trade"+(hits.length===1?"":"s")+" as "+sym)
                           : ("Untagged "+hits.length+" trade"+(hits.length===1?"":"s")));
          var ids={}; hits.forEach(function(t){ ids[t.id]=1; });
          state.transactions.forEach(function(t){
            if(!ids[t.id]) return;
            if(sym) t.inst=sym; else delete t.inst;
          });
          persistNow(); only.value=""; inst.value="";
          renderTradingPage(); render();
          flashSaved((sym?"tagged ":"untagged ")+hits.length);
        }
      });
    });
  }
