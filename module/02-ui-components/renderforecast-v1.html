function renderForecast(){
    var card=document.getElementById("fcCard");
    if(!card) return;
    var f=forecast(fcDays), sym=state.currency;
    // Nothing to project from: no rules and no spending history would just draw a
    // flat line and pretend it meant something.
    if(!f.events.length && f.burn.perDay<=0){ card.style.display="none"; return; }
    card.style.display="";
    var endEl=document.getElementById("fcEnd");
    endEl.textContent=fmt(f.end,sym);
    endEl.className="fc-num mono "+(f.end>=0?"up":"down");
    document.getElementById("fcCap").textContent="projected in "+fcDays+" days · "+fmt(f.start,sym)+" today";
    document.getElementById("fcChart").innerHTML=fcChartSvg(f);

    var warn=document.getElementById("fcWarn");
    if(f.zeroOn){
      var daysOut=Math.round((new Date(f.zeroOn)-new Date(todayStr()))/86400000);
      warn.innerHTML='<div class="fc-alert bad"><span>⚠️</span><span>On this path you run out around <b>'+esc(fcShortDate(f.zeroOn))+'</b> — about <b>'+daysOut+' days</b> away.</span></div>';
    } else if(f.low.bal < f.start*0.25 && f.low.bal>=0){
      warn.innerHTML='<div class="fc-alert warn"><span>⚡</span><span>Tightest point is <b>'+fmt(f.low.bal,sym)+'</b> around <b>'+esc(fcShortDate(f.low.date))+'</b>.</span></div>';
    } else warn.innerHTML="";

    var up=f.events.slice(0,5);
    document.getElementById("fcUpNext").style.display=up.length?"":"none";
    document.getElementById("fcEvents").innerHTML=up.map(function(e){
      return '<div class="fc-ev"><span class="fc-ev-d">'+esc(fcShortDate(e.date))+'</span>'
        +'<span class="fc-ev-n">'+esc(e.label)+'</span>'
        +'<span class="fc-ev-a '+(e.amount>=0?"up":"down")+'">'+(e.amount>=0?"+":"")+fmt(e.amount,sym)+'</span></div>';
    }).join("");

    var bits=[];
    if(f.burn.perDay>0) bits.push("everyday spending of about <b>"+fmt(f.burn.perDay,sym)+"/day</b> (from your last "+f.burn.days+" days)");
    if(f.events.length) bits.push("<b>"+f.events.length+"</b> scheduled item"+(f.events.length===1?"":"s"));
    document.getElementById("fcBasis").innerHTML="Based on "+bits.join(" and ")+". It's a projection from your own numbers, not a prediction — one unusual month changes it.";
  }
