function renderGoals(){
    var card=document.getElementById("goalCard");
    if(!card) return;
    var list=goalList();
    if(!list.length){ card.style.display="none"; return; }
    card.style.display="";
    var sym=state.currency;
    document.getElementById("goalList").innerHTML=list.map(function(g){
      var p=goalProgress(g);
      var cls = p.done ? "goal-done" : (p.byLate ? "goal-late" : "goal-on");
      var width=Math.round(p.pct*100);
      var right;
      if(p.done) right="reached";
      else if(p.perMonth<=0) right="not saving yet";
      else if(p.eta) right="≈ "+monthLong(p.eta);
      else right="—";
      return '<div class="goal-row '+cls+'">'
        +'<div class="goal-top"><span class="goal-n">'+esc(g.name||"Goal")+'</span>'
        +'<span class="goal-v mono">'+fmt(p.now,sym)+' / '+fmt(p.target,sym)+'</span></div>'
        +'<div class="goal-bar"><span style="width:'+width+'%"></span></div>'
        +'<div class="goal-foot"><span class="mono">'+width+'%</span>'
        +'<span class="goal-eta mono">'+esc(right)+(g.by&&!p.done?" · want "+monthLong(g.by):"")+'</span></div>'
        +(p.byLate&&!p.done&&p.eta ? '<div class="goal-warn">At '+fmt(p.perMonth,sym)+'/month you reach it around '+monthLong(p.eta)+' — after '+monthLong(g.by)+'. You\'d need '+fmt(goalNeededRate(g),sym)+'/month to make it.</div>' : "")
        +'</div>';
    }).join("");
  }
