function mtOcrUI(){
    if(typeof tradeInfoSync==="function") tradeInfoSync();   // v1.99 Stage 2
    var bar=document.getElementById("mtOcrBar");
    if(!bar) return;
    var on = mtOcrMode()!=="off" && mtOcrIsTradeType();
    bar.style.display = on?"flex":"none";
    if(!on) return;
    // one-time grey walkthrough
    var intro=document.getElementById("mtOcrIntro");
    var seen=false;try{seen=!!localStorage.getItem("mymoney_mtocr_intro");}catch(e){}
    intro.style.display=seen?"none":"block";
    var st=document.getElementById("mtOcrStatus");
    var btn=document.getElementById("mtOcrFillBtn");
    var prog=document.getElementById("mtOcrProg");
    var fill=document.getElementById("mtOcrProgFill");
    var imgs=mtOcrImages().length;
    var showProg = (mtOcr.state==="running"||mtOcr.state==="loading") && (mtOcrMode()==="auto"||mtOcr.wantFill);
    prog.style.display=showProg?"block":"none";
    if(showProg) fill.style.width=Math.round(mtOcr.progress*100)+"%";
    btn.style.display="none";
    if(!imgs){ st.textContent="⚡ Attach a statement photo to auto-read it"; return; }
    if(mtOcr.state==="loading"){ st.textContent="⚡ Getting the reader ready…"; }
    else if(mtOcr.state==="running"){
      st.textContent=(mtOcrMode()==="auto"||mtOcr.wantFill)
        ? "⚡ Reading the statement… "+Math.round(mtOcr.progress*100)+"%"
        : "⚡ Reading in the background…";
      if(mtOcrMode()==="silent"&&!mtOcr.wantFill) btn.style.display="inline-block";
    }
    else if(mtOcr.state==="done"){
      var n=mtOcr.result.trades.length, net=mtOcrNet(mtOcr.result.trades);
      st.textContent="⚡ Read "+n+" trade"+(n>1?"s":"")+", net "+(net>=0?"+":"")+net.toFixed(2);
      if(mtOcrMode()==="silent") btn.style.display="inline-block";
    }
    else if(mtOcr.state==="nota"){ st.textContent="⚡ That photo doesn't look like an MT4/MT5 statement — ignored"; }
    else if(mtOcr.state==="failed"){ st.textContent="⚡ "+(mtOcr.error||"Couldn't read the photo"); }
  }
