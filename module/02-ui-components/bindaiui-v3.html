function bindAiUI(){
    document.getElementById("aiBtn").addEventListener("click", openAiPanel);
    document.getElementById("aiPanelClose").addEventListener("click", closeAiPanel);
    document.getElementById("aiPanelOverlay").addEventListener("click", closeAiPanel);
    document.getElementById("aiHelpBtn").addEventListener("click", aiHelpModal);

    ["aiProvA","aiProvB","aiKeyA","aiKeyB","aiModelA","aiModelB"].forEach(function(id){
      // "input" not "change": change only fires on blur, so a key typed and immediately
      // tested could still be unsaved when the request goes out
      document.getElementById(id).addEventListener("input", function(){
        var prevA=aiCfg.provA, prevB=aiCfg.provB;
        aiCfg.provA=document.getElementById("aiProvA").value;
        aiCfg.provB=document.getElementById("aiProvB").value;
        aiCfg.keyA=aiCleanKey(document.getElementById("aiKeyA").value);
        aiCfg.keyB=aiCleanKey(document.getElementById("aiKeyB").value);
        aiCfg.modelA=document.getElementById("aiModelA").value.trim();
        aiCfg.modelB=document.getElementById("aiModelB").value.trim();
        // a model name is meaningless against a different provider — drop it on switch
        if(prevA!==aiCfg.provA){ aiCfg.modelA=""; aiCfg.verA=""; document.getElementById("aiModelA").value=""; }
        if(prevB!==aiCfg.provB){ aiCfg.modelB=""; aiCfg.verB=""; document.getElementById("aiModelB").value=""; }
        saveAiCfg(); refreshAiVisibility();
      });
    });
    document.getElementById("aiFindA").addEventListener("click", function(){ aiPickModels("A"); });
    document.getElementById("aiFindB").addEventListener("click", function(){ aiPickModels("B"); });
    document.getElementById("aiClearBtn").addEventListener("click", function(){
      appModal({title:"Forget API keys?", message:"Both keys are removed from this device. Your entries aren't affected.", confirmText:"Forget", danger:true, onConfirm:function(){
        aiCfg={provA:"gemini", keyA:"", modelA:"", verA:"", provB:"", keyB:"", modelB:"", verB:""};
        saveAiCfg(); fillProviderSelects(); refreshAiVisibility();
        document.getElementById("aiTestOut").textContent="";
        flashSaved("keys forgotten");
      }});
    });
    document.getElementById("aiTestBtn").addEventListener("click", async function(){
      var out=document.getElementById("aiTestOut");
      out.style.color="var(--mut)"; out.textContent="Testing…";
      var probe="Reply with exactly this JSON and nothing else: {\"transactions\":[]}";
      var msgs=[];
      var slots=[["A",aiCfg.provA,aiCfg.keyA,aiCfg.modelA,aiCfg.verA],["B",aiCfg.provB,aiCfg.keyB,aiCfg.modelB,aiCfg.verB]];
      for(var si=0; si<slots.length; si++){
        var slot=slots[si], name=slot[0], pk=slot[1], key=slot[2];
        if(!pk||!key){ msgs.push("Engine "+name+": not set"); continue; }
        var prov=AI_PROVIDERS[pk], tag="Engine "+name+" ("+prov.label+")";
        // Step 1 proves the KEY. Step 2 proves the MODEL. Collapsing them is why a dead
        // model name reads as "my key never works".
        var keyOk=false;
        if(prov.canList){
          try{ await prov.listModels(key); keyOk=true; msgs.push(tag+": ✅ key accepted"); }
          catch(e){ msgs.push(tag+": ❌ key problem — "+e.message); continue; }
        }
        try{
          await prov.run(key, probe, [], slot[3], slot[4]);
          msgs.push(tag+": ✅ model “"+(slot[3]||prov.model)+"” works");
        }catch(e2){
          msgs.push(tag+": "+(keyOk?"⚠️ key is fine, but the model failed":"❌")+" — "+e2.message);
        }
      }
      out.style.color="var(--mut)";
      out.innerHTML=msgs.map(esc).join("<br>");
    });

    document.getElementById("aiAddImgs").addEventListener("click", function(){ document.getElementById("aiFiles").click(); });
    document.getElementById("aiClearImgs").addEventListener("click", function(){ aiImages=[]; renderAiThumbs(); });
    document.getElementById("aiFiles").addEventListener("change", function(e){
      var files=Array.prototype.slice.call(e.target.files||[]);
      var left=files.length;
      if(!left) return;
      files.forEach(function(f){
        var r=new FileReader();
        r.onload=function(){
          var url=r.result;
          var comma=url.indexOf(",");
          aiImages.push({mime:f.type||"image/jpeg", b64:url.slice(comma+1), url:url});
          if(--left<=0) renderAiThumbs();
        };
        r.onerror=function(){ if(--left<=0) renderAiThumbs(); };
        r.readAsDataURL(f);
      });
      e.target.value=""; // so picking the same file twice still fires
    });

    document.getElementById("aiRunBtn").addEventListener("click", async function(){
      if(aiBusy || !aiImages.length) return;
      aiBusy=true;
      var btn=this; btn.disabled=true; btn.textContent="Reading…";
      var log=document.getElementById("aiLog"); log.innerHTML=""; log.style.display="block";
      document.getElementById("aiResults").style.display="none";
      var expected=parseInt(document.getElementById("aiExpected").value,10);
      if(!isFinite(expected) || expected<1) expected=0;
      try{
        var res=await aiExtract(aiImages, expected, aiLogLine);
        if(expected>0 && res.list.length!==expected){
          aiLogLine("You said "+expected+" but "+res.list.length+" came back — check the list.","warn");
        }
        renderAiResults(res);
      }catch(e){
        aiLogLine(e.message,"err");
        document.getElementById("aiResults").style.display="block";
        document.getElementById("aiResults").innerHTML='<div class="ai-banner bad"><b>Couldn\'t read the images.</b><br>'+esc(e.message)+'</div>';
      }
      aiBusy=false; renderAiThumbs();
    });

    window.addEventListener("online", refreshAiVisibility);
    window.addEventListener("offline", refreshAiVisibility);
  }
