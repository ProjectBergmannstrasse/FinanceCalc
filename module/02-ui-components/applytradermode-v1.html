function applyTraderMode(){
    var on=isTrader();
    var tab=document.querySelector('#tabbar .tab[data-page="pageTrading"]');
    if(tab) tab.style.display = on ? "" : "none";
    var tb=document.getElementById("tabbar");
    if(tb) tb.setAttribute("data-tabs", on ? "4" : "3");
    // Settings' bank/trading section is trading-only. The instrument field is NOT listed
    // here on purpose: typeUI() already shows it only for trade types, and a non-trader
    // can't select one. (This used to name "fInstWrap", an id that doesn't exist, so it
    // silently did nothing.)
    var sw=document.getElementById("sepTradingWrap");
    if(sw) sw.style.display = on ? "" : "none";
    var sel=document.getElementById("fType");
    if(sel) Array.prototype.forEach.call(sel.querySelectorAll("option"), function(o){
      if(/^(trade_win|trade_loss|to_trading|from_trading|alw_trading)$/.test(o.value)) o.hidden = !on;
    });
    // if they've turned trading off while sitting on the Trading tab, move them somewhere real
    if(!on && document.getElementById("pageTrading") && document.getElementById("pageTrading").classList.contains("on")) setTab("pageDash");
  }
