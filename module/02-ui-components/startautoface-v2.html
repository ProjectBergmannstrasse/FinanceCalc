function startAutoFace(ovEl, onMatch, setStatus){
    if(!autoFaceOn()) return;
    if(!faceCfg.on || !facePeople().length) return;
    if(cameraBlocked()) return;
    if(inFrame() && !window.faceapi) return;      // model can't load in a frame
    if(!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) return;

    cameraAlreadyGranted().then(function(granted){
      // don't ambush someone with a permission prompt at launch — only run
      // automatically once they've already allowed it for this site
      if(!granted) return;
      setStatus && setStatus("Looking for you\u2026");
      var video=document.createElement("video");
      video.autoplay=true; video.playsInline=true; video.muted=true;
      video.style.cssText="position:absolute;width:1px;height:1px;opacity:0;pointer-events:none";
      ovEl.appendChild(video);

      navigator.mediaDevices.getUserMedia({video:{facingMode:"user",width:320,height:240}})
        .then(function(stream){
          _autoFace.stream=stream; video.srcObject=stream;
          return loadFaceApi();
        })
        .then(function(){
          var tries=0;
          var tick=function(){
            if(!_autoFace.stream || !document.getElementById("lockScreen")) { _autoFace.stop(); return; }
            faceapi.detectSingleFace(video, new faceapi.TinyFaceDetectorOptions())
              .withFaceLandmarks(true).withFaceDescriptor()
              .then(function(det){
                if(!_autoFace.stream) return;
                if(det){
                  var who=matchFace(Array.from(det.descriptor));
                  if(who){
                    _autoFace.stop();
                    setStatus && setStatus("Welcome back, "+who.name);
                    setTimeout(function(){ onMatch(who); }, 350);
                    return;
                  }
                  // not a match — say nothing accusatory, just keep offering the PIN
                  if(++tries===6) setStatus && setStatus("Not recognised \u2014 use your PIN");
                }
                if(tries<40) _autoFace.raf=setTimeout(tick, 500);
                else { _autoFace.stop(); setStatus && setStatus("Enter your PIN"); }
              })
              .catch(function(){ _autoFace.raf=setTimeout(tick, 900); });
          };
          tick();
        })
        .catch(function(){ _autoFace.stop(); setStatus && setStatus("Enter your PIN"); });
    });
  }
  function showLockScreen(){
    if(!lockIsOn()) return;
    var ov=document.createElement("div");
    ov.id="lockScreen";
    ov.innerHTML='<div class="lk-box">'
      +'<div class="lk-icon">\uD83D\uDD12</div>'
      +'<div id="lkPadHost"></div>'
      +'</div>';
    document.body.appendChild(ov);
    document.body.style.overflow="hidden";

    function unlock(){
      _autoFace.stop();
      if(pad) pad.dispose();
      setTimeout(showAttemptBanner, 700);
      ov.style.opacity="0";
      document.body.style.overflow="";
      document.removeEventListener("visibilitychange", onHide);
      setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); }, 220);
    }

    // extras below the pad: face unlock and the recovery route
    var extra="";
    if(faceCfg.on && facePeople().length)
      extra+='<button class="btn ghost sm" id="lkBio">\uD83D\uDE00 Use face unlock</button>';
    if(recoveryOn())
      extra+='<button class="kp-link" id="lkForgot">Forgot your PIN?</button>';

    var pad=buildKeypad({
      title:"My Money",
      sub:"Enter your PIN",
      len:4, max:12,
      wrongText:"Wrong PIN \u2014 try again",
      extra:extra,
      validate:function(d, prog){
        var cd=lockCooldownLeft();
        if(cd>0) return Promise.resolve(false);
        return new Promise(function(res){
          verifyPin(d, function(ok){
            if(!ok){
              recordFailedAttempt("pin");
              var left=lockCooldownLeft();
              if(left>0) setTimeout(function(){ pad.say("Too many tries \u2014 wait "+left+"s"); }, 460);
            }
            res(ok);
          }, prog);
        });
      },
      onDone:unlock
    });
    document.getElementById("lkPadHost").appendChild(pad.el);

    var bio=pad.el.querySelector("#lkBio");
    if(bio) bio.addEventListener("click", function(){
      if(inFrame() && !window.faceapi){
        pad.say("Face unlock needs the app in its own tab"); return;
      }
      pad.say("Opening the camera\u2026");
      bio.disabled=true;
      var answered=false;
      var settle=function(ok){
        if(answered) return; answered=true; bio.disabled=false;
        if(ok) unlock();
        else { recordFailedAttempt("face"); pad.say("Face didn\u2019t match \u2014 use your PIN"); }
      };
      setTimeout(function(){
        if(answered) return; answered=true; bio.disabled=false;
        var sh=document.getElementById("faceOv");
        if(sh && sh.parentNode) sh.parentNode.removeChild(sh);
        pad.say("Camera didn\u2019t start \u2014 use your PIN");
      }, 15000);
      verifyFace(settle);
    });
    var fg=pad.el.querySelector("#lkForgot");
    if(fg) fg.addEventListener("click", function(){ openRecoveryUnlock(function(){ unlock(); }); });

    // look for the owner automatically, only while this screen is up
    startAutoFace(ov, unlock, function(msg){ pad.say(msg, /Welcome/.test(msg)?"ok":""); });
    var onHide=function(){ if(document.hidden) _autoFace.stop(); };
    document.addEventListener("visibilitychange", onHide);
  }

  /* ============================================================
     PWA INSTALL — single-file friendly
     A normal PWA needs manifest.json + sw.js as separate files,
     which would break the "one file you double-click" promise.
     Both are generated at runtime as blob URLs instead, so the
     app can be installed to the home screen and still ships as
     exactly one file.
     ============================================================ */
  var _deferredPrompt=null;
  function initPWA(){
    // manifest, built in memory
    try{
      var manifest={
        name:"My Money Tracker", short_name:"My Money",
        start_url:location.href, display:"standalone",
        background_color:"#0b0e14", theme_color:"#0b0e14",
        icons:[{src:pwaIcon(), sizes:"512x512", type:"image/png", purpose:"any maskable"}]
      };
      var mUrl=URL.createObjectURL(new Blob([JSON.stringify(manifest)],{type:"application/json"}));
      var link=document.createElement("link");
      link.rel="manifest"; link.href=mUrl;
      document.head.appendChild(link);
    }catch(e){}

    // service worker — only registrable over http(s), not file://
    if("serviceWorker" in navigator && location.protocol.indexOf("http")===0){
      try{
        var swSrc=[
          "const C='mymoney-v1';",
          "self.addEventListener('install',e=>{self.skipWaiting();});",
          "self.addEventListener('activate',e=>{e.waitUntil(self.clients.claim());});",
          // cache-first for the app shell so it opens with no connection at all
          "self.addEventListener('fetch',e=>{",
          "  if(e.request.method!=='GET') return;",
          "  e.respondWith(caches.open(C).then(c=>c.match(e.request).then(hit=>{",
          "    const net=fetch(e.request).then(r=>{ if(r&&r.status===200) c.put(e.request,r.clone()); return r; })",
          "      .catch(()=>hit);",
          "    return hit||net;",
          "  })));",
          "});"
        ].join("\n");
        var swUrl=URL.createObjectURL(new Blob([swSrc],{type:"application/javascript"}));
        navigator.serviceWorker.register(swUrl).catch(function(){});
      }catch(e){}
    }

    window.addEventListener("beforeinstallprompt", function(e){
      e.preventDefault(); _deferredPrompt=e;
      var b=document.getElementById("pwaInstallBtn");
      if(b){ b.style.display=""; b.disabled=false; }
    });
    window.addEventListener("appinstalled", function(){
      var b=document.getElementById("pwaInstallBtn");
      if(b){ b.textContent="✅ Installed"; b.disabled=true; }
    });
  }
  function pwaIcon(){
    // a plain generated icon so we don't ship a binary asset
    var c=document.createElement("canvas"); c.width=c.height=512;
    var x=c.getContext("2d");
    x.fillStyle="#0b0e14"; x.fillRect(0,0,512,512);
    x.fillStyle="#3ddc84"; x.font="bold 300px ui-monospace,monospace";
    x.textAlign="center"; x.textBaseline="middle"; x.fillText("$",256,270);
    return c.toDataURL("image/png");
  }
