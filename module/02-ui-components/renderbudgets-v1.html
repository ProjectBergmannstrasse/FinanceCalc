function renderBudgets(){
    var card=document.getElementById("budgetCard");
    if(!card) return;
    var rows=budgetRows(), sym=state.currency;
    if(!rows.length){ card.style.display="none"; return; }
    card.style.display="";
    document.getElementById("budgetMonth").textContent=monthLong(monthKey(todayStr()));
    document.getElementById("budgetList").innerHTML=rows.map(function(r){
      var state_= r.pct>=1 ? "over" : (r.pct>=0.8 ? "near" : "ok");
      var width=Math.min(100, Math.round(r.pct*100));
      var right = r.left>=0
        ? fmt(r.left,sym)+" left"
        : fmt(Math.abs(r.left),sym)+" over";
      return '<div class="bg-row bg-'+state_+'">'
        +'<div class="bg-top"><span class="bg-cat">'+esc(r.cat)+'</span>'
        +'<span class="bg-nums mono">'+fmt(r.spent,sym)+' / '+fmt(r.cap,sym)+'</span></div>'
        +'<div class="bg-bar"><span style="width:'+width+'%"></span></div>'
        +'<div class="bg-foot"><span class="bg-pct mono">'+Math.round(r.pct*100)+'%</span>'
        +'<span class="bg-left mono">'+right+'</span></div>'
        +'</div>';
    }).join("");
  }
