function renderAiResults(res){
    var box=document.getElementById("aiResults");
    box.style.display="block";
    var list=res.list||[];
    if(!list.length){
      box.innerHTML='<div class="ai-banner warn">Nothing financial found in those images.</div>';
      return;
    }
    var banner;
    if(res.conflict){
      var d=res.diff||{onlyA:[],onlyB:[]};
      banner='<div class="ai-banner bad"><b>⚠️ Needs your eyes — do not just accept this.</b><br>'+esc(res.reason)
        +'<br><br>The two engines read these images differently, so at least one is wrong. Check every row against the original image before adding.'
        +'<br><br><b>Only engine A said:</b><br>'+(d.onlyA.map(esc).join("<br>")||"—")
        +'<br><br><b>Only engine B said:</b><br>'+(d.onlyB.map(esc).join("<br>")||"—")
        +'</div>';
    } else if(res.verified){
      banner='<div class="ai-banner good">✅ Both engines read these identically'
        +(res.attempts>1?" (agreed on attempt "+res.attempts+")":"")+'. Safe to add.</div>';
    } else {
      banner='<div class="ai-banner warn"><b>Not cross-checked.</b> '+esc(res.reason||"")+' Worth a quick look before adding.</div>';
    }
    var rows=list.map(function(t,i){
      var meta=TYPE_META[t.type]||{label:t.type};
      var low=t.confidence<0.6;
      return '<div class="ai-row'+(low?" low":"")+'">'
        +'<input type="checkbox" class="ai-pick" data-i="'+i+'" checked>'
        +'<div class="r-main"><div><span class="badge">'+esc(meta.label)+'</span> '+esc(t.date)
        +(low?' <span style="color:var(--amber)">· unsure</span>':'')+'</div>'
        +'<div class="r-note">'+esc(t.note||t.cat||"")+'</div></div>'
        +'<div class="r-amt">'+fmt(t.amount)+'</div>'
        +'</div>';
    }).join("");
    box.innerHTML=banner+'<div class="ai-rows">'+rows+'</div>'
      +'<button class="btn primary sm" id="aiAddSel" style="width:100%;margin-top:6px">Add selected entries</button>';

    document.getElementById("aiAddSel").addEventListener("click", function(){
      var picks=Array.prototype.filter.call(box.querySelectorAll(".ai-pick"), function(c){ return c.checked; })
                     .map(function(c){ return list[+c.getAttribute("data-i")]; });
      if(!picks.length) return;
      picks.forEach(function(t){
        var tx={id:freshIdBase(), date:t.date, type:t.type, amount:t.amount};
        if(t.type==="spend" && t.cat) tx.cat=t.cat;
        if(t.note) tx.note=t.note;
        state.transactions.push(tx);
        mirrorNewTransaction(tx);
      });
      state.selectedMonths=null;
      persistNow(); render();
      flashSaved("added "+picks.length);
      closeAiPanel();
      aiImages=[]; renderAiThumbs();
      box.style.display="none"; box.innerHTML="";
      document.getElementById("aiLog").style.display="none";
      document.getElementById("aiLog").innerHTML="";
    });
  }
