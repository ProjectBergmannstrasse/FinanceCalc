function openCopyMenu(t, anchor){
    var hasPhoto=photosOf(t).length>0;
    var ov=document.createElement("div");
    ov.className="xsel-ov copy-ov";
    ov.innerHTML='<div class="copy-menu">'
      +'<div class="copy-mh">Copy transaction</div>'
      +'<button type="button" class="copy-opt" data-k="text">Copy as text'
        +'<span class="copy-sub">note + amount</span></button>'
      +'<button type="button" class="copy-opt" data-k="img">Copy as image'
        +'<span class="copy-sub">the card, exactly as shown</span></button>'
      +(hasPhoto?'<button type="button" class="copy-opt" data-k="imgphoto">Copy as image + photo'
        +'<span class="copy-sub">card with the receipt below</span></button>':'')
      +'<button type="button" class="copy-opt copy-cancel" data-k="cancel">Cancel</button>'
      +'</div>';
    document.body.appendChild(ov);
    requestAnimationFrame(function(){ ov.classList.add("in"); });
    var close=function(){ ov.classList.remove("in");
      document.removeEventListener("keydown",onKey);
      setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); },140); };
    var onKey=function(e){ if(e.key==="Escape"){ e.preventDefault(); close(); } };
    document.addEventListener("keydown",onKey);
    ov.addEventListener("click",function(e){ if(e.target===ov) close(); });
    ov.querySelectorAll(".copy-opt").forEach(function(b){
      b.addEventListener("click", function(){
        var k=b.getAttribute("data-k");
        if(k==="text"){ copyText(txAsText(t), "text copied"); close(); }
        else if(k==="img"){ flashSaved("rendering…"); renderTxImage(t,false,function(blob){ copyImageBlob(blob); close(); }); }
        else if(k==="imgphoto"){ flashSaved("rendering…"); renderTxImage(t,true,function(blob){ copyImageBlob(blob); close(); }); }
        else close();
      });
    });
  }
