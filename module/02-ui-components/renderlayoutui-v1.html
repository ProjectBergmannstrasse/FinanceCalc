function renderLayoutUI(){
    var box=document.getElementById("layoutList");
    if(!box) return;
    var cfg=layoutCfg(), order=layoutOrder();

    var presets='<div class="ly-presets">'+LAYOUT_PRESETS.map(function(p){
      var on=presetMatches(p);
      return '<button type="button" class="ly-preset'+(on?" on":"")+'" data-p="'+p.k+'">'
        +'<span class="ly-pv">'+LAYOUT_CARDS.map(function(c){
            return '<span class="ly-pv-c ly-pv-'+(p.cfg[c.id]||"off")+'"></span>';
          }).join("")+'</span>'
        +'<span class="ly-pt">'+esc(p.t)+'</span><span class="ly-pd">'+esc(p.d)+'</span>'
        +(on?'<span class="ly-pon">✓</span>':'')+'</button>';
    }).join("")+'</div>';

    var board=LAYOUT_PAGES.map(function(pg){
      var mine=order.filter(function(id){ return cfg[id]===pg.id; });
      var cards=mine.map(function(id,i){
        var c=LAYOUT_CARDS.filter(function(x){ return x.id===id; })[0];
        var picked=(lyPicked===id);
        return '<div class="ly-card'+(picked?" picked":"")+'" data-id="'+id+'" role="button" tabindex="0">'
          +layoutMiniHtml(id)
          +'<span class="ly-cn">'+esc(c.name)+'</span>'
          +'<span class="ly-ord">'
            +'<button type="button" class="ly-mv" data-mv="'+id+'" data-dir="-1"'+(i===0?" disabled":"")+' aria-label="Up">▲</button>'
            +'<button type="button" class="ly-mv" data-mv="'+id+'" data-dir="1"'+(i===mine.length-1?" disabled":"")+' aria-label="Down">▼</button>'
          +'</span></div>';
      }).join("");
      var empty = mine.length ? "" : '<div class="ly-empty">'+(lyPicked?"Tap to put it here":"nothing here")+'</div>';
      return '<div class="ly-zone'+(lyPicked?" live":"")+'" data-page="'+pg.id+'">'
        +'<div class="ly-zh">'+esc(pg.label)+'<span class="ly-zn">'+mine.length+'</span></div>'
        +'<div class="ly-cards">'+cards+empty+'</div></div>';
    }).join("");

    var help = lyPicked
      ? '<div class="ly-help live">Now tap a section to move it there.</div>'
      : '<div class="ly-help">Tap a box, then tap where you want it. ▲▼ reorders.</div>';
    box.innerHTML = presets + help + '<div class="ly-board">'+board+'</div>';

    Array.prototype.forEach.call(box.querySelectorAll(".ly-preset"), function(b){
      b.addEventListener("click", function(){ applyPreset(b.getAttribute("data-p")); });
    });
    Array.prototype.forEach.call(box.querySelectorAll(".ly-mv"), function(b){
      b.addEventListener("click", function(e){
        e.stopPropagation();   // the arrows sit inside the tappable box
        moveLayout(b.getAttribute("data-mv"), +b.getAttribute("data-dir"));
      });
    });
    Array.prototype.forEach.call(box.querySelectorAll(".ly-card"), function(c){
      c.addEventListener("click", function(e){ e.stopPropagation(); lyTap(c.getAttribute("data-id")); });
      c.addEventListener("keydown", function(e){
        if(e.key==="Enter"||e.key===" "){ e.preventDefault(); lyTap(c.getAttribute("data-id")); }
      });
    });
    Array.prototype.forEach.call(box.querySelectorAll(".ly-zone"), function(z){
      z.addEventListener("click", function(){ lyDrop(z.getAttribute("data-page")); });
    });
  }
