function renderPlan(list, sym){
    var card=document.getElementById("planCard");
    if(!card) return;
    var plan=tradingPlan(list);
    if(plan.thin){
      if(!list.length){ card.style.display="none"; return; }
      card.style.display="";
      document.getElementById("planSub").textContent="";
      document.getElementById("planToday").innerHTML="";
      document.getElementById("planRules").innerHTML='<div class="fc-alert warn"><span>◔</span><span>A plan built on <b>'+plan.n+'</b> trade'+(plan.n===1?"":"s")+' would just be a plan built on luck. At <b>'+PLAN_MIN_TRADES+'</b> this fills in with rules from your own record.</span></div>';
      document.getElementById("planFoot").textContent="";
      return;
    }
    card.style.display="";
    document.getElementById("planSub").textContent="from your "+plan.n+" trades";

    // today first — it's the only part that's actionable right now
    var st=planStatus(list, plan);
    var box=document.getElementById("planToday");
    if(st){
      var worst = [st.trades,st.loss,st.streak].reduce(function(a,m){
        return (m.state==="over") ? "over" : (m.state==="near" && a!=="over") ? "near" : a; },"ok");
      var msg = worst==="over" ? "You are past a limit you set from your own record."
              : worst==="near" ? "Getting close to a limit."
              : "Inside your plan.";
      box.className="pl-today "+worst;
      box.innerHTML='<div class="pl-th">Today · '+esc(msg)+'</div>'
        +planMeter("Trades", st.trades, sym, false)
        +planMeter("Lost today", st.loss, sym, true)
        +planMeter("Losses in a row", st.streak, sym, false);
    } else box.innerHTML="";

    var rules=[];
    rules.push({ic:"◇", t:"Risk no more than <b>"+fmt(plan.riskCap,sym)+"</b> on one trade",
      w:"Your average loss is <b>"+fmt(plan.stats.avgLoss,sym)+"</b>. Your worst was <b>"+fmt(plan.worstLoss,sym)+"</b> — "+Math.round((plan.worstLoss/(plan.stats.grossLoss||1))*100)+"% of everything you've lost, in one trade."});
    rules.push({ic:"◇", t:"Stop for the day at <b>"+fmt(plan.dailyStop,sym)+"</b> down",
      w:"Two average losses. Your worst single day was <b>"+fmt(plan.worstDay,sym)+"</b>."});
    if(plan.stopAfter>=2 && plan.stopSaves>0){
      rules.push({ic:"◇", t:"After <b>"+plan.stopAfter+"</b> losses in a row, stop for the day",
        w:"Replayed on your record this skips <b>"+plan.stopBlocked+"</b> trades and saves <b>"+fmt(plan.stopSaves,sym)+"</b> — missed wins already subtracted."});
    }
    rules.push({ic:"◇", t:"Around <b>"+plan.maxPerDay+"</b> trade"+(plan.maxPerDay===1?"":"s")+" a day",
      w:"That's your median. Your busiest day was <b>"+plan.busiest+"</b>."});
    if(plan.focus) rules.push({ic:"◆", t:"<b>"+esc(plan.focus.inst)+"</b> is where your edge is",
      w:"<b>"+fmt(plan.focus.stats.expectancy,sym)+"</b> per trade over "+plan.focus.stats.n+" trades."});
    if(plan.avoid) rules.push({ic:"◆", t:"<b>"+esc(plan.avoid.inst)+"</b> is where it leaks",
      w:"<b>"+fmt(plan.avoid.stats.expectancy,sym)+"</b> per trade over "+plan.avoid.stats.n+" trades."});

    document.getElementById("planRules").innerHTML=rules.map(function(r){
      return '<div class="pl-rule"><span class="pl-ic">'+r.ic+'</span>'
        +'<span class="pl-body"><span class="pl-t">'+r.t+'</span><span class="pl-w">'+r.w+'</span></span></div>';
    }).join("");
    document.getElementById("planFoot").textContent="Every rule is your own record played back — the numbers are shown so you can argue with them. It sets limits on how much and how often; it never says what to trade.";
  }
