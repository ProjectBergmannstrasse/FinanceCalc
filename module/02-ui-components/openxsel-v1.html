function openXsel(sel, btn){
    var opts=xselVisible(sel);
    if(!opts.length) return;
    var ov=document.createElement("div");
    ov.className="xsel-ov";
    ov.innerHTML='<div class="xsel-sheet" role="listbox">'
      +'<div class="xsel-h">'+esc(xselLabel(sel))+'</div>'
      +'<div class="xsel-list">'
      +opts.map(function(o){
          var on=(o.value===sel.value);
          return '<button type="button" class="xsel-o'+(on?" on":"")+'" data-v="'+esc(o.value)+'" role="option" aria-selected="'+on+'">'
            +'<span class="xsel-otx">'+esc(o.textContent)+'</span>'
            +(on?'<span class="xsel-tick">✓</span>':'')+'</button>';
        }).join("")
      +'</div></div>';
    document.body.appendChild(ov);
    requestAnimationFrame(function(){ ov.classList.add("in"); });
    var close=function(){
      ov.classList.remove("in");
      document.removeEventListener("keydown", onKey);
      setTimeout(function(){ if(ov.parentNode) ov.parentNode.removeChild(ov); }, 140);
      if(btn) btn.focus();
    };
    var onKey=function(e){ if(e.key==="Escape"){ e.preventDefault(); close(); } };
    document.addEventListener("keydown", onKey);
    ov.addEventListener("click", function(e){ if(e.target===ov) close(); });
    Array.prototype.forEach.call(ov.querySelectorAll(".xsel-o"), function(b){
      b.addEventListener("click", function(){
        var v=b.getAttribute("data-v");
        if(v!==sel.value){
          sel.value=v;
          sel.dispatchEvent(new Event("change",{bubbles:true}));  // whatever was bound still runs
        } else if(sel._xsync) sel._xsync();
        close();
      });
    });
    var cur=ov.querySelector(".xsel-o.on") || ov.querySelector(".xsel-o");
    if(cur) cur.focus();
  }
