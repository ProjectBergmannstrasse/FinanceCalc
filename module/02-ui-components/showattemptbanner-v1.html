function showAttemptBanner(){
    var list=unseenAttempts();
    if(!list.length) return;
    var host=document.createElement("div");
    host.className="ov in"; host.id="attemptOv";
    host.innerHTML=
       '<div class="sheet" style="max-width:420px">'
      +'  <div class="sheet-hd"><h3>⚠️ Failed unlock attempts</h3>'
      +'    <button class="x-btn" id="atX">✕</button></div>'
      +'  <div class="sheet-bd">'
      +'    <div class="rq-warn">There '+(list.length===1?"was <b>1</b> failed attempt":"were <b>"+list.length+"</b> failed attempts")
      +'      to unlock this app since you were last in.</div>'
      +'    <div class="rc-sec">When</div>'
      +      list.slice(0,12).map(function(a){
              return '<div class="rc-card"><div class="rc-ic">⛔</div>'
                +'<div class="rc-main"><div class="rc-t">'+esc(a.kind==="face"?"Face didn\u2019t match":"Wrong PIN")+'</div>'
                +'<div class="rc-s">'+esc(fmtLogTime(a.at))+' · '+agoText(a.at)+'</div></div></div>';
            }).join("")
      +'    <div class="rc-secd" style="margin-top:12px">If this was you, nothing to do. If it wasn’t, '
      +'      consider changing your PIN and exporting a backup.</div>'
      +'  </div>'
      +'  <div class="sheet-ft"><button class="btn ghost sm" id="atClear">Clear log</button>'
      +'    <button class="btn primary sm" id="atOk">Got it</button></div>'
      +'</div>';
    document.body.appendChild(host);
    function close(){ markAttemptsSeen(); if(host.parentNode) host.parentNode.removeChild(host); }
    host.querySelector("#atX").onclick=close;
    host.querySelector("#atOk").onclick=close;
    host.querySelector("#atClear").onclick=function(){
      attempts=[]; saveAttempts(); if(host.parentNode) host.parentNode.removeChild(host);
      flashSaved("attempt log cleared");
    };
  }
