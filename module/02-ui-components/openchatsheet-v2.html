function openChatSheet(){
    var host=document.getElementById("chatSheetHost");
    if(!host){ host=document.createElement("div"); host.id="chatSheetHost"; document.body.appendChild(host); }
    host.innerHTML=
       '<div class="ov in" id="chatOv">'
      +'  <div class="sheet chat-sheet">'
      +'    <div class="chat-hd">'
      +'      <h3>Ask AI</h3>'
      +'      <span class="chat-engine" id="chatEngine">checking…</span>'
      +'      <span style="flex:1"></span>'
      +'      <button class="x-btn" id="chatClear" title="Clear this conversation">\u21bb</button>'
      +'      <button class="x-btn" id="chatSheetX" title="Close">\u2715</button>'
      +'    </div>'
      +'    <div class="chat-body" id="chatLog"></div>'
      +'    <div class="chat-foot">'
      +'      <div class="chat-pending" id="chatFileStrip"></div>'
      +'      <div class="chat-input-row">'
      +'        <button class="chat-icon-btn" id="chatAttachBtn" title="Attach a file">\uD83D\uDCCE</button>'
      +'        <textarea id="chatInput" class="inp" rows="1" placeholder="Ask about your money\u2026"></textarea>'
      +'        <button class="chat-icon-btn chat-send" id="chatSend" title="Send" disabled>\u2191</button>'
      +'      </div>'
      +'      <div class="chat-hint">Runs on your device \u00b7 Enter to send, Shift+Enter for a new line</div>'
      +'    </div>'
      +'    <input type="file" id="chatFileInput" multiple style="display:none"'
      +'      accept=".pdf,.docx,.txt,.csv,.xlsx,.json,.html,.css,.js,.md,.log,.xml,.jpg,.jpeg,.png,.gif,.webp,.mp3,.wav,.mp4,.mov,.zip,image/*,text/*">'
      +'  </div></div>';
    bindChat();
    renderChatLog(); renderChatFiles();

    var close=function(){ host.innerHTML=""; document.removeEventListener("keydown", onEsc); };
    var onEsc=function(e){ if(e.key==="Escape") close(); };
    document.getElementById("chatSheetX").onclick=close;
    document.getElementById("chatOv").addEventListener("click", function(e){
      if(e.target.id==="chatOv") close();
    });
    document.addEventListener("keydown", onEsc);

    // grow the box as they type, up to a limit
    var ta=document.getElementById("chatInput");
    var send=document.getElementById("chatSend");
    var sync=function(){
      ta.style.height="auto";
      ta.style.height=Math.min(130, ta.scrollHeight)+"px";
      send.disabled = !ta.value.trim() && !chatFiles.length;
    };
    ta.addEventListener("input", sync);
    setTimeout(function(){ try{ ta.focus(); }catch(e){} sync(); },120);
    window._chatSync=sync;

    refreshChatEngine();
  }
